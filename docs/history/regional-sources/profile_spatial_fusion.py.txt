import gc
import hashlib
import json
import os
import shutil
import time
from pathlib import Path
from typing import Literal

import numpy as np
import torch
from pydantic import Field
from pydantic_settings import CliPositionalArg

import knee_abnormality.dataset as dataset_module
import knee_abnormality.model as model_module
from knee_abnormality.cli import CLISettings
from knee_abnormality.config import load_config
from knee_abnormality.dataset import build_data_loader, collate_studies
from knee_abnormality.profiling import sample_dataset, training_step
from knee_abnormality.runtime import configure_cuda_allocator, move_batch
from knee_abnormality.training import build_optimizer


class ProfileCLI(CLISettings):
    config: CliPositionalArg[Path]
    output_dir: Path
    mode: Literal["train", "inference"]
    samples: int = Field(default=16, ge=8)
    batch_size: int = Field(default=8, ge=1)
    image_multiplier: int = Field(default=1, ge=1)
    warmup: int = Field(default=3, ge=1)
    steps: int = Field(default=12, ge=1)
    workers: int = Field(default=16, ge=0)


def main() -> None:
    args = ProfileCLI()
    config = load_config(args.config)
    configure_cuda_allocator()
    os.environ["CUBLAS_WORKSPACE_CONFIG"] = ":4096:8"
    torch.set_num_threads(1)
    torch.set_float32_matmul_precision("high")
    torch.manual_seed(config.seed)
    torch.use_deterministic_algorithms(config.trainer.deterministic)
    torch.backends.cudnn.benchmark = False
    args.output_dir.mkdir(parents=True, exist_ok=False)
    shutil.copy2(__file__, args.output_dir / "profile_spatial_fusion.py")
    source = Path(model_module.__file__).parent
    shutil.copytree(
        source, args.output_dir / "source", ignore=shutil.ignore_patterns("__pycache__")
    )
    dataset, labels = sample_dataset(config, args.samples)
    dataset.image_limit_multiplier = args.image_multiplier
    dataset.training = args.mode == "train"
    selections = []
    original = dataset_module.select_slices

    def record(*values, **kwargs):
        selected = original(*values, **kwargs)
        selections.append([str(value.path) for value in selected])
        return selected

    dataset_module.select_slices = record
    loading = []
    manifests = []
    items = []
    for index in range(len(dataset)):
        started = time.perf_counter()
        item = dataset[index]
        loading.append(time.perf_counter() - started)
        items.append(item)
        manifests.append(
            {
                "study_uid": item["study_uid"],
                "series": [
                    {
                        "uid": value.uid,
                        "pixels_sha256": hashlib.sha256(
                            value.images.numpy().tobytes()
                        ).hexdigest(),
                        "positions": value.positions.tolist(),
                        "shape": list(value.images.shape),
                    }
                    for acquisitions in item["series"]
                    for value in acquisitions
                ],
            }
        )
    dataset_module.select_slices = original
    (args.output_dir / "inputs.json").write_text(
        json.dumps({"studies": manifests, "selected": selections}, indent=2) + "\n"
    )
    batch = move_batch(collate_studies(items[: args.batch_size]), torch.device("cuda"))
    initialization_started = time.perf_counter()
    model = model_module.KneeAbnormalityModel(
        config.model, config.data, len(labels.targets), load_pretrained=False
    ).cuda()
    model.train(args.mode == "train")
    optimizer = (
        build_optimizer(model, config.optimizer) if args.mode == "train" else None
    )

    def step(current):
        if optimizer is not None:
            return training_step(model, optimizer, current, config)
        with torch.inference_mode(), torch.autocast("cuda", dtype=torch.bfloat16):
            output = model(current)
        if not torch.isfinite(output).all():
            raise RuntimeError("Non-finite inference")
        return float(output.mean())

    initialization_seconds = time.perf_counter() - initialization_started
    warmup_seconds = []
    durations = []
    values = []
    for index in range(args.warmup + args.steps):
        torch.manual_seed(config.seed + index % args.steps)
        torch.cuda.synchronize()
        if index == args.warmup:
            torch.cuda.reset_peak_memory_stats()
        started = time.perf_counter()
        value = step(batch)
        torch.cuda.synchronize()
        if index < args.warmup:
            warmup_seconds.append(time.perf_counter() - started)
        if index >= args.warmup:
            durations.append(time.perf_counter() - started)
            values.append(value)
    peak = torch.cuda.max_memory_allocated()
    with torch.profiler.profile(
        activities=[
            torch.profiler.ProfilerActivity.CPU,
            torch.profiler.ProfilerActivity.CUDA,
        ]
    ) as profile:
        step(batch)
        torch.cuda.synchronize()
    profile.export_chrome_trace(str(args.output_dir / "trace.json"))
    (args.output_dir / "operators.txt").write_text(
        profile.key_averages().table(sort_by="self_cuda_time_total", row_limit=60)
    )
    backend = [
        event.key
        for event in profile.key_averages()
        if "scaled_dot_product" in event.key or "attention" in event.key
    ]
    del batch
    gc.collect()
    torch.cuda.empty_cache()
    loader = build_data_loader(
        dataset,
        config.trainer,
        batch_size=args.batch_size,
        num_workers=args.workers,
        persistent_workers=args.workers > 0,
        shuffle=False,
    )
    end_to_end = []
    for repeat in range(4):
        torch.cuda.synchronize()
        started = time.perf_counter()
        iteration = iter(loader)
        batch_times = []
        while True:
            try:
                current = next(iteration)
            except StopIteration:
                break
            step(move_batch(current, torch.device("cuda")))
            torch.cuda.synchronize()
            finished = time.perf_counter()
            batch_times.append(finished - started)
            started = finished
        end_to_end.append(batch_times)
    result = {
        "mode": args.mode,
        "config": config.model_dump(mode="json"),
        "options": args.model_dump(mode="json"),
        "source_root": str(source),
        "source_sha256": {
            str(path.relative_to(source)): hashlib.sha256(path.read_bytes()).hexdigest()
            for path in source.rglob("*.py")
        },
        "label_provenance": labels.provenance,
        "gpu": torch.cuda.get_device_name(),
        "first_visit_seconds": loading,
        "initialization_seconds": initialization_seconds,
        "warmup_seconds": warmup_seconds,
        "model_seconds": durations,
        "mean_model_seconds": float(np.mean(durations)),
        "peak_allocated_bytes": peak,
        "end_to_end_batch_seconds": end_to_end,
        "attention_operators": backend,
        "values": values,
    }
    (args.output_dir / "results.json").write_text(json.dumps(result, indent=2) + "\n")
    print(
        json.dumps(
            {
                key: result[key]
                for key in (
                    "mode",
                    "mean_model_seconds",
                    "peak_allocated_bytes",
                    "attention_operators",
                    "end_to_end_batch_seconds",
                )
            }
        ),
        flush=True,
    )


if __name__ == "__main__":
    main()
