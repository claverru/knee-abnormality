import torch
from torch import nn
from torch.nn import functional as F

FUSION_IDENTITY = {
    "algorithm": "per_plane_spatial_queries_v1",
    "query_layout": "cartesian_3x3x3",
    "query_offsets_mm": [-35.0, 0.0, 35.0],
    "execution": "eager_pytorch",
    "coordinate_scale_mm": 100.0,
    "proximity_scale_mm": 35.0,
    "coverage_radius_mm": 18.0,
    "coverage": "nearest_valid_anchor_within_fixed_radius_v1",
    "width": 64,
    "heads": 1,
    "plane_order": ["Axial", "Coronal", "Sagittal"],
    "feature_grid": "rexnet_symmetric_padding_stride32_origin0",
    "preparation": "rounded_resampling_half_pixel_crop_v1",
    "augmentation": "kornia_normalized_affine_align_corners_false_v1",
    "eligibility": "accepted_reference_two_covered_planes_per_query_v1",
    "validity": "all_channel_anchor_bilinear_support_inside_rect",
    "aggregation": "global_and_regional_target_attention_mil_v1",
}


def feature_geometry(
    patient_maps: torch.Tensor,
    valid_rects: torch.Tensor,
    native_positions: torch.Tensor,
    geometry_valid: torch.Tensor,
    inverse_affine: torch.Tensor,
    height: int,
    width: int,
) -> tuple[torch.Tensor, torch.Tensor, torch.Tensor, torch.Tensor]:
    with torch.autocast(device_type=patient_maps.device.type, enabled=False):
        y, x = torch.meshgrid(
            torch.arange(height, device=patient_maps.device) * 32,
            torch.arange(width, device=patient_maps.device) * 32,
            indexing="ij",
        )
        anchors = torch.stack(
            (x.flatten(), y.flatten(), torch.ones_like(x.flatten())), -1
        ).float()
        prepared = torch.einsum("nij,kj->nki", inverse_affine.float(), anchors)
        top, left, rows, columns = valid_rects.float().unbind(-1)
        px = prepared[:, None, :, 0]
        py = prepared[:, None, :, 1]
        valid = (
            (px >= left[..., None])
            & (px <= (left + columns - 1)[..., None])
            & (py >= top[..., None])
            & (py <= (top + rows - 1)[..., None])
        ).all(1) & geometry_valid[:, None]
        middle = patient_maps.shape[1] // 2
        mapped = patient_maps[:, middle].float() @ inverse_affine.float()
        points = torch.einsum("nij,kj->nki", mapped, anchors)
        u = F.normalize(mapped[:, :, 0], dim=-1)
        v = F.normalize(mapped[:, :, 1], dim=-1)
        normal = F.normalize(torch.linalg.cross(u, v), dim=-1)
        axes = torch.stack((u, v, normal), dim=1)
        depth = torch.einsum(
            "nci,ni->nc",
            native_positions.float() - native_positions[:, middle : middle + 1].float(),
            normal,
        )
    return points, axes, depth, valid


class SpatialPooling(nn.Module):
    def __init__(self, channels: int, output_width: int, planes: list[int]) -> None:
        super().__init__()
        width = FUSION_IDENTITY["width"]
        self.projection = nn.Linear(channels, width)
        self.type_embedding = nn.Embedding(len(planes), width)
        self.position = nn.Sequential(
            nn.Linear(3, width), nn.GELU(), nn.Linear(width, width)
        )
        self.query = nn.Parameter(torch.zeros(width))
        self.key_value = nn.Linear(width, 2 * width)
        self.query_projection = nn.Linear(width, width)
        self.geometry = nn.Sequential(nn.Linear(12, 16), nn.GELU(), nn.Linear(16, 1))
        nn.init.zeros_(self.geometry[-1].weight)
        nn.init.zeros_(self.geometry[-1].bias)
        self.norm = nn.LayerNorm(width)
        self.mlp = nn.Sequential(
            nn.Linear(width, 2 * width), nn.GELU(), nn.Linear(2 * width, width)
        )
        self.output = nn.Linear(3 * width, output_width)
        self.output_position = nn.Sequential(
            nn.Linear(3, width), nn.GELU(), nn.Linear(width, output_width)
        )
        self.source = nn.Parameter(torch.zeros(output_width))
        offsets = torch.tensor(FUSION_IDENTITY["query_offsets_mm"])
        self.register_buffer(
            "offsets", torch.cartesian_prod(offsets, offsets, offsets), persistent=False
        )
        self.register_buffer("planes", torch.tensor(planes), persistent=False)

    def forward(
        self,
        maps: torch.Tensor,
        points: torch.Tensor,
        axes: torch.Tensor,
        depth: torch.Tensor,
        valid: torch.Tensor,
        study: torch.Tensor,
        types: torch.Tensor,
        centers: torch.Tensor,
        center_valid: torch.Tensor,
        input_slots: torch.Tensor,
        max_inputs: int,
    ) -> tuple[torch.Tensor, torch.Tensor]:
        batch_size = len(centers)
        grid_size = maps.shape[-2] * maps.shape[-1]
        projected = self.projection(maps.permute(0, 2, 3, 1).flatten(1, 2))
        local = projected + self.type_embedding(types)[:, None]
        kv = self.key_value(local)
        packed = kv.new_zeros((batch_size, max_inputs, grid_size, kv.shape[-1]))
        packed[study, input_slots] = kv
        keys, values = packed.flatten(1, 2).chunk(2, -1)
        mask = torch.zeros(
            (batch_size, max_inputs, grid_size), device=maps.device, dtype=torch.bool
        )
        mask[study, input_slots] = valid
        local_planes = torch.full(
            (batch_size, max_inputs), -1, device=maps.device, dtype=torch.long
        )
        local_planes[study, input_slots] = self.planes[types]
        local_points = centers.new_zeros((batch_size, max_inputs, grid_size, 3))
        local_axes = centers.new_zeros((batch_size, max_inputs, 3, 3))
        local_depth = centers.new_zeros((batch_size, max_inputs, 3))
        local_points[study, input_slots] = points - centers[study, None]
        local_axes[study, input_slots] = axes
        local_depth[study, input_slots] = depth
        delta = local_points[:, None] - self.offsets[None, :, None, None]
        displacement = torch.einsum("bnki,bnai->bnka", local_points, local_axes) / 100.0
        context = torch.cat((local_axes[:, :, :2].flatten(-2), local_depth / 100.0), -1)
        geometric = torch.cat(
            (displacement, context[:, :, None].expand(-1, -1, grid_size, -1)), -1
        )
        base = self.geometry[0](geometric)
        query_local = torch.einsum("qi,bnai->bqna", self.offsets / 100.0, local_axes)
        query_hidden = F.linear(query_local, self.geometry[0].weight[:, :3])
        hidden = self.geometry[1](base[:, None] - query_hidden[:, :, :, None])
        distance_squared = delta.square().sum(-1)
        bias = -distance_squared / (2 * FUSION_IDENTITY["proximity_scale_mm"] ** 2)
        bias = bias + self.geometry[2](hidden).squeeze(-1).float()
        within_radius = (
            distance_squared <= FUSION_IDENTITY["coverage_radius_mm"] ** 2
        ).flatten(2, 3)
        bias = bias.flatten(2, 3)
        position = self.offsets / 100.0
        queries = self.query_projection(self.position(position) + self.query)
        queries = queries[None, None].expand(batch_size, -1, -1, -1)
        first_token = torch.arange(max_inputs * grid_size, device=maps.device) == 0
        plane_vectors = []
        plane_coverage = []
        for plane in range(3):
            plane_valid = (mask & (local_planes == plane)[..., None]).flatten(1, 2)
            covered = (within_radius & plane_valid[:, None]).any(-1)
            attention_valid = (plane_valid[:, None] & covered[..., None]) | (
                ~covered[..., None] & first_token
            )
            plane_bias = bias.masked_fill(~attention_valid, -torch.inf)
            plane_vectors.append(
                F.scaled_dot_product_attention(
                    queries,
                    keys[:, None],
                    values[:, None],
                    attn_mask=plane_bias[:, None].to(keys.dtype),
                )[:, 0]
            )
            plane_coverage.append(covered)
        covered = torch.stack(plane_coverage, dim=-1)
        regional = torch.stack(plane_vectors, dim=2)
        regional = regional + self.mlp(self.norm(regional))
        regional = regional.masked_fill(~covered[..., None], 0).flatten(2, 3)
        available = center_valid[:, None] & (covered.sum(-1) >= 2)
        return self.output(regional) + self.output_position(position)[
            None
        ] + self.source, available
