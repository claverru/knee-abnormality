import hashlib
import json
import math
import shutil
from datetime import UTC, datetime
from pathlib import Path

import numpy as np
import polars as pl
import torch
from pydantic import Field
from pydantic_settings import CliPositionalArg
from torch.utils.data import Subset

import knee_abnormality
from knee_abnormality.cli import CLISettings
from knee_abnormality.config import load_config
from knee_abnormality.dataset import (
    KneeStudyDataset,
    build_data_loader,
    build_series_index,
)
from knee_abnormality.labels import load_fold_labels
from knee_abnormality.provenance import sha256_file
from knee_abnormality.spatial_pooling import FUSION_IDENTITY, feature_geometry


class CoverageCLI(CLISettings):
    config: CliPositionalArg[Path]
    output_dir: Path
    samples: int = Field(default=128, ge=1)
    workers: int = Field(default=8, ge=0)


def main() -> None:
    args = CoverageCLI()
    config = load_config(args.config)
    if config.data.image_size != 224 or config.data.slices_per_input != 3:
        raise ValueError(
            "Coverage selection requires 224-pixel triplets and 7x7 anchors"
        )
    torch.set_num_threads(1)
    labels = load_fold_labels(config, config.validation.folds[0])
    index = build_series_index(config.paths.data_root, config.data, "train")
    dataset = KneeStudyDataset(
        labels.training, index, config.data, labels.targets, True, config.seed
    )
    selected = sorted(
        np.random.default_rng(config.seed)
        .choice(len(dataset), min(args.samples, len(dataset)), replace=False)
        .tolist()
    )
    args.output_dir.mkdir(parents=True, exist_ok=False)
    snapshot = args.output_dir / "source"
    shutil.copytree(
        Path(knee_abnormality.__file__).parent,
        snapshot / "knee_abnormality",
        ignore=shutil.ignore_patterns("__pycache__"),
    )
    shutil.copy2(__file__, snapshot / Path(__file__).name)
    loader = build_data_loader(
        Subset(dataset, selected),
        config.trainer,
        batch_size=1,
        num_workers=args.workers,
        pin_memory=False,
        persistent_workers=False,
        generator=torch.Generator().manual_seed(config.seed),
    )
    offsets = torch.tensor(FUSION_IDENTITY["query_offsets_mm"])
    queries = torch.cartesian_prod(offsets, offsets, offsets)
    planes = ("Axial", "Coronal", "Sagittal")
    records = []
    inputs = []
    for batch in loader:
        present = batch["slice_mask"]
        geometry = {
            name: batch[name][present]
            for name in (
                "patient_maps",
                "valid_rects",
                "native_positions",
                "geometry_valid",
            )
        }
        points, _, _, valid = feature_geometry(
            **geometry,
            inverse_affine=torch.eye(3)[None].expand(
                len(geometry["patient_maps"]), -1, -1
            ),
            height=7,
            width=7,
        )
        types = batch["series_type_index"][present.nonzero(as_tuple=True)[0]]
        token_planes = torch.tensor(
            [planes.index(config.data.series_types[kind].plane) for kind in types]
        )
        accepted = bool(batch["center_valid"][0])
        points = points - batch["study_centers"][0]
        for plane, name in enumerate(planes):
            anchors = points[valid & (token_planes == plane)[:, None]]
            distances = (
                (queries[:, None] - anchors[None]).square().sum(-1).amin(-1).sqrt()
                if len(anchors) and accepted
                else None
            )
            for query in range(len(queries)):
                records.append(
                    {
                        "StudyInstanceUID": batch["study_uid"][0],
                        "center_accepted": accepted,
                        "query": query,
                        "plane": name,
                        "valid_anchors": len(anchors),
                        "nearest_anchor_mm": float(distances[query])
                        if distances is not None
                        else None,
                    }
                )
        digest = hashlib.sha256()
        for name, tensor in sorted(
            {**geometry, "study_centers": batch["study_centers"]}.items()
        ):
            digest.update(name.encode())
            digest.update(str(tuple(tensor.shape)).encode())
            digest.update(tensor.numpy().tobytes())
        inputs.append(
            {
                "study_uid": batch["study_uid"][0],
                "geometry_sha256": digest.hexdigest(),
                "selected_paths": batch["selected_paths"],
                "failures": batch["failures"],
            }
        )
    frame = pl.DataFrame(records)
    distances = frame["nearest_anchor_mm"].drop_nulls().to_numpy()
    if not len(distances) or not np.isfinite(distances).all():
        raise ValueError(
            "Coverage selection requires finite accepted training geometry"
        )
    radius = math.ceil(float(np.quantile(distances, 0.95)))
    coverage = frame.with_columns(
        (pl.col("nearest_anchor_mm") <= radius).fill_null(False).alias("covered")
    )
    regional = coverage.group_by("StudyInstanceUID", "query").agg(
        (pl.col("covered").sum() >= 2).alias("valid")
    )
    frame.write_parquet(args.output_dir / "nearest_anchors.parquet")
    (args.output_dir / "inputs.json").write_text(json.dumps(inputs, indent=2) + "\n")
    result = {
        "created_at": datetime.now(UTC).isoformat(),
        "config": config.model_dump(mode="json"),
        "options": args.model_dump(mode="json"),
        "training_membership_sha256": hashlib.sha256(
            labels.training.select("StudyInstanceUID").write_csv().encode()
        ).hexdigest(),
        "label_split_provenance": labels.provenance,
        "source_sha256": {
            str(path.relative_to(snapshot)): sha256_file(path)
            for path in sorted(snapshot.rglob("*.py"))
        },
        "inputs_sha256": sha256_file(args.output_dir / "inputs.json"),
        "nearest_anchors_sha256": sha256_file(
            args.output_dir / "nearest_anchors.parquet"
        ),
        "sampling": "uniform_seeded_training_subset_epoch0_training_triplet_budgets",
        "augmentation": "none",
        "series_dropout": False,
        "query_offsets_mm": queries.tolist(),
        "radius_rule": "ceil_mm_p95_nearest_valid_anchor_in_accepted_present_plane_groups",
        "coverage_radius_mm": radius,
        "studies": len(selected),
        "accepted_centers": frame.filter(pl.col("center_accepted"))[
            "StudyInstanceUID"
        ].n_unique(),
        "measured_query_plane_groups": len(distances),
        "nearest_anchor_quantiles_mm": {
            str(q): float(np.quantile(distances, q))
            for q in (0.5, 0.9, 0.95, 0.99, 1.0)
        },
        "covered_fraction_by_plane": coverage.group_by("plane")
        .agg(pl.col("covered").mean())
        .sort("plane")
        .to_dicts(),
        "valid_regional_fraction": regional["valid"].mean(),
    }
    (args.output_dir / "results.json").write_text(json.dumps(result, indent=2) + "\n")
    print(
        json.dumps(
            {
                key: result[key]
                for key in (
                    "studies",
                    "accepted_centers",
                    "coverage_radius_mm",
                    "nearest_anchor_quantiles_mm",
                    "valid_regional_fraction",
                )
            },
            indent=2,
        )
    )


if __name__ == "__main__":
    main()
