import ast
import json
import shutil
from concurrent.futures import ProcessPoolExecutor
from datetime import UTC, datetime
from importlib.metadata import version
from pathlib import Path

import polars as pl
from pydantic import Field

from knee_abnormality.cli import CLISettings
from knee_abnormality.config import load_config
from knee_abnormality.dataset import KneeStudyDataset, build_series_index
from knee_abnormality.preprocessing.dicom import PREPARATION_ERRORS
from knee_abnormality.preprocessing.geometry import NativePose, PoseLayout
from knee_abnormality.preprocessing.laterality import (
    EXPLICIT_FIELDS,
    SIDE_POLICY,
    TEXT_FIELDS,
    header_side,
    resolve_knee_side,
    series_side,
    side_record,
)
from knee_abnormality.preprocessing.series import KNOWN_BAD_INSTANCES, SliceRecord
from knee_abnormality.provenance import sha256_file

STUDY = "StudyInstanceUID"
SERIES = "SeriesInstanceUID"
SPLITS = ("train", "test")
_DATA = None


class PolicyCLI(CLISettings):
    config: Path = Path("configs/baseline.yaml")
    geometry_headers: Path = Path(
        "outputs/analysis/knee-side-geometry-20260914/headers.parquet"
    )
    reference_studies: Path = Path(
        "outputs/analysis/knee-laterality-20260914/reviewed/studies.parquet"
    )
    output_dir: Path = Path("outputs/analysis/knee-side-policy-20260914")
    production_studies: Path | None = None
    workers: int = Field(default=8, ge=1, le=16)


def _initialize(data) -> None:
    global _DATA
    _DATA = data


def production_study(task) -> dict:
    split, study = task
    dataset = KneeStudyDataset(
        pl.DataFrame({STUDY: [study.uid]}), {study.uid: study}, _DATA, (), False, 0
    )
    dataset._index_study_geometry(study)
    candidates = {c.uid: c for values in study.slots for c in values}
    records = {}
    for uid, candidate in sorted(candidates.items()):
        indexed = dataset._slice_index.get(candidate.directory)
        records[uid] = {
            "files": sum(1 for _ in candidate.directory.glob("*.dcm")),
            "indexed_records": None if indexed is None else len(indexed.records),
        }
    return {
        "split": split,
        **side_record(study.uid, dataset._study_sides[study.uid]),
        "series_records": json.dumps(records, sort_keys=True),
    }


def _pose(row) -> NativePose | None:
    try:
        position = tuple(
            float(v) for v in ast.literal_eval(row["ImagePositionPatient"])
        )
        orientation = tuple(
            float(v) for v in ast.literal_eval(row["ImageOrientationPatient"])
        )
        spacing = tuple(float(v) for v in ast.literal_eval(row["PixelSpacing"]))
        if len(position) != 3 or len(orientation) != 6 or len(spacing) != 2:
            return None
        return NativePose(
            position,
            PoseLayout(
                orientation,
                spacing,
                int(row["Rows"]),
                int(row["Columns"]),
                row["FrameOfReferenceUID"] or None,
            ),
        )
    except (*PREPARATION_ERRORS, SyntaxError):
        return None


def snapshot_sides(headers: pl.DataFrame, indexes: dict) -> pl.DataFrame:
    supported = {
        (split, study.uid, candidate.uid)
        for split, index in indexes.items()
        for study in index.values()
        for values in study.slots
        for candidate in values
    }
    grouped = {}
    for row in headers.sort("path").iter_rows(named=True):
        path = Path(row["path"])
        key = (row["split"], row[STUDY], row[SERIES])
        if key not in supported or path.stem in KNOWN_BAD_INSTANCES:
            continue
        entry = grouped.setdefault(key, ([], []))
        entry[0].append(
            header_side(
                {
                    field: row[field]
                    for field in (
                        *EXPLICIT_FIELDS,
                        *TEXT_FIELDS,
                        "Manufacturer",
                        "ManufacturerModelName",
                    )
                }
            )
        )
        entry[1].append(SliceRecord(path, 0.0, len(entry[1]), pose=_pose(row)))
    studies = {}
    for (split, study, series), (sides, records) in grouped.items():
        studies.setdefault((split, study), []).append(
            (series, series_side(tuple(records), tuple(sides)))
        )
    rows = []
    for split, index in indexes.items():
        for uid in sorted(index):
            side = resolve_knee_side(tuple(studies.get((split, uid), ())))
            rows.append({"split": split, **side_record(uid, side)})
    return pl.from_dicts(rows, infer_schema_length=None)


def counts(frame: pl.DataFrame, *columns: str) -> list[dict]:
    return frame.group_by(columns).len().sort(columns).to_dicts()


def agreement(frame: pl.DataFrame, reference: str, column: str) -> dict:
    subset = frame.filter(
        (pl.col("reference_status") == reference)
        & pl.col("reference_side").is_not_null()
    )
    expected = pl.col("reference_side").replace_strict(
        {"L": "left", "R": "right"}, return_dtype=pl.String
    )
    assigned = subset.filter(pl.col(column) != "unknown")
    agree = assigned.filter(pl.col(column) == expected).height
    return {
        "reference_studies": subset.height,
        "assigned": assigned.height,
        "unknown": subset.height - assigned.height,
        "agree": agree,
        "disagree": assigned.height - agree,
        "agreement": agree / assigned.height if assigned.height else None,
        "cells": counts(subset, "reference_side", column),
    }


def main() -> None:
    settings = PolicyCLI()
    settings.output_dir.mkdir(parents=True, exist_ok=False)
    started = datetime.now(UTC)
    config = load_config(settings.config)
    indexes = {
        split: build_series_index(config.paths.data_root, config.data, split)
        for split in SPLITS
    }
    headers = pl.read_parquet(settings.geometry_headers)
    snapshot = snapshot_sides(headers, indexes)
    if settings.production_studies:
        production = pl.read_parquet(settings.production_studies)
    else:
        tasks = [
            (split, index[uid])
            for split, index in indexes.items()
            for uid in sorted(index)
        ]
        rows = []
        with ProcessPoolExecutor(
            settings.workers, initializer=_initialize, initargs=(config.data,)
        ) as pool:
            for number, row in enumerate(
                pool.map(production_study, tasks, chunksize=8), 1
            ):
                rows.append(row)
                if number % 500 == 0:
                    print(
                        json.dumps({"studies": number, "total": len(tasks)}), flush=True
                    )
        production = pl.from_dicts(rows, infer_schema_length=None)
    production.write_parquet(settings.output_dir / "production.parquet")
    snapshot.write_parquet(settings.output_dir / "snapshot.parquet")
    folds = pl.read_csv(config.split_dir / "studies.csv").select(STUDY, "fold")
    reference = pl.read_parquet(settings.reference_studies).select(
        "split",
        STUDY,
        pl.col("status").alias("reference_status"),
        pl.col("side").alias("reference_side"),
    )
    compared = [
        "knee_side",
        "side",
        "source",
        "reason",
        "explicit_sides",
        "text_sides",
        "coordinate_side",
        "coordinate_reason",
        "center_x_mm",
        "coordinate_series",
    ]
    studies = (
        production.join(
            snapshot.select(
                "split", STUDY, *[pl.col(c).alias(f"snapshot_{c}") for c in compared]
            ),
            on=["split", STUDY],
            how="left",
            validate="1:1",
        )
        .join(folds, on=STUDY, how="left", validate="m:1")
        .join(reference, on=["split", STUDY], how="left", validate="1:1")
        .with_columns(
            (pl.col("side") != pl.col("snapshot_side")).alias("side_differs"),
            (pl.col("explicit_sides") != pl.col("snapshot_explicit_sides")).alias(
                "explicit_differs"
            ),
            (pl.col("text_sides") != pl.col("snapshot_text_sides")).alias(
                "text_differs"
            ),
            (pl.col("coordinate_side") != pl.col("snapshot_coordinate_side")).alias(
                "coordinate_differs"
            ),
        )
        .sort("split", STUDY)
    )
    studies.write_parquet(settings.output_dir / "studies.parquet")
    differences = studies.filter(
        pl.col("side_differs")
        | pl.col("explicit_differs")
        | pl.col("text_differs")
        | pl.col("coordinate_differs")
    )
    differences.drop("series_records").write_csv(
        settings.output_dir / "reconciliation.csv"
    )
    studies.group_by("split", "manufacturers", "side").len().sort(
        "split", "manufacturers", "side"
    ).write_csv(settings.output_dir / "manufacturer_sides.csv")
    studies.group_by("split", "models", "side").len().sort(
        "split", "models", "side"
    ).write_csv(settings.output_dir / "model_sides.csv")
    train = studies.filter(pl.col("split") == "train")
    series_records = [
        value
        for text in studies["series_records"]
        for value in json.loads(text).values()
    ]
    summary = {
        "created_at": started.isoformat(),
        "settings": settings.model_dump(mode="json"),
        "policy": SIDE_POLICY,
        "versions": {name: version(name) for name in ("polars", "pydicom", "numpy")},
        "cohorts": {
            split: {
                "studies": frame.height,
                "sides": counts(frame, "side"),
                "sources": counts(frame, "source", "side"),
                "reasons": counts(frame, "reason"),
                "coordinate_reasons": counts(frame, "coordinate_reason"),
                "coverage": frame.filter(pl.col("side") != "unknown").height
                / frame.height,
                "snapshot_sides": counts(frame, "snapshot_side"),
                "snapshot_reasons": counts(frame, "snapshot_reason"),
                "snapshot_versus_production": counts(frame, "snapshot_side", "side"),
                "explicit_evidence_differs": frame["explicit_differs"].sum(),
                "text_evidence_differs": frame["text_differs"].sum(),
                "coordinate_side_differs": frame["coordinate_differs"].sum(),
            }
            for split, frame in (
                (split, studies.filter(pl.col("split") == split)) for split in SPLITS
            )
        },
        "train_folds": counts(train, "fold", "side"),
        "train_fold_sources": counts(train, "fold", "source"),
        "train_conflicts": counts(
            train.filter(
                pl.col("reason").is_in(
                    [
                        "bilateral_explicit",
                        "conflicting_explicit_or_text",
                        "coordinates_contradict_metadata",
                    ]
                )
            ),
            "reason",
            "manufacturers",
        ),
        "train_agreement": {
            "final_side_vs_sampled_explicit_reference": agreement(
                train, "explicit", "side"
            ),
            "final_side_vs_sampled_text_reference": agreement(
                train, "text_inferred", "side"
            ),
            "coordinate_side_vs_sampled_explicit_reference": agreement(
                train, "explicit", "coordinate_side"
            ),
            "coordinate_side_vs_sampled_text_reference": agreement(
                train, "text_inferred", "coordinate_side"
            ),
        },
        "train_reference_versus_final": counts(train, "reference_status", "side"),
        "indexed_records": {
            "series": len(series_records),
            "files": sum(value["files"] for value in series_records),
            "indexed_records": sum(
                value["indexed_records"] or 0 for value in series_records
            ),
            "unindexed_series": sum(
                value["indexed_records"] is None for value in series_records
            ),
        },
        "snapshot_header_rows": headers.height,
        "example_test": studies.filter(pl.col("split") == "test")
        .drop("series_records")
        .to_dicts(),
        "limitations": [
            "Production assignments use every readable header of loader-supported series; the snapshot uses the audit's three lexicographic headers per series.",
            "Explicit and text references come from the sampled reviewed audit and are metadata evidence, not anatomical truth.",
            "Coordinate agreement is an empirical acquisition convention; hidden-test metadata coverage is not measurable locally.",
            "CPU header reads only; no pixels, models, reports, labels, tests or GPU access.",
        ],
    }
    (settings.output_dir / "summary.json").write_text(
        json.dumps(summary, indent=2, default=int) + "\n"
    )
    shutil.copy2(__file__, settings.output_dir / Path(__file__).name)
    paths = [
        settings.config,
        settings.geometry_headers,
        settings.reference_studies,
        config.split_dir / "studies.csv",
        *(config.raw_dir / f"{split}_series.csv" for split in SPLITS),
        Path(__file__),
        Path("src/knee_abnormality/preprocessing/laterality.py"),
        Path("src/knee_abnormality/preprocessing/dicom.py"),
        Path("src/knee_abnormality/preprocessing/series.py"),
        Path("src/knee_abnormality/dataset.py"),
        Path("uv.lock"),
    ]
    if settings.production_studies:
        paths.append(settings.production_studies)
    manifest = {
        "created_at": summary["created_at"],
        "inputs_and_sources": [
            {"path": str(path), "sha256": sha256_file(path)} for path in paths
        ],
        "outputs": [
            {"path": str(path), "sha256": sha256_file(path)}
            for path in sorted(settings.output_dir.iterdir())
            if path.is_file()
        ],
    }
    (settings.output_dir / "manifest.json").write_text(
        json.dumps(manifest, indent=2) + "\n"
    )
    print(
        json.dumps(
            {
                split: {
                    key: summary["cohorts"][split][key]
                    for key in ("studies", "sides", "reasons", "coverage")
                }
                for split in SPLITS
            },
            indent=2,
            default=int,
        ),
        flush=True,
    )


if __name__ == "__main__":
    main()
