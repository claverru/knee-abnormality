import json
import shutil
from pathlib import Path
from zipfile import ZipFile

import numpy as np
import polars as pl
import torch
from pydantic import Field
from sklearn.metrics import roc_auc_score
from torch.nn import functional as F

from knee_abnormality.cli import CLISettings
from knee_abnormality.metrics import aucs, validation_metrics
from knee_abnormality.provenance import sha256_file
from scripts.adhoc.compare_public_labels import bootstrap_auc

STUDY = "StudyInstanceUID"
PAIRS = {
    "meniscus": ("Medial Meniscus", "Lateral Meniscus"),
    "oa": ("Medial OA", "Lateral OA"),
}


class AnalyzeCLI(CLISettings):
    reference: Path = Path("outputs/reflection-sigma40")
    candidate: Path = Path("outputs/knee-side-embedding")
    output_dir: Path = Path("outputs/analysis/knee-side-embedding-20260914")
    headers: Path = Path(
        "outputs/analysis/sharing-disagreement-20260914/study_headers.parquet"
    )
    bootstrap_draws: int = Field(default=2000, ge=100)
    seed: int = 20260914


def fingerprint(path):
    return {"path": str(path), "sha256": sha256_file(path)}


def differences(reference, candidate):
    return {
        key: {"reference": reference.get(key), "candidate": candidate.get(key)}
        for key in sorted(reference.keys() | candidate.keys())
        if reference.get(key) != candidate.get(key)
    }


def binary_cross_entropy(truth, values):
    p = np.clip(values, 1e-7, 1 - 1e-7)
    return float(-(truth * np.log(p) + (1 - truth) * np.log1p(-p)).mean())


def analyze(settings):
    torch.set_num_threads(1)
    settings.output_dir.mkdir(parents=True, exist_ok=False)
    roots = {"reference": settings.reference, "candidate": settings.candidate}
    configs = {
        name: json.loads((root / "effective_config.json").read_text())
        for name, root in roots.items()
    }
    config_diff = differences(configs["reference"], configs["candidate"])
    model_diff = differences(
        configs["reference"]["model"], configs["candidate"]["model"]
    )
    if config_diff.keys() - {"name", "model"} or model_diff.keys() - {"side_dropout"}:
        raise ValueError("Comparison requires matching configurations except side")
    config = configs["candidate"]
    if config["validation"]["folds"] != [0]:
        raise ValueError("This investigation requires fold 0")
    split = Path(config["paths"]["data_root"]) / "folds" / config["validation"]["split"]
    targets_path, studies_path = split / "targets.csv", split / "studies.csv"
    labels = pl.read_csv(targets_path)
    targets = labels.columns[1:]
    side_paths = sorted(
        (settings.candidate / "fold_0" / "epochs").glob(
            "knee_sides_validation_epoch_*.csv"
        )
    )
    sides = [pl.read_csv(path) for path in side_paths]
    if not sides or not all(frame.equals(sides[0]) for frame in sides):
        raise ValueError("Validation side manifests are missing or differ by epoch")
    frame = (
        pl.read_csv(studies_path)
        .filter(pl.col("fold") == 0)
        .join(labels, on=STUDY, validate="1:1")
        .join(
            pl.read_parquet(settings.headers).select(STUDY, "vendor"),
            on=STUDY,
            validate="1:1",
            how="left",
        )
        .join(
            sides[0].select(STUDY, "side", "source", "reason"),
            on=STUDY,
            validate="1:1",
            how="left",
        )
        .sort(STUDY)
    )
    if frame["vendor"].null_count() or frame["side"].null_count():
        raise ValueError("Missing acquisition or side metadata")
    truth = frame.select(targets).to_numpy().astype(float)
    hard = (truth >= 0.5).astype(int)
    is_gold = frame["is_gold"].to_numpy()
    indices = {name: [targets.index(t) for t in pair] for name, pair in PAIRS.items()}
    predictions, runs, inputs = {}, {}, {}
    for name, root in roots.items():
        fold = root / "fold_0"
        selected = json.loads((fold / "selected_checkpoint.json").read_text())
        checkpoint_path = fold / Path(selected["path"]).name
        checkpoint = torch.load(checkpoint_path, map_location="cpu", weights_only=False)
        epoch = checkpoint["epoch"]
        vectors = checkpoint["state_dict"].get("model.knee_side_embedding")
        side_vectors = None
        if vectors is not None:
            vectors = vectors.detach().float()
            side_vectors = {
                "left_norm": float(vectors[0].norm()),
                "right_norm": float(vectors[1].norm()),
                "left_right_distance": float((vectors[0] - vectors[1]).norm()),
                "left_right_cosine": float(
                    F.cosine_similarity(vectors[0], vectors[1], dim=0)
                ),
                "plane_embedding_norms": checkpoint["state_dict"][
                    "model.spatial_tokens.plane.weight"
                ]
                .float()
                .norm(dim=1)
                .tolist(),
                "contrast_embedding_norms": checkpoint["state_dict"][
                    "model.spatial_tokens.contrast.weight"
                ]
                .float()
                .norm(dim=1)
                .tolist(),
            }
        del checkpoint
        prediction_path = fold / "validation_predictions.csv"
        prediction = pl.read_csv(prediction_path).sort(STUDY)
        if prediction[STUDY].to_list() != frame[STUDY].to_list():
            raise ValueError(f"Prediction membership mismatch: {name}")
        predictions[name] = prediction.select(targets).to_numpy().astype(float)
        metrics = validation_metrics(truth, predictions[name], targets, is_gold)
        saved = json.loads(
            (fold / "epochs" / f"metrics_epoch_{epoch:03d}.json").read_text()
        )
        if abs(metrics["macro_auc"] - saved["macro_auc"]) > 1e-12:
            raise ValueError(
                f"Selected predictions do not reproduce recorded AUC: {name}"
            )
        inputs[name] = json.loads((fold / "input_identity.json").read_text())
        epoch_paths = sorted((fold / "epochs").glob("metrics_epoch_*.json"))
        files = [
            root / "effective_config.json",
            root / "provenance.json",
            root / "source_snapshot.zip",
            checkpoint_path,
            prediction_path,
            *sorted(fold.glob("*.json")),
            *epoch_paths,
        ]
        runs[name] = {
            "selected_epoch": epoch + 1,
            "metrics": metrics,
            "knee_side_embedding": side_vectors,
            "epochs": [
                {
                    "epoch": int(p.stem.rsplit("_", 1)[1]) + 1,
                    "macro_auc": (m := json.loads(p.read_text()))["macro_auc"],
                    "soft_target_bce": m["soft_target_bce"],
                    "gold_macro_auc": m["gold"]["macro_auc"],
                }
                for p in epoch_paths
            ],
            "timing": json.loads((fold / "timing.json").read_text()),
            "files": [fingerprint(p) for p in files],
        }
    identity_diff = differences(inputs["reference"], inputs["candidate"])
    if identity_diff.keys() - {"preprocessing_identity"}:
        raise ValueError(f"Unexpected input identity differences: {identity_diff}")
    with (
        ZipFile(settings.reference / "source_snapshot.zip") as left,
        ZipFile(settings.candidate / "source_snapshot.zip") as right,
    ):
        source_changes = [
            name
            for name in sorted(set(left.namelist()) | set(right.namelist()))
            if name.startswith(("src/", "configs/"))
            and (
                name not in left.namelist()
                or name not in right.namelist()
                or left.read(name) != right.read(name)
            )
        ]

    def subset(mask, selected_targets):
        ix = [targets.index(t) for t in selected_targets]
        result = {
            "studies": int(mask.sum()),
            "positive_counts": dict(
                zip(
                    selected_targets,
                    hard[mask][:, ix].sum(axis=0).tolist(),
                    strict=True,
                )
            ),
        }
        for name, values in predictions.items():
            per_target = aucs(
                hard[mask][:, ix], values[mask][:, ix], selected_targets, strict=False
            )
            result[name] = {
                **per_target,
                "soft_target_bce": binary_cross_entropy(
                    truth[mask][:, ix], values[mask][:, ix]
                )
                if mask.any()
                else None,
            }
        return result

    conditional = []
    exclusive = {}
    codes = {name: hard[:, ix[0]] + 2 * hard[:, ix[1]] for name, ix in indices.items()}
    for family, ix in indices.items():
        other = "oa" if family == "meniscus" else "meniscus"
        for side, index in enumerate(ix):
            for condition, flags in [
                ("opposite_compartment", hard[:, ix[1 - side]]),
                (f"any_{other}", (codes[other] > 0).astype(int)),
            ]:
                for value in (0, 1):
                    conditional.append(
                        {
                            "family": family,
                            "target": targets[index],
                            "condition": condition,
                            "value": value,
                            **subset(flags == value, [targets[index]]),
                        }
                    )
        mask = np.isin(codes[family], [1, 2])
        exclusive[family] = {
            "studies": int(mask.sum()),
            "medial_only": int((codes[family] == 1).sum()),
            "lateral_only": int((codes[family] == 2).sum()),
        }
        for name, values in predictions.items():
            p = np.clip(values[mask][:, ix], 1e-7, 1 - 1e-7)
            logits = np.log(p) - np.log1p(-p)
            margin = logits[:, 0] - logits[:, 1]
            medial = codes[family][mask] == 1
            exclusive[family][name] = {
                "logit_margin_auc": float(roc_auc_score(medial, margin)),
                "correct_side_fraction": float(
                    (
                        (margin > 0) * medial
                        + (margin < 0) * ~medial
                        + 0.5 * (margin == 0)
                    ).mean()
                ),
                "mean_signed_logit_margin": float(
                    (margin * np.where(medial, 1, -1)).mean()
                ),
            }
            for side in ("left", "right", "unknown"):
                selected = frame["side"].to_numpy()[mask] == side
                exclusive[family][name][f"logit_margin_auc_side={side}"] = (
                    float(roc_auc_score(medial[selected], margin[selected]))
                    if len(np.unique(medial[selected])) == 2
                    else None
                )
    strata = {
        f"{column}={value}": subset(frame[column].to_numpy() == value, targets)
        for column in ("side", "source", "vendor")
        for value in sorted(frame[column].unique().to_list())
    }
    strata.update(
        {
            f"vendor={vendor}|side={side}": subset(
                (frame["vendor"].to_numpy() == vendor)
                & (frame["side"].to_numpy() == side),
                targets,
            )
            for vendor in sorted(frame["vendor"].unique().to_list())
            for side in ("left", "right", "unknown")
        }
    )
    groups, inverse = np.unique(frame["report_group"].to_numpy(), return_inverse=True)
    rng = np.random.default_rng(settings.seed)
    group_counts = rng.multinomial(
        len(groups),
        np.full(len(groups), 1 / len(groups)),
        size=settings.bootstrap_draws,
    )[:, inverse]
    families = {
        "all": list(range(len(targets))),
        **indices,
        "other_eight": [
            i
            for i in range(len(targets))
            if i not in indices["meniscus"] + indices["oa"]
        ],
    }
    masks = {
        "all_studies": np.ones(frame.height, dtype=bool),
        **{
            f"side={side}": frame["side"].to_numpy() == side
            for side in ("left", "right", "unknown")
        },
        "known_side": frame["side"].to_numpy() != "unknown",
    }
    intervals = {}
    for mask_name, mask in masks.items():
        counts = group_counts * mask
        positive = counts @ hard
        valid = ((positive > 0) & (positive < counts.sum(axis=1, keepdims=True))).all(
            axis=1
        )
        for family, ix in families.items():
            key = f"{mask_name}:{family}"
            if not valid.any():
                intervals[key] = {"valid_draws": 0, "delta_percentile_95": None}
                continue
            delta = bootstrap_auc(
                hard[:, ix], predictions["candidate"][:, ix], counts[valid]
            ) - bootstrap_auc(
                hard[:, ix], predictions["reference"][:, ix], counts[valid]
            )
            intervals[key] = {
                "valid_draws": int(valid.sum()),
                "delta_percentile_95": np.percentile(delta, [2.5, 97.5]).tolist(),
                "delta_mean": float(delta.mean()),
            }
    per_target = [
        {
            "target": target,
            "positives": int(hard[:, index].sum()),
            **{name: runs[name]["metrics"]["per_target_auc"][target] for name in runs},
        }
        for index, target in enumerate(targets)
    ]
    for row in per_target:
        row["delta"] = row["candidate"] - row["reference"]
    report = {
        "settings": settings.model_dump(mode="json"),
        "studies": frame.height,
        "targets": targets,
        "runs": runs,
        "per_target": per_target,
        "family_means": {
            family: {
                name: float(
                    np.mean(
                        [
                            runs[name]["metrics"]["per_target_auc"][targets[i]]
                            for i in ix
                        ]
                    )
                )
                for name in runs
            }
            for family, ix in families.items()
        },
        "prediction_change": {
            "mean_abs_probability_difference": float(
                np.abs(predictions["candidate"] - predictions["reference"]).mean()
            ),
            "per_target_spearman": {
                target: float(
                    np.corrcoef(
                        np.argsort(np.argsort(predictions["candidate"][:, i])),
                        np.argsort(np.argsort(predictions["reference"][:, i])),
                    )[0, 1]
                )
                for i, target in enumerate(targets)
            },
        },
        "config_differences": config_diff,
        "input_identity_differences": sorted(identity_diff),
        "changed_source_archive_members": source_changes,
        "side_manifest": {
            "epochs_identical": len(side_paths),
            "counts": sides[0]
            .group_by("side", "source")
            .len()
            .sort("side", "source")
            .to_dicts(),
            "file": fingerprint(side_paths[0]),
        },
        "inputs": [
            fingerprint(p) for p in [targets_path, studies_path, settings.headers]
        ],
        "analysis_sources": [
            fingerprint(Path(p))
            for p in [
                __file__,
                "scripts/adhoc/compare_public_labels.py",
                "src/knee_abnormality/metrics.py",
            ]
        ],
        "conditional_auc": conditional,
        "exclusive_positive": exclusive,
        "strata": strata,
        "paired_report_group_bootstrap": {
            "groups": len(groups),
            "draws": settings.bootstrap_draws,
            "seed": settings.seed,
            "unit": "report group; strata keep the full-fold draw and zero out other studies",
            "intervals": intervals,
        },
        "limitations": [
            "Saved predictions and CPU-loaded checkpoint tensors only; no model execution or GPU access.",
            "One training seed and one development fold; intervals exclude training-seed and checkpoint-selection variability.",
            "Side and source strata use metadata-derived sides, not anatomical labels; vendor is an acquisition proxy.",
            "The run changes metadata inference, embedding and side dropout together; their separate contributions are not identifiable.",
        ],
    }
    (settings.output_dir / "comparison.json").write_text(
        json.dumps(report, indent=2, allow_nan=False) + "\n"
    )
    source_dir = settings.output_dir / "source"
    source_dir.mkdir()
    for item in report["analysis_sources"]:
        shutil.copy2(item["path"], source_dir / Path(item["path"]).name)
    print(
        json.dumps(
            {
                "artifact": str(settings.output_dir / "comparison.json"),
                "selected": {name: run["selected_epoch"] for name, run in runs.items()},
                "macro_auc": {
                    name: run["metrics"]["macro_auc"] for name, run in runs.items()
                },
                "family_means": report["family_means"],
                "intervals": intervals,
            },
            indent=2,
        )
    )


if __name__ == "__main__":
    analyze(AnalyzeCLI())
