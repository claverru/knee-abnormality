import json
from datetime import UTC, datetime
from pathlib import Path
from typing import Literal

import numpy as np
import polars as pl
import pydicom

from knee_abnormality.cli import NonExitingCLISettings
from knee_abnormality.provenance import sha256_file
from scripts.adhoc.analyze_sharing_generalization import (
    RUNS,
    auc,
    weighted_bootstrap_auc,
)

TARGETS = ["Medial Meniscus", "Lateral Meniscus", "Medial OA", "Lateral OA", "PF OA"]
PAIRS = {"meniscus": TARGETS[:2], "oa": TARGETS[2:4]}
TAGS = [
    "Manufacturer",
    "ManufacturerModelName",
    "MagneticFieldStrength",
    "PatientSex",
    "SeriesDescription",
    "ProtocolName",
    "Rows",
    "Columns",
    "PixelSpacing",
    "SliceThickness",
    "SpacingBetweenSlices",
    "RepetitionTime",
    "EchoTime",
]
STATES = {"positive", "negative", "below_threshold", "uncertain", "unmentioned"}


class AuditCLI(NonExitingCLISettings):
    phase: Literal["prepare", "summarize"]
    output: Path
    seed: int = 20260916


def save(path: Path, value: object) -> None:
    path.write_text(json.dumps(value, indent=2, ensure_ascii=False) + "\n")


def load_data() -> tuple[pl.DataFrame, dict[str, pl.DataFrame], list[str]]:
    frozen = pl.read_csv("data/folds/all_5fold_v3/targets.csv")
    targets = frozen.columns[1:]
    frame = pl.read_csv("data/folds/all_5fold_v3/studies.csv").join(
        frozen, on="StudyInstanceUID", validate="1:1"
    )
    val = frame.filter(pl.col("fold") == 0).sort("StudyInstanceUID")
    predictions = {}
    for name, run in RUNS.items():
        p = pl.read_csv(
            Path("outputs") / run / "fold_0/validation_predictions.csv"
        ).sort("StudyInstanceUID")
        if p["StudyInstanceUID"].to_list() != val["StudyInstanceUID"].to_list():
            raise ValueError(f"Study mismatch: {name}")
        values = p.select(targets).to_numpy()
        if not np.isfinite(values).all() or not ((values >= 0) & (values <= 1)).all():
            raise ValueError(f"Invalid predictions: {name}")
        predictions[name] = p
    return frame, predictions, targets


def contribution(y: np.ndarray, p: np.ndarray) -> np.ndarray:
    values = np.empty(len(y))
    positive = p[y == 1]
    negative = p[y == 0]
    comparisons = (positive[:, None] > negative).astype(float) + 0.5 * (
        positive[:, None] == negative
    )
    values[y == 1] = comparisons.mean(axis=1)
    values[y == 0] = comparisons.mean(axis=0)
    return values


def scanner_name(vendor: str, strength: str | None) -> str:
    return vendor + "/" + (f"{float(strength):g}" if strength else "missing")


def prepare(args: AuditCLI) -> None:
    args.output.mkdir(parents=True, exist_ok=False)
    frame, predictions, targets = load_data()
    val = frame.filter(pl.col("fold") == 0).sort("StudyInstanceUID")
    y = (val.select(targets).to_numpy() >= 0.5).astype(int)
    contributions = {
        name: np.column_stack(
            [
                contribution(y[:, i], p[target].to_numpy())
                for i, target in enumerate(targets)
            ]
        )
        for name, p in predictions.items()
    }
    rng = np.random.default_rng(args.seed)
    selected = []
    used = set()
    for family, pair in PAIRS.items():
        indices = [targets.index(t) for t in pair]
        code = y[:, indices[0]] + 2 * y[:, indices[1]]
        advantage = (
            contributions["1B"][:, indices] - contributions["reflection"][:, indices]
        ).mean(axis=1)
        divergence = np.abs(
            contributions["1C"][:, indices] - contributions["1B"][:, indices]
        ).mean(axis=1)
        for group, group_name in enumerate(
            ["neither", "medial_only", "lateral_only", "both"]
        ):
            pool = np.flatnonzero(code == group)
            choices = [
                (
                    "1B_ranking_gain",
                    pool[np.argsort(-advantage[pool], kind="stable")],
                    2,
                ),
                (
                    "reflection_ranking_gain",
                    pool[np.argsort(advantage[pool], kind="stable")],
                    2,
                ),
                (
                    "1B_1C_divergence",
                    pool[np.argsort(-divergence[pool], kind="stable")],
                    1,
                ),
                ("random_control", rng.permutation(pool), 1),
            ]
            for reason, ordered, count in choices:
                available = [int(i) for i in ordered if int(i) not in used][:count]
                if len(available) != count:
                    raise ValueError(
                        f"Insufficient cases: {family}/{group_name}/{reason}"
                    )
                for i in available:
                    used.add(i)
                    selected.append(
                        {
                            "StudyInstanceUID": val["StudyInstanceUID"][i],
                            "family": family,
                            "group": group_name,
                            "selection": reason,
                            "ranking_advantage_1B_over_reflection": float(advantage[i]),
                            "is_gold": bool(val["is_gold"][i]),
                            "frozen": {t: float(val[t][i]) for t in TARGETS},
                            "predictions": {
                                n: {t: float(p[t][i]) for t in TARGETS}
                                for n, p in predictions.items()
                            },
                        }
                    )
    rng.shuffle(selected)
    reports = dict(
        pl.read_csv("data/raw/train.csv")
        .select("StudyInstanceUID", "Report")
        .iter_rows()
    )
    blind = []
    for i, row in enumerate(selected, 1):
        row["case_id"] = f"C{i:02}"
        blind.append(
            {"case_id": row["case_id"], "report": reports[row["StudyInstanceUID"]]}
        )
    save(args.output / "selection_key.json", selected)
    save(args.output / "blind_reports.json", blind)
    rubric = {
        "targets": TARGETS,
        "positive": "Report explicitly supports a definite meniscal tear or extensive qualifying high-grade compartment cartilage loss/advanced OA.",
        "negative": "Report explicitly describes the relevant meniscus/cartilage as normal or excludes the target finding.",
        "below_threshold": "Explicit degenerative intrameniscal signal without a definite tear, or mild/low-grade cartilage disease without qualifying high-grade loss.",
        "uncertain": "Finding ambiguous, severity/extent or compartment insufficient, or postoperative appearance without clear recurrent tear evidence.",
        "unmentioned": "No interpretable report evidence for this target; absence of mention is not a negative label.",
        "instructions": "Review complete report with model predictions, source labels, selection reason and acquisition identity hidden. Record five states, language, evidence and flags. Freeze reviews.json before unblinding. This is an assistant report-text audit, not independent radiologist image adjudication.",
        "host_source": "https://www.kaggle.com/competitions/rsna-knee-abnormality-detection/discussion/733343",
        "selection": "48 unique studies: two families x four frozen joint-label groups x (two cases favoring each model's ranking, one 1B/1C divergence, one random control). Ranking contribution uses comparisons with opposite-class validation studies.",
    }
    save(args.output / "rubric.json", rubric)
    paths = [
        Path(__file__),
        Path("data/raw/train.csv"),
        Path("data/raw/train_series.csv"),
        Path("data/folds/all_5fold_v3/studies.csv"),
        Path("data/folds/all_5fold_v3/targets.csv"),
    ]
    paths.extend(
        Path("outputs") / run / "fold_0/validation_predictions.csv"
        for run in RUNS.values()
    )
    save(
        args.output / "manifest.json",
        {
            "created_at": datetime.now(UTC).isoformat(),
            "seed": args.seed,
            "hashes": {str(p): sha256_file(p) for p in paths},
        },
    )
    (args.output / Path(__file__).name).write_bytes(Path(__file__).read_bytes())
    print(
        json.dumps({"output": str(args.output), "blinded_cases": len(blind)}),
        flush=True,
    )
    collect_headers(args.output)


def collect_headers(output: Path) -> None:
    series = (
        pl.read_csv("data/raw/train_series.csv")
        .with_columns(
            (
                (pl.col("Anatomical_Plane") == "Coronal").cast(pl.Int8) * 4
                + pl.col("Fluid_Sensitive") * 2
                + pl.col("Fat_Suppression")
            ).alias("priority")
        )
        .sort(
            ["StudyInstanceUID", "priority", "SeriesInstanceUID"],
            descending=[False, True, False],
        )
    )
    counts = series.group_by("StudyInstanceUID").len().rename({"len": "series_count"})
    representatives = series.unique(
        "StudyInstanceUID", keep="first", maintain_order=True
    ).join(counts, on="StudyInstanceUID")
    records = []
    for i, row in enumerate(representatives.iter_rows(named=True), 1):
        directory = (
            Path("data/raw/train_series")
            / row["StudyInstanceUID"]
            / row["SeriesInstanceUID"]
        )
        record = {
            k: row[k]
            for k in [
                "StudyInstanceUID",
                "SeriesInstanceUID",
                "Anatomical_Plane",
                "Fluid_Sensitive",
                "Fat_Suppression",
                "series_count",
            ]
        }
        record.update({tag: None for tag in TAGS})
        try:
            path = min(directory.glob("*.dcm"))
            dataset = pydicom.dcmread(path, stop_before_pixels=True, specific_tags=TAGS)
            record.update(
                {
                    tag: None if dataset.get(tag) is None else str(dataset.get(tag))
                    for tag in TAGS
                }
            )
            record.update(
                {"path": str(path), "file_bytes": path.stat().st_size, "error": None}
            )
        except (OSError, ValueError, pydicom.errors.InvalidDicomError) as exc:
            record.update({"path": None, "file_bytes": None, "error": str(exc)})
        manufacturer = (record["Manufacturer"] or "missing").upper()
        record["vendor"] = next(
            (
                normalized
                for token, normalized in [
                    ("SIEMENS", "Siemens"),
                    ("PHILIPS", "Philips"),
                    ("GE", "GE"),
                    ("TOSHIBA", "Toshiba"),
                    ("HITACHI", "Hitachi"),
                    ("CANON", "Canon"),
                ]
                if token in manufacturer
            ),
            manufacturer,
        )
        record["scanner_group"] = scanner_name(
            record["vendor"], record["MagneticFieldStrength"]
        )
        records.append(record)
        if i % 1000 == 0:
            print(json.dumps({"headers_read": i}), flush=True)
    pl.from_dicts(records, infer_schema_length=None).write_parquet(
        output / "study_headers.parquet"
    )
    print(
        json.dumps(
            {
                "headers_complete": len(records),
                "errors": sum(r["error"] is not None for r in records),
            }
        ),
        flush=True,
    )


def summarize(args: AuditCLI) -> None:
    review_path = args.output / "reviews.json"
    reviews = json.loads(review_path.read_text())
    key = json.loads((args.output / "selection_key.json").read_text())
    if {r["case_id"] for r in reviews} != {r["case_id"] for r in key} or len(
        reviews
    ) != len(key):
        raise ValueError("Review IDs must match the complete blinded packet")
    if any(
        set(r["labels"]) != set(TARGETS) or not set(r["labels"].values()) <= STATES
        for r in reviews
    ):
        raise ValueError("Every reviewed case requires five valid target states")
    lock = args.output / "review_lock.json"
    if lock.exists():
        if json.loads(lock.read_text())["sha256"] != sha256_file(review_path):
            raise ValueError("Locked annotations changed after unblinding")
    else:
        save(
            lock,
            {
                "locked_at": datetime.now(UTC).isoformat(),
                "sha256": sha256_file(review_path),
            },
        )
    review_by_id = {r["case_id"]: r for r in reviews}
    rows = []
    for case in key:
        review = review_by_id[case["case_id"]]
        for target in TARGETS:
            state = review["labels"][target]
            reviewed = (
                1
                if state == "positive"
                else 0
                if state in {"negative", "below_threshold"}
                else None
            )
            rows.append(
                {
                    "case_id": case["case_id"],
                    "StudyInstanceUID": case["StudyInstanceUID"],
                    "family": case["family"],
                    "group": case["group"],
                    "selection": case["selection"],
                    "language": review["language"],
                    "is_gold": case["is_gold"],
                    "target": target,
                    "state": state,
                    "reviewed_label": reviewed,
                    "frozen_soft": case["frozen"][target],
                    "frozen_binary": int(case["frozen"][target] >= 0.5),
                    "flags": "|".join(review["flags"]),
                    "evidence": review["evidence"],
                    **{name: p[target] for name, p in case["predictions"].items()},
                }
            )
    table = pl.from_dicts(rows)
    table.write_csv(args.output / "unblinded_review.csv")
    source_paths = {
        "steven_v1": Path(
            "data/labels/public/stevenleehans/version-6/llm_labels_full.csv"
        ),
        "steven_v2": Path(
            "data/labels/public/stevenleehans/version-6/llm_labels_v2.csv"
        ),
        "lixin": Path("data/labels/public/lixin73/version-1/labels_llm_gpt56sol.csv"),
    }
    conflicts = table.filter(
        pl.col("reviewed_label").is_not_null()
        & (pl.col("reviewed_label") != pl.col("frozen_binary"))
    )
    lineage = conflicts.select(
        "case_id", "StudyInstanceUID", "target", "is_gold", "frozen_soft", "state"
    )
    for name, path in source_paths.items():
        source = (
            pl.read_csv(path)
            .select("StudyInstanceUID", *TARGETS)
            .unpivot(index="StudyInstanceUID", variable_name="target", value_name=name)
        )
        lineage = lineage.join(
            source, on=["StudyInstanceUID", "target"], validate="m:1"
        )
    lineage.write_csv(args.output / "source_label_conflicts.csv")
    details = {}
    for target in TARGETS:
        t = table.filter(pl.col("target") == target)
        known = t.filter(pl.col("reviewed_label").is_not_null())
        frozen_positive = t.filter(pl.col("frozen_binary") == 1)
        details[target] = {
            "states": t.group_by("state").len().to_dicts(),
            "frozen_positives": len(frozen_positive),
            "frozen_positive_states": frozen_positive.group_by("state")
            .len()
            .to_dicts(),
            "scoreable_report_cases": len(known),
            "report_label_disagreements": int(
                known.filter(pl.col("reviewed_label") != pl.col("frozen_binary")).height
            ),
            "models": {
                name: {
                    "report_auc": auc(
                        known["reviewed_label"].to_numpy(), known[name].to_numpy()
                    ),
                    "frozen_auc_on_same_cases": auc(
                        known["frozen_binary"].to_numpy(), known[name].to_numpy()
                    ),
                }
                for name in RUNS
            },
        }
    frame, predictions, targets = load_data()
    headers = pl.read_parquet(args.output / "study_headers.parquet")
    headers = headers.with_columns(
        pl.Series(
            "scanner_group",
            [
                scanner_name(row["vendor"], row["MagneticFieldStrength"])
                for row in headers.iter_rows(named=True)
            ],
        )
    )
    joined = frame.join(headers, on="StudyInstanceUID", validate="1:1")
    val = joined.filter(pl.col("fold") == 0).sort("StudyInstanceUID")
    exclusion_models = {}
    for name, p in predictions.items():
        values = {}
        for target in targets:
            excluded = conflicts.filter(
                (~pl.col("is_gold")) & (pl.col("target") == target)
            )["StudyInstanceUID"]
            keep = ~val["StudyInstanceUID"].is_in(excluded.implode()).to_numpy()
            values[target] = auc(
                (val[target].to_numpy()[keep] >= 0.5).astype(int),
                p[target].to_numpy()[keep],
            )
        exclusion_models[name] = {
            "macro_auc": float(np.mean(list(values.values()))),
            "targets": values,
        }
    save(
        args.output / "conflict_exclusion_sensitivity.json",
        {
            "method": "Exclude only reviewed report-derived conflicting target cells; retain gold and all remaining frozen labels. Post-hoc enriched sample, not a replacement validation metric.",
            "excluded_cells": conflicts.filter(~pl.col("is_gold")).height,
            "models": exclusion_models,
        },
    )
    strata_rows = []
    distributions = []
    for column in ["vendor", "scanner_group", "PatientSex"]:
        for group in sorted(joined[column].drop_nulls().unique().to_list()):
            for fold_name, subset in [
                ("train", joined.filter(pl.col("fold") != 0)),
                ("validation", val),
            ]:
                group_frame = subset.filter(pl.col(column) == group)
                distributions.append(
                    {
                        "stratifier": column,
                        "group": group,
                        "split": fold_name,
                        "studies": len(group_frame),
                        **{
                            t: float((group_frame[t] >= 0.5).mean())
                            if len(group_frame)
                            else None
                            for t in TARGETS
                        },
                    }
                )
            mask = (val[column] == group).fill_null(False).to_numpy()
            if mask.sum() < 30:
                continue
            for target in targets:
                y = (val[target].to_numpy() >= 0.5).astype(int)[mask]
                row = {
                    "stratifier": column,
                    "group": group,
                    "target": target,
                    "studies": int(mask.sum()),
                    "positives": int(y.sum()),
                }
                row.update(
                    {
                        name: auc(y, p[target].to_numpy()[mask])
                        for name, p in predictions.items()
                    }
                )
                strata_rows.append(row)
    pl.from_dicts(strata_rows).write_csv(args.output / "stratified_auc.csv")
    pl.from_dicts(distributions).write_csv(args.output / "metadata_distributions.csv")
    metadata_prior = {}
    train = joined.filter(pl.col("fold") != 0)
    for column in ["vendor", "scanner_group", "PatientSex"]:
        metadata_prior[column] = {}
        for target in TARGETS:
            rates = dict(
                train.with_columns((pl.col(target) >= 0.5).cast(pl.Float64))
                .group_by(column)
                .agg(pl.col(target).mean())
                .iter_rows()
            )
            default = float((train[target] >= 0.5).mean())
            p = np.array([rates.get(value, default) for value in val[column].to_list()])
            metadata_prior[column][target] = auc(
                (val[target].to_numpy() >= 0.5).astype(int), p
            )
    rng = np.random.default_rng(args.seed + 1)
    vendor_intervals = {}
    vendor_sensitivity = {}
    for vendor in sorted(val["vendor"].unique().to_list()):
        mask = (val["vendor"] == vendor).to_numpy()
        if mask.sum() < 30:
            continue
        n = int(mask.sum())
        weights = rng.multinomial(n, np.full(n, 1 / n), size=2000)
        vendor_intervals[vendor] = {}
        for target in TARGETS:
            y = (val[target].to_numpy()[mask] >= 0.5).astype(int)
            delta = weighted_bootstrap_auc(
                y, predictions["1B"][target].to_numpy()[mask], weights
            ) - weighted_bootstrap_auc(
                y, predictions["reflection"][target].to_numpy()[mask], weights
            )
            finite = delta[np.isfinite(delta)]
            vendor_intervals[vendor][target] = {
                "95_interval": np.quantile(finite, [0.025, 0.975]).tolist(),
                "finite_draws": len(finite),
            }
        for multiplier in [2, 4]:
            w = np.where(mask, float(multiplier), 1.0)
            vendor_sensitivity[f"{vendor}_x{multiplier}"] = {
                name: float(
                    np.mean(
                        [
                            auc(
                                (val[t].to_numpy() >= 0.5).astype(int),
                                p[t].to_numpy(),
                                w,
                            )
                            for t in targets
                        ]
                    )
                )
                for name, p in predictions.items()
            }
    (args.output / "summary_source.py").write_bytes(Path(__file__).read_bytes())
    save(
        args.output / "summary.json",
        {
            "created_at": datetime.now(UTC).isoformat(),
            "review_sha256": sha256_file(review_path),
            "headers_sha256": sha256_file(args.output / "study_headers.parquet"),
            "review_cases": len(reviews),
            "gold_cases": sum(c["is_gold"] for c in key),
            "header_studies": len(headers),
            "header_errors": headers["error"].is_not_null().sum(),
            "target_review": details,
            "training_group_rate_validation_auc": metadata_prior,
            "vendor_1B_minus_reflection_bootstrap": vendor_intervals,
            "vendor_weight_sensitivity_macro_auc": vendor_sensitivity,
            "summary_source_sha256": sha256_file(Path(__file__)),
            "source_label_hashes": {
                str(path): sha256_file(path) for path in source_paths.values()
            },
            "report_conflict_cells": len(conflicts),
            "report_conflict_gold_cells": int(conflicts["is_gold"].sum()),
            "limits": [
                "Model-blinded assistant report review, not radiologist image adjudication",
                "Case selection enriches disagreements; counts and AUC do not estimate population performance",
                "Unknown report states excluded from report-label AUC",
                "Representative header per study does not inventory every acquisition or identify true hospital sites",
                "Metadata associations use report-derived labels and do not prove model shortcut reliance or hidden shift",
                "No GPU use, pixel decoding, model execution, training, inference changes or automated tests",
            ],
        },
    )
    print(
        json.dumps(
            {"output": str(args.output), "cases": len(reviews), "headers": len(headers)}
        )
    )


def main() -> None:
    args = AuditCLI()
    if args.phase == "prepare":
        prepare(args)
    else:
        summarize(args)


if __name__ == "__main__":
    main()
