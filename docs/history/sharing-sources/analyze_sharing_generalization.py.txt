import hashlib
import json
from datetime import UTC, datetime
from pathlib import Path

import numpy as np
import polars as pl
from pydantic import Field
from scipy.stats import pearsonr, spearmanr
from sklearn.metrics import roc_auc_score

from knee_abnormality.cli import NonExitingCLISettings


class AnalysisCLI(NonExitingCLISettings):
    output: Path
    bootstrap_samples: int = Field(default=2000, ge=100)
    seed: int = 20260914


RUNS = {
    "reflection": "spatial-tokens-reflection",
    "1A": "meniscus-classifier",
    "1B": "meniscus-oa-classifiers",
    "1C": "disease-compartment-attention",
    "sigma40": "overnight-20260913T214822Z-74udtekr/runs/02-central-sampling",
}
SUBMISSIONS = {
    "reflection": "knee-reflection-20260913",
    "1A": "knee-1a-dual-gpu-20260914-v2",
    "1B": "knee-meniscus-oa-20260913",
    "1C": "knee-1c-batch4-20260914",
    "sigma40": "knee-sigma40-20260914",
}


def digest(path: Path) -> str:
    return hashlib.file_digest(path.open("rb"), "sha256").hexdigest()


def auc(
    y: np.ndarray, p: np.ndarray, weights: np.ndarray | None = None
) -> float | None:
    if len(np.unique(y)) != 2:
        return None
    return float(roc_auc_score(y, p, sample_weight=weights))


def bce(y: np.ndarray, p: np.ndarray) -> np.ndarray:
    clipped = np.clip(p, 1e-7, 1 - 1e-7)
    return -(y * np.log(clipped) + (1 - y) * np.log1p(-clipped))


def weighted_bootstrap_auc(
    y: np.ndarray, p: np.ndarray, weights: np.ndarray
) -> np.ndarray:
    order = np.argsort(p, kind="stable")
    starts = np.r_[0, np.flatnonzero(np.diff(p[order])) + 1]
    positive = np.add.reduceat(weights[:, order] * y[order], starts, axis=1)
    negative = np.add.reduceat(weights[:, order] * (1 - y[order]), starts, axis=1)
    denominator = positive.sum(axis=1) * negative.sum(axis=1)
    numerator = (positive * (negative.cumsum(axis=1) - 0.5 * negative)).sum(axis=1)
    return np.divide(
        numerator,
        denominator,
        out=np.full(len(weights), np.nan),
        where=denominator > 0,
    )


def correlation(rows: list[dict]) -> dict:
    x = [row["local_auc"] for row in rows]
    y = [row["public_auc"] for row in rows]
    counts = {"agree": 0, "reverse": 0, "public_tie": 0}
    for i in range(len(rows)):
        for j in range(i):
            product = (x[i] - x[j]) * (y[i] - y[j])
            counts[
                "agree" if product > 0 else "reverse" if product < 0 else "public_tie"
            ] += 1
    return {
        "rows": rows,
        "pearson": float(pearsonr(x, y).statistic),
        "spearman": float(spearmanr(x, y).statistic),
        "pairwise": counts,
    }


def main() -> None:
    args = AnalysisCLI()
    output = args.output.resolve()
    output.mkdir(parents=True, exist_ok=False)
    frozen_path = Path("data/folds/all_5fold_v3/targets.csv")
    membership_path = frozen_path.with_name("studies.csv")
    frozen = pl.read_csv(frozen_path)
    targets = frozen.columns[1:]
    frame = (
        pl.read_csv(membership_path)
        .filter(pl.col("fold") == 0)
        .join(frozen, on="StudyInstanceUID", validate="1:1")
        .sort("StudyInstanceUID")
    )
    y_soft = frame.select(targets).to_numpy()
    y = (y_soft >= 0.5).astype(int)
    gold = frame["is_gold"].to_numpy()
    groups, group_index, group_sizes = np.unique(
        frame["report_group"].to_numpy(), return_inverse=True, return_counts=True
    )
    rng = np.random.default_rng(args.seed)
    group_weights = rng.multinomial(
        len(groups), np.full(len(groups), 1 / len(groups)), args.bootstrap_samples
    )[:, group_index].astype(float)
    group_weights = np.vstack([np.ones(len(frame)), group_weights])
    summary = {
        "created_at": datetime.now(UTC).isoformat(),
        "seed": args.seed,
        "bootstrap_samples": args.bootstrap_samples,
        "bootstrap_unit": "normalized report group, paired across models and targets",
        "studies": len(frame),
        "gold_studies": int(gold.sum()),
        "report_groups": len(groups),
        "largest_group_studies": int(group_sizes.max()),
        "hashes": {
            str(frozen_path): digest(frozen_path),
            str(membership_path): digest(membership_path),
            str(Path(__file__)): digest(Path(__file__)),
        },
        "runs": {},
        "target_counts": {},
        "comparisons": {},
    }
    for i, target in enumerate(targets):
        summary["target_counts"][target] = {
            "positive": int(y[:, i].sum()),
            "negative": int((1 - y[:, i]).sum()),
            "exact_half": int((y_soft[:, i] == 0.5).sum()),
            "between_01_09": int(((y_soft[:, i] > 0.1) & (y_soft[:, i] < 0.9)).sum()),
        }
    predictions = {}
    draws = {}
    target_rows = []
    for name, location in RUNS.items():
        run = Path("outputs") / location
        prediction_path = run / "fold_0/validation_predictions.csv"
        prediction = pl.read_csv(prediction_path).sort("StudyInstanceUID")
        if (
            prediction["StudyInstanceUID"].to_list()
            != frame["StudyInstanceUID"].to_list()
        ):
            raise ValueError(f"Study mismatch for {name}")
        p = prediction.select(targets).to_numpy()
        if not np.isfinite(p).all() or not ((p >= 0) & (p <= 1)).all():
            raise ValueError(f"Invalid probabilities for {name}")
        predictions[name] = p
        metrics = {}
        draws[name] = np.column_stack(
            [
                weighted_bootstrap_auc(y[:, i], p[:, i], group_weights)
                for i in range(len(targets))
            ]
        )
        for i, target in enumerate(targets):
            confident = (y_soft[:, i] <= 0.1) | (y_soft[:, i] >= 0.9)
            exact = (y_soft[:, i] == 0) | (y_soft[:, i] == 1)
            keep = group_index != group_sizes.argmax()
            metrics[target] = {
                "auc": auc(y[:, i], p[:, i]),
                "gold_auc": auc(y[gold, i], p[gold, i]),
                "silver_auc": auc(y[~gold, i], p[~gold, i]),
                "confident_auc": auc(y[confident, i], p[confident, i]),
                "confident_studies": int(confident.sum()),
                "exact_binary_auc": auc(y[exact, i], p[exact, i]),
                "group_balanced_auc": auc(
                    y[:, i], p[:, i], 1 / group_sizes[group_index]
                ),
                "without_largest_group_auc": auc(y[keep, i], p[keep, i]),
                "soft_bce": float(bce(y_soft[:, i], p[:, i]).mean()),
                "mean_probability": float(p[:, i].mean()),
                "label_mean": float(y_soft[:, i].mean()),
                "logit_sd": float(
                    np.std(
                        np.log(
                            np.clip(p[:, i], 1e-7, 1 - 1e-7)
                            / np.clip(1 - p[:, i], 1e-7, 1)
                        )
                    )
                ),
            }
            target_rows.append({"model": name, "target": target, **metrics[target]})
        selected = json.loads((run / "fold_0/selected_checkpoint.json").read_text())
        epochs = []
        for path in sorted((run / "fold_0/epochs").glob("metrics_epoch_*.json")):
            value = json.loads(path.read_text())
            epochs.append(
                {
                    "epoch": int(path.stem.rsplit("_", 1)[-1]) + 1,
                    "auc": value["macro_auc"],
                    "bce": value["soft_target_bce"],
                    "gold_auc": value["gold"]["macro_auc"],
                }
            )
            summary["hashes"][str(path)] = digest(path)
        record_path = Path("outputs/submissions") / (SUBMISSIONS[name] + ".json")
        record = json.loads(record_path.read_text())
        history = {
            item["state"]: datetime.fromisoformat(item["at"])
            for item in record["history"]
        }
        summary["runs"][name] = {
            "run": str(run),
            "targets": metrics,
            "aggregate": {
                key: float(np.mean([value[key] for value in metrics.values()]))
                for key in [
                    "auc",
                    "gold_auc",
                    "silver_auc",
                    "confident_auc",
                    "exact_binary_auc",
                    "group_balanced_auc",
                    "without_largest_group_auc",
                    "soft_bce",
                ]
            },
            "epochs": epochs,
            "selected_checkpoint": selected,
            "public_auc": float(record["submission"]["publicScore"]),
            "submission_seconds": (
                history["submission_complete"] - history["submitted"]
            ).total_seconds(),
        }
        for path in [
            prediction_path,
            record_path,
            run / "effective_config.json",
            run / "source_snapshot.zip",
            run / "fold_0/input_identity.json",
        ]:
            summary["hashes"][str(path)] = digest(path)
    for candidate, reference in [
        ("1A", "reflection"),
        ("1B", "reflection"),
        ("1C", "reflection"),
        ("1B", "1A"),
        ("1C", "1B"),
        ("sigma40", "1C"),
    ]:
        difference = draws[candidate] - draws[reference]
        observed = difference[0]
        bootstrap = difference[1:].mean(axis=1)
        summary["comparisons"][candidate + "_minus_" + reference] = {
            "macro_auc_delta": float(observed.mean()),
            "paired_report_group_bootstrap_95_interval": np.quantile(
                bootstrap[np.isfinite(bootstrap)], [0.025, 0.975]
            ).tolist(),
            "finite_draws": int(np.isfinite(bootstrap).sum()),
            "per_target_delta": dict(zip(targets, observed.tolist(), strict=True)),
            "mean_auc_delta_excluding_contusion_fracture": float(observed[:-2].mean()),
            "probability_mean_abs_change": float(
                np.abs(predictions[candidate] - predictions[reference]).mean()
            ),
        }
    rows = [
        {
            "name": name,
            "local_auc": value["aggregate"]["auc"],
            "public_auc": value["public_auc"],
        }
        for name, value in summary["runs"].items()
    ]
    earlier_path = Path("outputs/analysis/1b-public-result-20260913/analysis.json")
    earlier = json.loads(earlier_path.read_text())["updated_fixed_label_local_public"][
        "rows"
    ]
    summary["earlier_correlation_rows"] = earlier
    summary["current_four_correlation"] = correlation(rows[:4])
    summary["current_five_correlation"] = correlation(rows)
    summary["hashes"][str(earlier_path)] = digest(earlier_path)
    summary["limits"] = [
        "Saved predictions only; no training, inference, GPU use or tests",
        "Bootstrap measures fixed-model validation sampling uncertainty, not seed variance or hidden label shift",
        "Confidence subsets change case mix and remain report-derived",
        "Submission time includes queue, evaluation and polling",
    ]
    pl.DataFrame(target_rows).write_csv(output / "target_metrics.csv")
    (output / "analysis.json").write_text(json.dumps(summary, indent=2) + "\n")
    (output / Path(__file__).name).write_bytes(Path(__file__).read_bytes())
    print(
        json.dumps(
            {
                "output": str(output),
                "studies": len(frame),
                "report_groups": len(groups),
                "runs": {
                    name: value["aggregate"] for name, value in summary["runs"].items()
                },
                "comparisons": summary["comparisons"],
            },
            indent=2,
        )
    )


if __name__ == "__main__":
    main()
