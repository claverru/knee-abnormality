import json
from datetime import UTC, datetime
from pathlib import Path

import numpy as np
import polars as pl
from pydantic import Field

from knee_abnormality.cli import NonExitingCLISettings
from scripts.adhoc.analyze_sharing_generalization import (
    RUNS,
    auc,
    digest,
    weighted_bootstrap_auc,
)


class CaseMixCLI(NonExitingCLISettings):
    output: Path
    samples: int = Field(default=2000, ge=100)
    seed: int = 20260915


PAIRS = {
    "meniscus": ("Medial Meniscus", "Lateral Meniscus"),
    "oa": ("Medial OA", "Lateral OA"),
}
COMPARISONS = [
    ("1A", "reflection", -0.001),
    ("1B", "reflection", -0.006),
    ("1C", "1B", 0.003),
    ("sigma40", "1C", 0.003),
]


def main() -> None:
    args = CaseMixCLI()
    args.output.mkdir(parents=True, exist_ok=False)
    target_path = Path("data/folds/all_5fold_v3/targets.csv")
    study_path = target_path.with_name("studies.csv")
    frozen = pl.read_csv(target_path)
    targets = frozen.columns[1:]
    all_studies = pl.read_csv(study_path).join(
        frozen, on="StudyInstanceUID", validate="1:1"
    )
    val = all_studies.filter(pl.col("fold") == 0).sort("StudyInstanceUID")
    y = (val.select(targets).to_numpy() >= 0.5).astype(int)
    output = {
        "created_at": datetime.now(UTC).isoformat(),
        "seed": args.seed,
        "draws": args.samples,
        "hashes": {
            str(path): digest(path)
            for path in [target_path, study_path, Path(__file__)]
        },
        "distributions": {},
        "resampling": {},
        "case_mix": {},
        "conditional_auc": {},
        "limits": [
            "CPU-only saved-prediction analysis; no training, inference, GPU access or tests",
            "390 and 910 approximate public/private sizes from about 1300 studies and a 30/70 split",
            "IID study bootstrap with replacement assumes the empirical local population and ignores report-group dependence",
            "Resampling frequencies are sensitivity diagnostics, not probabilities of hidden results or formal significance tests",
            "Scenario weights are hypothetical; hidden target and case-mix distributions are unavailable",
            "All classifications use frozen >=0.5 labels; gold and report-derived populations are not interchangeable",
        ],
    }
    for name, frame in {
        "training_fold0": all_studies.filter(pl.col("fold") != 0),
        "validation_fold0": val,
        "gold_all": all_studies.filter(pl.col("is_gold")),
    }.items():
        soft = frame.select(targets).to_numpy()
        hard = (soft >= 0.5).astype(int)
        joint = {}
        for family, pair in PAIRS.items():
            code = hard[:, targets.index(pair[0])] + 2 * hard[:, targets.index(pair[1])]
            joint[family] = dict(
                zip(
                    ["neither", "medial_only", "lateral_only", "both"],
                    np.bincount(code, minlength=4).tolist(),
                    strict=True,
                )
            )
        output["distributions"][name] = {
            "studies": len(frame),
            "positive_counts": dict(
                zip(targets, hard.sum(axis=0).tolist(), strict=True)
            ),
            "soft_means": dict(zip(targets, soft.mean(axis=0).tolist(), strict=True)),
            "joint": joint,
        }
    rng = np.random.default_rng(args.seed)
    sample_weights = {
        n: rng.multinomial(n, np.full(len(val), 1 / len(val)), size=args.samples)
        for n in [390, 910]
    }
    codes = {
        family: y[:, targets.index(pair[0])] + 2 * y[:, targets.index(pair[1])]
        for family, pair in PAIRS.items()
    }
    scenarios = {"original": np.ones(len(val))}
    for family, code in codes.items():
        counts = np.bincount(code, minlength=4)
        scenarios[family + "_four_groups_equal_mass"] = 1 / counts[code]
        scenarios[family + "_lateral_only_x2"] = np.where(code == 2, 2.0, 1.0)
        scenarios[family + "_lateral_only_x4"] = np.where(code == 2, 4.0, 1.0)
        scenarios[family + "_medial_only_x2"] = np.where(code == 1, 2.0, 1.0)
        scenarios[family + "_any_x0.5"] = np.where(code > 0, 0.5, 1.0)
        scenarios[family + "_any_x2"] = np.where(code > 0, 2.0, 1.0)
    cooccurs = (codes["meniscus"] > 0) & (codes["oa"] > 0)
    scenarios["meniscus_oa_cooccurrence_x0.5"] = np.where(cooccurs, 0.5, 1.0)
    scenarios["meniscus_oa_cooccurrence_x2"] = np.where(cooccurs, 2.0, 1.0)
    conditions = {}
    for family, pair in PAIRS.items():
        other = "oa" if family == "meniscus" else "meniscus"
        for target, counterpart in [pair, pair[::-1]]:
            for status in [0, 1]:
                conditions[(target, "opposite_compartment", status)] = (
                    y[:, targets.index(counterpart)] == status
                )
                conditions[(target, "other_disease_present", status)] = (
                    codes[other] > 0
                ) == status
    draws = {n: {} for n in sample_weights}
    for name, location in RUNS.items():
        path = Path("outputs") / location / "fold_0/validation_predictions.csv"
        predictions = pl.read_csv(path).sort("StudyInstanceUID")
        if (
            predictions["StudyInstanceUID"].to_list()
            != val["StudyInstanceUID"].to_list()
        ):
            raise ValueError(f"Study mismatch: {name}")
        p = predictions.select(targets).to_numpy()
        if not np.isfinite(p).all() or not ((p >= 0) & (p <= 1)).all():
            raise ValueError(f"Invalid probabilities: {name}")
        output["hashes"][str(path)] = digest(path)
        for n, weights in sample_weights.items():
            draws[n][name] = np.column_stack(
                [
                    weighted_bootstrap_auc(y[:, i], p[:, i], weights)
                    for i in range(len(targets))
                ]
            ).mean(axis=1)
        output["case_mix"][name] = {}
        for scenario, weights in scenarios.items():
            values = [auc(y[:, i], p[:, i], weights) for i in range(len(targets))]
            output["case_mix"][name][scenario] = {
                "macro_auc": float(np.mean(values)),
                "targets": dict(zip(targets, values, strict=True)),
                "effective_studies": float(weights.sum() ** 2 / (weights**2).sum()),
            }
        output["conditional_auc"][name] = {}
        for (target, condition, status), mask in conditions.items():
            i = targets.index(target)
            output["conditional_auc"][name][f"{target}/{condition}/{status}"] = {
                "studies": int(mask.sum()),
                "positives": int(y[mask, i].sum()),
                "auc": auc(y[mask, i], p[mask, i]),
            }
    for n, models in draws.items():
        output["resampling"][n] = {}
        for candidate, reference, public_delta in COMPARISONS:
            delta = models[candidate] - models[reference]
            delta = delta[np.isfinite(delta)]
            output["resampling"][n][candidate + "_minus_" + reference] = {
                "mean_delta": float(delta.mean()),
                "95_interval": np.quantile(delta, [0.025, 0.975]).tolist(),
                "candidate_wins_fraction": float((delta > 0).mean()),
                "public_delta_from_rounded_scores": public_delta,
                "fraction_at_least_as_far_in_public_direction": float(
                    (
                        (delta <= public_delta)
                        if public_delta < 0
                        else (delta >= public_delta)
                    ).mean()
                ),
                "finite_draws": len(delta),
            }
    (args.output / "analysis.json").write_text(json.dumps(output, indent=2) + "\n")
    (args.output / Path(__file__).name).write_bytes(Path(__file__).read_bytes())
    (args.output / "generalization_analysis_source.py").write_bytes(
        Path("scripts/adhoc/analyze_sharing_generalization.py").read_bytes()
    )
    print(
        json.dumps(
            {"output": str(args.output), "resampling": output["resampling"]}, indent=2
        )
    )


if __name__ == "__main__":
    main()
