import json
import multiprocessing
import sys
from concurrent.futures import ProcessPoolExecutor
from pathlib import Path

import polars as pl
from PIL import Image, ImageDraw

sys.path.insert(0, str(Path(__file__).parent))
from common import OUT, load
from sheets import FONT, projected, tile

SIZE = 170
PER_ROW = 5
HEADER = 16
TABLE = None
BY_STUDY = {}


def caption(row):
    lines = [
        f"{row['series8']} {row['slot_name']} n{row['n_records']}",
        (row["desc"] or "")[:26],
    ]
    if row["offset_mm"] is not None:
        lines.append(
            f"off {row['offset_x']:.0f},{row['offset_y']:.0f},{row['offset_z']:.0f} fo{row['fov_outside_mm']:.0f} or{row['outside_range_mm']:.0f}"
        )
    lines.append(f"fov {row['fov_rows_mm']:.0f}" if row["fov_rows_mm"] is not None else "")
    return lines


def study_block(item):
    study, flagged, note = item
    rows = BY_STUDY[study]
    count = len(rows)
    lines = -(-count // PER_ROW)
    block = Image.new("RGB", (PER_ROW * SIZE, HEADER + lines * SIZE), (0, 0, 0))
    draw = ImageDraw.Draw(block)
    draw.text((3, 2), f"{study[-8:]} {note}"[:140], fill=(120, 220, 255), font=FONT)
    for index, row in enumerate(rows):
        x = (index % PER_ROW) * SIZE
        y = HEADER + (index // PER_ROW) * SIZE
        try:
            image = tile(row["sample_paths"][1], projected(row["SeriesInstanceUID"], study))
            image = image.resize((SIZE, SIZE))
        except Exception as error:
            image = Image.new("RGB", (SIZE, SIZE))
            ImageDraw.Draw(image).text((3, 60), repr(error)[:30], fill=(255, 0, 0), font=FONT)
        block.paste(image, (x, y))
        color = (255, 60, 60) if row["SeriesInstanceUID"] in flagged else (255, 255, 90)
        for line, text in enumerate(caption(row)):
            draw.text((x + 2, y + 1 + 12 * line), text, fill=color, font=FONT, stroke_width=1, stroke_fill=(0, 0, 0))
        if row["SeriesInstanceUID"] in flagged:
            draw.rectangle((x, y, x + SIZE - 1, y + SIZE - 1), outline=(255, 0, 0), width=3)
    return block


def render(items, prefix, max_height=1500, table=None):
    global BY_STUDY
    folder = OUT / "sheets"
    folder.mkdir(exist_ok=True)
    table = load() if table is None else table
    wanted = {item[0] for item in items}
    BY_STUDY = {}
    for row in table.filter(pl.col("StudyInstanceUID").is_in(wanted)).sort("slot", "SeriesInstanceUID").iter_rows(named=True):
        BY_STUDY.setdefault(row["StudyInstanceUID"], []).append(row)
    with ProcessPoolExecutor(16, mp_context=multiprocessing.get_context("fork")) as pool:
        blocks = list(pool.map(study_block, items, chunksize=1))
    sheets, current, height = [], [], 0
    for item, block in zip(items, blocks, strict=True):
        if current and height + block.height > max_height:
            sheets.append(current)
            current, height = [], 0
        current.append((item[0], block))
        height += block.height + 4
    if current:
        sheets.append(current)
    manifest = {}
    for number, chunk in enumerate(sheets):
        canvas = Image.new("RGB", (PER_ROW * SIZE, sum(b.height + 4 for _, b in chunk)), (60, 60, 60))
        offset = 0
        for _, block in chunk:
            canvas.paste(block, (0, offset))
            offset += block.height + 4
        name = folder / f"{prefix}-{number:03d}.png"
        canvas.save(name)
        manifest[name.name] = [study for study, _ in chunk]
    (folder / f"{prefix}-manifest.json").write_text(json.dumps(manifest, indent=1))
    return list(manifest)


if __name__ == "__main__":
    spec = pl.read_parquet(sys.argv[1])
    prefix = sys.argv[2]
    items = [
        (study, set(flagged), note)
        for study, flagged, note in spec.select("StudyInstanceUID", "flagged", "note").iter_rows()
    ]
    for name in render(items, prefix):
        print(name)
