ROOT = Path("/home/claudio/src/knee-abnormality")
sys.path.insert(0, str(ROOT / "src"))

from knee_abnormality.config import load_config

QUEUE = ROOT / "outputs/training/overnight-20260916T214520Z-t8k_kw2z/runs"
RUNS = ("00-control", "01-control-seed-43", "02-control-seed-44")
OUT = ROOT / "outputs/analysis/series-filtering-20260917"
RAW = OUT / "inference"
WORKERS = 12


def variant_specs():
    flags = pl.read_parquet(OUT / "series_flags.parquet").filter((pl.col("split") == "train") & (pl.col("fold") == 0))

    def uids(mask):
        return set(flags.filter(mask)["SeriesInstanceUID"])

    def moves(mask):
        return dict(flags.filter(mask).select("SeriesInstanceUID", "relabel_slot").iter_rows())

    drop_rule = uids(pl.col("action") == "drop")
    contralateral = uids(pl.col("f_contralateral"))
    lowres = uids(pl.col("f_lowres"))
    t1fs = uids(pl.col("f_t1fs_in_t1pd"))
    zero_te = uids(pl.col("f_zero_te"))
    small_fov = uids(pl.col("f_small_fov_axial"))
    oblique = uids(pl.col("f_oblique"))
    relabel_contrast = moves(pl.col("f_fluidfs_in_t1pd_confirmed") | pl.col("f_nonfs_in_fluid_confirmed"))
    relabel_plane = moves(pl.col("f_swapped_plane_study"))
    relabel_all = {**relabel_contrast, **relabel_plane}
    return {
        "drop_rule_far_and_stray": {"drop": drop_rule, "move": {}},
        "drop_contralateral_minor_side": {"drop": contralateral, "move": {}},
        "drop_lowres": {"drop": lowres, "move": {}},
        "drop_t1fs_in_t1pd": {"drop": t1fs, "move": {}},
        "drop_zero_te": {"drop": zero_te, "move": {}},
        "drop_small_fov_axial": {"drop": small_fov, "move": {}},
        "drop_oblique_gt_30": {"drop": oblique, "move": {}},
        "relabel_contrast": {"drop": set(), "move": relabel_contrast},
        "relabel_plane_swap": {"drop": set(), "move": relabel_plane},
        "relabel_all": {"drop": set(), "move": relabel_all},
        "proposed_rule": {"drop": drop_rule, "move": relabel_all},
        "proposed_plus_contralateral": {"drop": drop_rule | contralateral, "move": relabel_all},
        "all_drops_and_relabels": {"drop": drop_rule | contralateral | lowres | t1fs | zero_te, "move": relabel_all},
    }


def apply(index, spec):
    drop, move = spec["drop"], spec["move"]
    result = {}
    for study, item in index.items():
        touched = any(c.uid in drop or c.uid in move for slot in item.slots for c in slot)
        if not touched:
            result[study] = item
            continue
        slots = [[] for _ in item.slots]
        for number, values in enumerate(item.slots):
            for candidate in values:
                if candidate.uid in drop:
                    continue
                slots[move.get(candidate.uid, number)].append(candidate)
        ordered = tuple(tuple(sorted(values, key=lambda c: (c.image_count, c.uid))) for values in slots)
        result[study] = replace(item, slots=ordered)
    return result


def predict(model, config, targets, studies, index, multiplier, device):
    dataset = KneeStudyDataset(studies, index, config.data, targets, False, config.seed, multiplier)
    loader = build_data_loader(dataset, config.trainer, 1, num_workers=WORKERS)
    ids, outputs, presence, sampling = [], [], [], []
    with torch.inference_mode():
        for batch in loader:
            ids.extend(batch["study_uid"])
            presence.append(batch["series_presence"].numpy())
            sampling.extend(
                [sum(meta["triplets"] for meta in item) for item in batch["sampling"]]
            )
            device_batch = move_batch(batch, device)
            with torch.autocast("cuda", dtype=torch.bfloat16):
                outputs.append(torch.sigmoid(model(device_batch).float()).cpu())
    return ids, torch.cat(outputs).numpy(), np.concatenate(presence), sampling


def run(name, specs):
    started = time.perf_counter()
    run_dir = QUEUE / name
    config = load_config(run_dir / "effective_config.json")
    labels = load_fold_labels(config, 0)
    targets = labels.targets
    validation = labels.validation
    selected = json.loads((run_dir / "fold_0/selected_checkpoint.json").read_text())
    checkpoint = run_dir / "fold_0" / Path(selected["path"]).name
    device = torch.device("cuda")
    model = load_checkpoint_model(config, targets, checkpoint, device, expected_fold=0)
    multiplier = prediction_image_multiplier(config, "validation")
    index = build_series_index(config.paths.data_root, config.data, "train")
    saved = pl.read_csv(run_dir / "fold_0/validation_predictions.csv")
    if saved[STUDY].to_list() != validation[STUDY].to_list():
        raise ValueError("Saved prediction order differs")
    base = saved.select(targets).to_numpy()
    order = {uid: i for i, uid in enumerate(validation[STUDY].to_list())}
    members = set(order)
    results, report = {}, {}
    affected_all = set()
    modified = {}
    for variant, spec in specs.items():
        changed = apply(index, spec)
        affected = sorted(uid for uid in members if changed[uid] is not index[uid])
        modified[variant] = (changed, affected)
        affected_all.update(affected)
    union = sorted(affected_all, key=order.get)
    ids, intact, intact_presence, intact_triplets = predict(
        model, config, targets, validation.filter(pl.col(STUDY).is_in(union)), index, multiplier, device
    )
    rows = [order[uid] for uid in ids]
    reproduction = float(np.abs(intact - base[rows]).max())
    print(name, "intact recompute vs saved max abs", reproduction, "studies", len(ids), flush=True)
    intact_lookup = {uid: (intact[i], intact_presence[i], intact_triplets[i]) for i, uid in enumerate(ids)}
    base = base.copy()
    base[rows] = intact
    for variant, (changed, affected) in modified.items():
        values = base.copy()
        detail = {"affected_studies": len(affected)}
        if affected:
            v_ids, v_pred, v_presence, v_triplets = predict(
                model, config, targets, validation.filter(pl.col(STUDY).is_in(affected)), changed, multiplier, device
            )
            for i, uid in enumerate(v_ids):
                values[order[uid]] = v_pred[i]
            deltas = np.abs(np.stack([v_pred[i] - intact_lookup[uid][0] for i, uid in enumerate(v_ids)]))
            detail.update(
                mean_abs_delta=float(deltas.mean()),
                max_abs_delta=float(deltas.max()),
                emptied_slots=int(sum((intact_lookup[uid][1] & ~v_presence[i]).sum() for i, uid in enumerate(v_ids))),
                filled_slots=int(sum((~intact_lookup[uid][1] & v_presence[i]).sum() for i, uid in enumerate(v_ids))),
                triplets_before=int(sum(intact_lookup[uid][2] for uid in v_ids)),
                triplets_after=int(sum(v_triplets)),
            )
        results[variant] = values
        report[variant] = detail
        print(name, variant, json.dumps(detail), flush=True)
    RAW.mkdir(parents=True, exist_ok=True)
    np.savez_compressed(
        RAW / f"{name}.npz",
        study=np.asarray(validation[STUDY].to_list()),
        targets=np.asarray(targets),
        variants=np.asarray(["intact", *results]),
        predictions=np.stack([base, *results.values()]),
    )
    return {
        "run": name,
        "checkpoint": str(checkpoint),
        "seed": config.seed,
        "validation_image_multiplier": multiplier,
        "intact_recompute_max_abs_vs_saved": reproduction,
        "variants": report,
        "seconds": time.perf_counter() - started,
    }


def main():
    configure_cuda_allocator()
    torch.set_float32_matmul_precision("high")
    torch.backends.cudnn.benchmark = False
    specs = variant_specs()
    summary = {
        variant: {"dropped": sorted(spec["drop"]), "moved": spec["move"]}
        for variant, spec in specs.items()
    }
    reports = [run(name, specs) for name in RUNS]
    RAW.mkdir(parents=True, exist_ok=True)
    (RAW / "inference.json").write_text(
        json.dumps(
            {
                "runs": reports,
                "variant_series": summary,
                "precision": "bf16 autocast",
                "removal": "before preparation: series removed from or moved within the series index; study reference re-estimated",
                "torch": torch.__version__,
                "gpu": torch.cuda.get_device_name(0),
            },
            indent=1,
        )
        + "\n"
    )


if __name__ == "__main__":
    main()
