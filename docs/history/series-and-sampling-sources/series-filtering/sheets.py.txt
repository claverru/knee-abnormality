import sys
from concurrent.futures import ProcessPoolExecutor
from pathlib import Path

import numpy as np
import polars as pl
from PIL import Image, ImageDraw, ImageFont

ROOT = Path("/home/claudio/src/knee-abnormality")
sys.path.insert(0, str(ROOT / "src"))

from knee_abnormality.data.preprocessing.dicom import read_frame

OUT = ROOT / "outputs/analysis/series-filtering-20260917"
TILE = 190
HEADER = 30
ROWS_PER_SHEET = 6
FONT = ImageFont.load_default(size=11)
PATHS = {
    r[0]: r[1:]
    for r in pl.read_parquet(OUT / "census_series.parquet")
    .select("SeriesInstanceUID", "sample_paths", "middle_position", "middle_orientation", "pixel_spacing")
    .rows()
}
POINTS = dict(pl.read_parquet(OUT / "robust_studies.parquet").select("StudyInstanceUID", "robust_point").rows())


def tile(path, marker=None):
    frame = read_frame(Path(path), None)
    pixels = frame.pixels.astype(np.float32)
    lo, hi = np.quantile(pixels, [0.01, 0.995])
    scaled = np.clip((pixels - lo) / max(hi - lo, 1e-6), 0, 1)
    height_mm = pixels.shape[0] * frame.row_spacing
    width_mm = pixels.shape[1] * frame.column_spacing
    image = Image.fromarray((scaled * 255).astype(np.uint8))
    factor = TILE / max(height_mm, width_mm)
    size = (max(1, round(width_mm * factor)), max(1, round(height_mm * factor)))
    image = image.resize(size, Image.BILINEAR).convert("RGB")
    canvas = Image.new("RGB", (TILE, TILE), (40, 0, 40))
    canvas.paste(image, ((TILE - size[0]) // 2, (TILE - size[1]) // 2))
    if marker is not None:
        u, v = marker
        x = (TILE - size[0]) // 2 + u * factor
        y = (TILE - size[1]) // 2 + v * factor
        draw = ImageDraw.Draw(canvas)
        x = min(max(x, 3), TILE - 4)
        y = min(max(y, 3), TILE - 4)
        draw.ellipse((x - 3, y - 3, x + 3, y + 3), outline=(255, 0, 0), width=2)
    return canvas


def projected(uid, study):
    point = POINTS.get(study)
    info = PATHS.get(uid)
    if point is None or info is None or info[1] is None:
        return None
    _, position, orientation, spacing = info
    axes = np.asarray(orientation).reshape(2, 3)
    delta = np.asarray(point) - np.asarray(position)
    return float(np.dot(delta, axes[0])), float(np.dot(delta, axes[1]))


def render_row(item):
    uid, study, title, reference = item
    row = Image.new("RGB", (4 * TILE, TILE + HEADER), (0, 0, 0))
    draw = ImageDraw.Draw(row)
    for line, text in enumerate(title.split("\n")[:2]):
        draw.text((3, 2 + line * 13), text[:130], fill=(255, 255, 90), font=FONT)
    paths = PATHS[uid][0]
    marker = projected(uid, study)
    for column, path in enumerate(paths):
        try:
            row.paste(tile(path, marker if column == 1 else None), (column * TILE, HEADER))
        except Exception as error:
            draw.text((column * TILE + 5, HEADER + 5), repr(error)[:30], fill=(255, 0, 0), font=FONT)
    if reference is not None and reference in PATHS:
        try:
            row.paste(tile(PATHS[reference][0][1], projected(reference, study)), (3 * TILE, HEADER))
            draw.text((3 * TILE + 3, HEADER + 2), "REF", fill=(0, 255, 0), font=FONT)
        except Exception:
            pass
    draw.line((0, TILE + HEADER - 1, 4 * TILE, TILE + HEADER - 1), fill=(90, 90, 90))
    return row


def render(items, prefix):
    folder = OUT / "sheets"
    folder.mkdir(exist_ok=True)
    with ProcessPoolExecutor(16) as pool:
        rows = list(pool.map(render_row, items, chunksize=2))
    names = []
    for start in range(0, len(rows), ROWS_PER_SHEET):
        chunk = rows[start : start + ROWS_PER_SHEET]
        sheet = Image.new("RGB", (4 * TILE, len(chunk) * (TILE + HEADER)))
        for index, row in enumerate(chunk):
            sheet.paste(row, (0, index * (TILE + HEADER)))
        name = folder / f"{prefix}-{start // ROWS_PER_SHEET:03d}.png"
        sheet.save(name)
        names.append(name)
    return names


if __name__ == "__main__":
    table = pl.read_parquet(sys.argv[1])
    prefix = sys.argv[2]
    items = list(table.select("SeriesInstanceUID", "StudyInstanceUID", "title", "reference").iter_rows())
    for name in render(items, prefix):
        print(name)
