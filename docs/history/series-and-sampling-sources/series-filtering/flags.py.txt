import numpy as np
import polars as pl

from common import OUT, load

SIDE_GAP_MM = 150.0
FAR_OUTSIDE_MM = 15.0
FAR_FOV_OUTSIDE_MM = 20.0
FAR_OFFSET_MM = 90.0
LOWRES_MM = 1.0
OBLIQUE_DEG = 30.0


def sides(frame):
    rows = []
    for (study,), group in frame.group_by("StudyInstanceUID"):
        x = group["mx"].to_numpy()
        uids = group["SeriesInstanceUID"].to_list()
        planes = group["plane"].to_list()
        dropped = group["robust_dropped"].to_list()
        if x.max() - x.min() <= SIDE_GAP_MM:
            rows.extend({"SeriesInstanceUID": u, "side_cluster": "single", "bilateral_minor": False, "bilateral_exam": False} for u in uids)
            continue
        order = np.sort(x)
        cut = order[int(np.argmax(np.diff(order)))] + 1e-6
        right = x > cut
        count_right, count_left = int(right.sum()), int((~right).sum())
        drop_right = sum(d for d, r in zip(dropped, right, strict=True) if r)
        drop_left = sum(d for d, r in zip(dropped, right, strict=True) if not r)
        if count_right != count_left:
            minor_is_right = count_right < count_left
        else:
            minor_is_right = drop_right >= drop_left
        minor = right if minor_is_right else ~right
        minor_planes = {p for p, m in zip(planes, minor, strict=True) if m}
        exam = len(minor_planes) >= 2 and int(minor.sum()) >= 3
        for u, m in zip(uids, minor, strict=True):
            rows.append({"SeriesInstanceUID": u, "side_cluster": "minor" if m else "major", "bilateral_minor": bool(m), "bilateral_exam": exam})
    return pl.DataFrame(rows)


def main():
    frame = (
        load()
        .join(pl.read_parquet(OUT / "series_mid_centres.parquet"), on="SeriesInstanceUID")
        .join(pl.read_parquet(OUT / "contrast_signals.parquet"), on="SeriesInstanceUID", how="left")
        .join(pl.read_parquet(OUT / "plane_signals.parquet"), on="SeriesInstanceUID", how="left")
    )
    frame = frame.join(sides(frame), on="SeriesInstanceUID")
    headers_present = pl.col("te").is_not_null() | pl.col("h_ScanOptions").is_not_null()
    fs_header = pl.col("header_class") == "fatsup"
    t1_timing = pl.col("weighting") == "t1_fatsup"
    frame = frame.with_columns(
        (pl.col("pixel_spacing").list.max()).alias("pixel_max_mm"),
        pl.max_horizontal("fov_rows_mm", "fov_cols_mm").alias("fov_max_mm"),
        pl.min_horizontal("fov_rows_mm", "fov_cols_mm").alias("fov_min_mm"),
    ).with_columns(
        f_contralateral=pl.col("bilateral_minor"),
        f_stray_contralateral=pl.col("bilateral_minor") & ~pl.col("bilateral_exam"),
        f_far=(
            pl.col("robust_accepted")
            & ~pl.col("bilateral_minor")
            & (pl.col("pixel_max_mm") <= LOWRES_MM)
            & (
                (pl.col("outside_range_mm") > FAR_OUTSIDE_MM)
                | (pl.col("fov_outside_mm") > FAR_FOV_OUTSIDE_MM)
                | (pl.col("offset_mm") > FAR_OFFSET_MM)
            )
        ).fill_null(False),
        f_robust_dropped=pl.col("robust_dropped") & pl.col("robust_accepted"),
        f_projects_outside=(pl.col("robust_accepted") & ((pl.col("outside_range_mm") > 0) | (pl.col("fov_outside_mm") > 0))).fill_null(False),
        f_study_rejected=~pl.col("robust_accepted"),
        f_lowres=pl.col("pixel_max_mm") > LOWRES_MM,
        f_small_fov_axial=(pl.col("plane") == "Axial") & (pl.col("fov_min_mm") < 110),
        f_oblique=pl.col("obliquity_deg") > OBLIQUE_DEG,
        f_desc_plane=(pl.col("desc_plane").is_in(["Axial", "Coronal", "Sagittal"]) & (pl.col("desc_plane") != pl.col("plane"))),
        f_t1fs_in_t1pd=(pl.col("fluid") == 0) & fs_header & t1_timing,
        f_fluidfs_in_t1pd=(pl.col("fluid") == 0) & fs_header & ~t1_timing,
        f_fluidfs_in_t1pd_confirmed=(pl.col("fluid") == 0) & fs_header & ~t1_timing & (pl.col("p_intensity") > 0.5),
        f_nonfs_in_fluid=(pl.col("fluid") == 1) & ~fs_header & headers_present,
        f_nonfs_in_fluid_confirmed=(pl.col("fluid") == 1) & ~fs_header & headers_present & (pl.col("p_intensity") < 0.5),
        f_intensity_disagrees=((pl.col("fluid") == 0) & (pl.col("p_intensity") > 0.9)) | ((pl.col("fluid") == 1) & (pl.col("p_intensity") < 0.1)),
        f_t2_nonfs_in_t1pd=(pl.col("fluid") == 0) & (pl.col("weighting") == "t2_nonfs"),
        f_zero_te=pl.col("te") < 1,
        f_3d=pl.col("h_MRAcquisitionType") == "3D",
        f_post_contrast=pl.col("desc").str.contains(r"(?i)\+c|post\b|_post") | pl.col("h_ContrastBolusAgent").is_in(["Dotarem", "gadovist"]),
        f_reformat=pl.col("h_ImageType").str.contains("PROJECTION IMAGE") & pl.col("desc").str.contains("MPR"),
        f_swapped_plane_study=pl.col("StudyInstanceUID").is_in(
            frame.filter(
                pl.col("desc_plane").is_in(["Axial", "Coronal", "Sagittal"])
                & (pl.col("desc_plane") != pl.col("plane"))
                & (pl.col("obliquity_deg") < 25)
            )
            .group_by("StudyInstanceUID")
            .agg(pl.len().alias("n"), pl.col("plane").n_unique().alias("planes"))
            .filter((pl.col("n") >= 3) & (pl.col("planes") >= 2))["StudyInstanceUID"]
            .implode()
        )
        & (pl.col("desc_plane").is_in(["Axial", "Coronal", "Sagittal"]) & (pl.col("desc_plane") != pl.col("plane"))),
    )
    frame = frame.with_columns(
        action=pl.when(pl.col("f_stray_contralateral") | pl.col("f_far"))
        .then(pl.lit("drop"))
        .when(pl.col("f_fluidfs_in_t1pd_confirmed") | pl.col("f_nonfs_in_fluid_confirmed") | pl.col("f_swapped_plane_study"))
        .then(pl.lit("relabel"))
        .otherwise(pl.lit("keep")),
        relabel_slot=pl.when(pl.col("f_fluidfs_in_t1pd_confirmed"))
        .then(pl.col("slot") + 1)
        .when(pl.col("f_nonfs_in_fluid_confirmed"))
        .then(pl.col("slot") - 1)
        .when(pl.col("f_swapped_plane_study"))
        .then(
            pl.col("desc_plane").replace_strict({"Axial": 0, "Coronal": 2, "Sagittal": 4}, return_dtype=pl.Int64)
            + pl.col("fluid")
        )
        .otherwise(pl.col("slot")),
    )
    keep = [c for c in frame.columns if c.startswith("f_")] + [
        "SeriesInstanceUID", "StudyInstanceUID", "split", "fold", "report_group", "slot", "slot_name", "plane", "fluid",
        "series8", "study8", "desc", "action", "relabel_slot", "side_cluster", "bilateral_exam", "pixel_max_mm", "fov_min_mm",
        "fov_max_mm", "offset_mm", "offset_z", "outside_range_mm", "fov_outside_mm", "obliquity_deg", "weighting",
        "header_class", "p_intensity", "p_both", "te", "tr", "n_records", "h_Manufacturer",
    ]
    frame.select(keep).write_parquet(OUT / "series_flags.parquet")
    pl.Config.set_tbl_rows(60)
    pl.Config.set_tbl_width_chars(200)
    print(frame.group_by("split", "action").len().sort("split", "action"))
    print(frame.select([pl.col(c).sum() for c in frame.columns if c.startswith("f_")]).transpose(include_header=True))


if __name__ == "__main__":
    main()
