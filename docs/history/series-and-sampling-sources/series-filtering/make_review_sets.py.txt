import polars as pl

from common import OUT, load

frame = load().join(pl.read_parquet(OUT / "contrast_signals.parquet"), on="SeriesInstanceUID", how="left").join(
    pl.read_parquet(OUT / "plane_signals.parquet"), on="SeriesInstanceUID", how="left"
)
frame = frame.with_columns(
    ((pl.col("header_class") == "fatsup").cast(pl.Int64) != pl.col("fat")).alias("hdr_dis"),
    (((pl.col("fat") == 0) & (pl.col("p_intensity") > 0.9)) | ((pl.col("fat") == 1) & (pl.col("p_intensity") < 0.1))).alias("int_dis"),
    (pl.col("desc_plane").is_in(["Axial", "Coronal", "Sagittal"]) & (pl.col("desc_plane") != pl.col("plane"))).alias("desc_plane_mismatch"),
)
train = frame.filter(pl.col("split") == "train")


def reference(row, contrast):
    same = frame.filter(
        (pl.col("StudyInstanceUID") == row["StudyInstanceUID"])
        & (pl.col("plane") == row["plane"])
        & (pl.col("SeriesInstanceUID") != row["SeriesInstanceUID"])
    )
    if contrast:
        other = same.filter(pl.col("fat") != row["fat"])
        same = other if other.height else same
    if same.height == 0:
        same = frame.filter((pl.col("StudyInstanceUID") == row["StudyInstanceUID"]) & (pl.col("SeriesInstanceUID") != row["SeriesInstanceUID"]))
    return same.sort(pl.col("inplane_mm").fill_null(999)).row(0, named=True)["SeriesInstanceUID"] if same.height else None


def rows(selection, tag, contrast):
    out = []
    for row in selection.iter_rows(named=True):
        title = (
            f"{tag} {row['study8']} {row['series8']} LABEL {row['slot_name']} n{row['n_records']} {row['desc'][:28]} {row['h_Manufacturer'] or ''}\n"
            f"TE{row['te']} TR{row['tr']} TI{row['ti']} flip{row['flip']} opt {(row['h_ScanOptions'] or '')[:20]} seq {row['h_ScanningSequence']} w={row['weighting']} pInt {row['p_intensity']:.2f} obl {row['obliquity_deg'] or 0:.0f}"
        )
        out.append({"SeriesInstanceUID": row["SeriesInstanceUID"], "StudyInstanceUID": row["StudyInstanceUID"], "title": title, "reference": reference(row, contrast), "tag": tag})
    return out


both = train.filter(pl.col("hdr_dis") & pl.col("int_dis"))
header_only = train.filter(pl.col("hdr_dis") & ~pl.col("int_dis")).sample(fraction=1.0, shuffle=True, seed=0).group_by("slot_name", "weighting", maintain_order=True).head(4)
intensity_only = train.filter(~pl.col("hdr_dis") & pl.col("int_dis")).sample(fraction=1.0, shuffle=True, seed=0).group_by("slot_name", "weighting", maintain_order=True).head(6)
contrast_rows = rows(both, "C-BOTH", True) + rows(header_only, "C-HDR", True) + rows(intensity_only, "C-INT", True)
pl.DataFrame(contrast_rows).write_parquet(OUT / "rows_contrast.parquet")
print("contrast", both.height, header_only.height, intensity_only.height)

mismatch = train.filter(pl.col("desc_plane_mismatch"))
low = mismatch.filter(pl.col("obliquity_deg").fill_null(0) < 25)
high = mismatch.filter(pl.col("obliquity_deg").fill_null(0) >= 25).sample(n=8, seed=0)
oblique = train.filter((pl.col("obliquity_deg") > 30) & ~pl.col("desc_plane_mismatch")).sample(n=12, seed=0)
plane_rows = rows(low, "P-LOW", False) + rows(high, "P-HIGH", False) + rows(oblique, "P-OBL", False)
pl.DataFrame(plane_rows).write_parquet(OUT / "rows_plane.parquet")
print("plane", low.height, high.height, oblique.height)

special = train.filter(pl.col("te") < 1).sample(n=6, seed=0)
thin = train.filter(pl.col("slab_mm") < 25).sample(n=6, seed=0)
contrast_agent = train.filter(pl.col("desc").str.contains(r"(?i)\+c|post|pre\b") | pl.col("h_ContrastBolusAgent").is_in(["Dotarem", "gadovist"]))
special_rows = rows(special, "S-UTE", False) + rows(thin, "S-THIN", False) + rows(contrast_agent, "S-GAD", True)
pl.DataFrame(special_rows).write_parquet(OUT / "rows_special.parquet")
print("special", special.height, thin.height, contrast_agent.height)
frame.write_parquet(OUT / "signals_all.parquet")
