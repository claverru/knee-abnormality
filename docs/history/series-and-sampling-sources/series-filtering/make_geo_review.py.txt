import polars as pl

from common import OUT, load

f = load()
geo = (
    pl.col("robust_dropped")
    | (pl.col("fov_outside_mm") > 0)
    | (pl.col("outside_range_mm") > 0)
    | (pl.col("offset_mm") > 80)
)
flagged = f.filter(geo.fill_null(False))
studies = (
    flagged.group_by("StudyInstanceUID")
    .agg(
        pl.col("SeriesInstanceUID").alias("flagged"),
        pl.col("robust_accepted").first(),
        pl.col("split").first(),
        pl.col("fold").first(),
    )
    .with_columns(
        pl.format(
            "geo-flag accepted={} fold={} split={}", "robust_accepted", "fold", "split"
        ).alias("note")
    )
)
rejected = (
    f.filter(~pl.col("robust_accepted"))
    .group_by("StudyInstanceUID")
    .agg(pl.col("robust_accepted").first(), pl.col("split").first(), pl.col("fold").first())
    .with_columns(
        pl.lit([], dtype=pl.List(pl.String)).alias("flagged"),
        pl.format("ROBUST-REJECTED fold={}", "fold").alias("note"),
    )
)
tests = (
    f.filter(pl.col("split") == "test")
    .group_by("StudyInstanceUID")
    .agg(pl.col("robust_accepted").first(), pl.col("split").first(), pl.col("fold").first())
    .with_columns(pl.lit([], dtype=pl.List(pl.String)).alias("flagged"), pl.lit("TEST example").alias("note"))
)
spec = pl.concat([studies, rejected.select(studies.columns), tests.select(studies.columns)]).unique("StudyInstanceUID", keep="first").sort("note", "StudyInstanceUID")
print(spec.height, spec.group_by("robust_accepted", "split").len())
spec.write_parquet(OUT / "review_geo_spec.parquet")
