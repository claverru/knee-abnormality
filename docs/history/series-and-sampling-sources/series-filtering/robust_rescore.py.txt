@dataclass(frozen=True, slots=True)
class RobustReference(StudyReference):
    mode: str = ""
    dropped: tuple[tuple[str, float], ...] = ()


def fit(usable):
    counts = Counter(plane for _, plane, _ in usable)
    if len(counts) < 2:
        return {"reason": "insufficient_planes"}
    matrix = np.zeros((3, 3))
    rhs = np.zeros(3)
    parts = []
    for uid, plane, geometry in usable:
        unit = np.asarray(geometry.normal) / np.linalg.norm(geometry.normal)
        projection = np.eye(3) - np.outer(unit, unit)
        weight = 1 / counts[plane]
        matrix += weight * projection
        rhs += weight * projection @ np.asarray(geometry.center)
        parts.append((uid, weight, projection, np.asarray(geometry.center)))
    condition = float(np.linalg.cond(matrix))
    if not np.isfinite(condition) or condition > RELIABILITY["maximum_condition"]:
        return {"reason": "ill_conditioned"}
    point = np.linalg.solve(matrix, rhs)
    residuals = {uid: float(np.linalg.norm(p @ (point - c))) for uid, _, p, c in parts}
    rms = float(np.sqrt(sum(w * residuals[uid] ** 2 for uid, w, _, _ in parts) / len(counts)))
    return {"reason": None, "point": point, "residuals": residuals, "rms": rms, "condition": condition}


def outside_range_mm(geometry, point):
    normal = np.asarray(geometry.normal, dtype=float)
    lo, hi = sorted(geometry.bounds)
    depth = float(np.dot(point, normal))
    return max(lo - depth, depth - hi, 0.0) / float(np.linalg.norm(normal))


def fov_outside_mm(indexed, point):
    middle = indexed.records[len(indexed.records) // 2]
    if middle.pose is None:
        return 0.0
    layout = middle.pose.layout
    orientation = np.asarray(layout.orientation, dtype=float).reshape(2, 3)
    delta = point - np.asarray(middle.pose.position, dtype=float)
    u = float(np.dot(delta, orientation[0]))
    v = float(np.dot(delta, orientation[1]))
    width = (layout.columns - 1) * layout.spacing[1]
    height = (layout.rows - 1) * layout.spacing[0]
    return float(np.hypot(max(-u, u - width, 0.0), max(-v, v - height, 0.0)))


def robust_reference(eligible, indexed_by_uid, mode):
    usable = sorted((uid, plane, geometry) for uid, plane, geometry in eligible if geometry.reason is None)
    retained = list(usable)
    dropped = []
    while True:
        result = fit(retained)
        if result["reason"] is not None:
            return None, result["reason"] + ("_after_drop" if dropped else "")
        point = result["point"]
        over = [(r, uid) for uid, r in result["residuals"].items() if r > THRESHOLD]
        if mode == "range_gated":
            geometry = {uid: g for uid, _, g in retained}
            over = [
                (r, uid)
                for r, uid in over
                if outside_range_mm(geometry[uid], point) > 0 or fov_outside_mm(indexed_by_uid[uid], point) > 0
            ]
        if not over:
            break
        residual, worst = max(over)
        dropped.append((worst, residual))
        retained = [item for item in retained if item[0] != worst]
    if result["rms"] > RELIABILITY["maximum_rms_mm"]:
        return None, "excessive_rms"
    point = result["point"]
    residuals = []
    for uid, _, geometry in usable:
        unit = np.asarray(geometry.normal) / np.linalg.norm(geometry.normal)
        offset = point - np.asarray(geometry.center)
        residuals.append((uid, float(np.linalg.norm(offset - np.dot(offset, unit) * unit))))
    frames = tuple(sorted({f for _, _, g in usable for f in g.frame_uids}))
    reference = RobustReference(
        tuple(point),
        tuple(uid for uid, _, _ in usable),
        result["condition"],
        result["rms"],
        tuple(residuals),
        frames,
        None,
        mode,
        tuple(dropped),
    )
    return reference, "accepted"


class RobustDataset(KneeStudyDataset):
    mode = "robust20"

    def _eligible(self, study):
        candidates = {}
        for slot, values in enumerate(study.slots):
            for candidate in values:
                candidates.setdefault(candidate.uid, (candidate, self.data.series_types[slot].plane))
        eligible, indexed = [], {}
        for uid, (candidate, plane) in sorted(candidates.items()):
            try:
                records = self._records(candidate)
            except PREPARATION_ERRORS:
                continue
            if len(records) >= SLICES_PER_INPUT:
                eligible.append((uid, plane, self._slice_index[candidate.directory].geometry))
                indexed[uid] = self._slice_index[candidate.directory]
        return eligible, indexed

    def _index_study_geometry(self, study):
        if study.uid in self._study_references:
            return
        super()._index_study_geometry(study)
        if self._study_references[study.uid].reason is None:
            return
        eligible, indexed = self._eligible(study)
        reference, _ = robust_reference(eligible, indexed, self.mode)
        if reference is not None:
            self._study_references[study.uid] = reference


def gated_build_pool(sources, plane, reference, failed_paths=None, failed_sources=None):
    if not isinstance(reference, RobustReference):
        return build_pool(sources, plane, reference, failed_paths, failed_sources)
    point = np.asarray(reference.point)
    inside, outside = [], []
    for uid, indexed in sources:
        geometry = indexed.geometry
        within = geometry.reason is None and geometry.bounds is not None and outside_range_mm(geometry, point) == 0
        (inside if within else outside).append((uid, indexed))
    local_reference = StudyReference(None, reference.contributors, None, None, (), reference.frame_uids, "outside_acquired_range")
    first = build_pool(tuple(inside), plane, reference, failed_paths, failed_sources)
    second = build_pool(tuple(outside), plane, local_reference, failed_paths, failed_sources)
    diagnostics = {
        key: first.diagnostics[key] + second.diagnostics[key]
        for key in ("singleton_runs", "split_sources", "plane_disagreements", "no_valid_run_sources")
    }
    diagnostics["sources"] = {**first.diagnostics["sources"], **second.diagnostics["sources"]}
    if first.physical:
        candidates, physical = first.candidates, True
        diagnostics["excluded_local_sources"] = sorted(
            uid for uid in {v.uid for v in second.candidates} | {uid for uid, s in first.diagnostics["sources"].items() if s != "physical"}
        )
    else:
        candidates, physical = first.candidates + second.candidates, False
        diagnostics["excluded_local_sources"] = []
    diagnostics["path"] = "physical" if physical else "local" if candidates else "unusable"
    return CandidatePool(candidates, physical, diagnostics)

def rescue_sets(config, index, rejected):
    rows, sets = [], {v: [] for v in VARIANTS}
    for uid in sorted(rejected[STUDY]):
        base = KneeStudyDataset(pl.DataFrame({STUDY: [uid]}), {uid: index[uid]}, config.data, (), False, config.seed)
        base._index_study_geometry(index[uid])
        original = base._study_references[uid]
        row = {STUDY: uid, "current_reason": original.reason, "current_rms_mm": original.rms_mm}
        for variant in VARIANTS:
            dataset = RobustDataset(pl.DataFrame({STUDY: [uid]}), {uid: index[uid]}, config.data, (), False, config.seed)
            dataset.mode = variant
            dataset._slice_index = base._slice_index
            eligible, indexed = dataset._eligible(index[uid])
            reference, reason = robust_reference(eligible, indexed, variant)
            row[f"{variant}_reason"] = reason
            row[f"{variant}_rms_mm"] = reference.rms_mm if reference else None
            row[f"{variant}_dropped"] = len(reference.dropped) if reference else None
            row[f"{variant}_point"] = list(reference.point) if reference else None
            if reference:
                sets[variant].append(uid)
                row[f"{variant}_series_inside"] = sum(
                    outside_range_mm(g, np.asarray(reference.point)) == 0 for _, _, g in eligible if g.reason is None
                )
                row[f"{variant}_series_outside"] = sum(
                    outside_range_mm(g, np.asarray(reference.point)) > 0 for _, _, g in eligible if g.reason is None
                )
        rows.append(row)
    return pl.DataFrame(rows, infer_schema_length=None), sets


def summarize(item):
    series = []
    for slot, (values, metadata) in enumerate(zip(item["series"], item["sampling"], strict=True)):
        prepared = {v.uid: v for v in values}
        excluded = set(metadata.get("excluded_local_sources", []))
        for uid, status in metadata["sources"].items():
            value = prepared.get(uid)
            series.append(
                {
                    "slot": slot,
                    "SeriesInstanceUID": uid,
                    "source_status": status,
                    "slot_path": metadata["path"],
                    "series_path": "excluded" if uid in excluded else value.sampling_path if value else "unused",
                    "triplets": metadata.get("realized_counts", {}).get(uid, 0),
                    "geometry_valid_all": bool(value.geometry_valid.all()) if value else None,
                    "shortfall": metadata["shortfall"],
                    "max_error_mm": max((a["error"] for a in metadata["assignments"] if a["uid"] == uid), default=None),
                    "attempts": metadata["attempts"],
                    "failed_sources": len(metadata["failed_sources"]),
                    "failed_paths": len(metadata["failed_paths"]),
                }
            )
    return {"study_uid": item["study_uid"], "failures": list(item["failures"]), "series": series}


def collect(items):
    return [summarize(item) for item in items]


def sampling_census(dataset_class, config, studies, index, multiplier, mode):
    dataset = dataset_class(studies, index, config.data, (), False, config.seed, multiplier)
    if mode:
        dataset.mode = mode
    loader = torch.utils.data.DataLoader(dataset, batch_size=1, num_workers=WORKERS, collate_fn=collect)
    rows, failures = [], {}
    for batch in loader:
        for item in batch:
            failures[item["study_uid"]] = item["failures"]
            rows.extend({STUDY: item["study_uid"], "variant": mode if dataset_class is RobustDataset else "intact", **s} for s in item["series"])
    return rows, failures


def predict(dataset_class, mode, model, config, targets, studies, index, multiplier, device):
    dataset = dataset_class(studies, index, config.data, targets, False, config.seed, multiplier)
    if mode:
        dataset.mode = mode
    loader = build_data_loader(dataset, config.trainer, 1, num_workers=WORKERS)
    ids, outputs = [], []
    with torch.inference_mode():
        for batch in loader:
            ids.extend(batch["study_uid"])
            device_batch = move_batch(batch, device)
            with torch.autocast("cuda", dtype=torch.bfloat16):
                outputs.append(torch.sigmoid(model(device_batch).float()).cpu())
    return ids, torch.cat(outputs).numpy()


def infer(sets):
    union = sorted(set().union(*sets.values()))
    reports = []
    for name in RUNS:
        started = time.perf_counter()
        run_dir = QUEUE / name
        config = load_config(run_dir / "effective_config.json")
        labels = load_fold_labels(config, 0)
        targets, validation = labels.targets, labels.validation
        selected = json.loads((run_dir / "fold_0/selected_checkpoint.json").read_text())
        checkpoint = run_dir / "fold_0" / Path(selected["path"]).name
        device = torch.device("cuda")
        model = load_checkpoint_model(config, targets, checkpoint, device, expected_fold=0)
        multiplier = prediction_image_multiplier(config, "validation")
        index = build_series_index(config.paths.data_root, config.data, "train")
        saved = pl.read_csv(run_dir / "fold_0/validation_predictions.csv")
        if saved[STUDY].to_list() != validation[STUDY].to_list():
            raise ValueError("Saved prediction order differs")
        order = {uid: i for i, uid in enumerate(validation[STUDY].to_list())}
        base = saved.select(targets).to_numpy()
        ids, intact = predict(KneeStudyDataset, None, model, config, targets, validation.filter(pl.col(STUDY).is_in(union)), index, multiplier, device)
        rows = [order[uid] for uid in ids]
        reproduction = float(np.abs(intact - base[rows]).max())
        base = base.copy()
        base[rows] = intact
        lookup = dict(zip(ids, intact, strict=True))
        stacks, detail = [base], {}
        for variant in VARIANTS:
            values = base.copy()
            v_ids, v_pred = predict(RobustDataset, variant, model, config, targets, validation.filter(pl.col(STUDY).is_in(sets[variant])), index, multiplier, device)
            for i, uid in enumerate(v_ids):
                values[order[uid]] = v_pred[i]
            deltas = np.abs(np.stack([v_pred[i] - lookup[uid] for i, uid in enumerate(v_ids)]))
            detail[variant] = {"studies": len(v_ids), "mean_abs_delta": float(deltas.mean()), "max_abs_delta": float(deltas.max())}
            stacks.append(values)
        print(name, "reproduction", reproduction, json.dumps(detail), round(time.perf_counter() - started), flush=True)
        np.savez_compressed(
            RAW / f"{name}.npz",
            study=np.asarray(validation[STUDY].to_list()),
            targets=np.asarray(targets),
            variants=np.asarray(["intact", *VARIANTS]),
            predictions=np.stack(stacks),
        )
        reports.append({"run": name, "checkpoint": str(checkpoint), "seed": config.seed, "validation_image_multiplier": multiplier, "intact_recompute_max_abs_vs_saved": reproduction, "variants": detail, "seconds": time.perf_counter() - started})
        del model
        torch.cuda.empty_cache()
    return reports


def analyze(sets, reports):
    data = [np.load(RAW / f"{run}.npz") for run in RUNS]
    study = data[0]["study"]
    targets = data[0]["targets"].tolist()
    config = load_config(QUEUE / RUNS[0] / "effective_config.json")
    validation = load_fold_labels(config, 0).validation
    if validation[STUDY].to_list() != study.tolist():
        raise ValueError("Label order differs")
    hard = (validation.select([f"target_{t}" for t in targets]).to_numpy() >= TARGET_THRESHOLD).astype(int)
    groups_frame = pl.read_csv(ROOT / "data/folds/all_5fold_v3/studies.csv").select(STUDY, "report_group")
    groups = validation.select(STUDY).join(groups_frame, on=STUDY, how="left")["report_group"].to_numpy()
    unique, inverse = np.unique(groups, return_inverse=True)
    rng = np.random.default_rng(SEED)
    counts = rng.multinomial(len(unique), np.full(len(unique), 1 / len(unique)), size=DRAWS)[:, inverse]
    positive = counts @ hard
    valid = ((positive > 0) & (positive < counts.sum(axis=1, keepdims=True))).all(axis=1)
    counts = counts[valid]
    full = np.ones((1, len(study)))
    macro_rows, target_rows = [], []
    only = sorted(set(sets["robust20"]) - set(sets["range_gated"]))
    gated_rows = np.isin(study, sets["range_gated"])
    same_centre_max_abs = float(max(np.abs(item["predictions"][1][gated_rows] - item["predictions"][2][gated_rows]).max() for item in data))
    members = {**sets, "robust20_not_range_gated": only}

    def candidate_of(item, variant):
        if variant in VARIANTS:
            return item["predictions"][1 + VARIANTS.index(variant)]
        values = item["predictions"][0].copy()
        rows = np.isin(study, only)
        values[rows] = item["predictions"][1][rows]
        return values

    for variant in (*VARIANTS, "robust20_not_range_gated"):
        point, draws, shifts = [], [], []
        per_target_point = np.zeros((len(RUNS), len(targets)))
        per_target_draws = np.zeros((len(RUNS), len(targets), len(counts)))
        affected = np.isin(study, members[variant])
        for r, item in enumerate(data):
            intact = item["predictions"][0]
            candidate = candidate_of(item, variant)
            shifts.append(np.abs(candidate[affected] - intact[affected]))
            point.append(float(bootstrap_auc(hard, candidate, full)[0] - bootstrap_auc(hard, intact, full)[0]))
            draws.append(bootstrap_auc(hard, candidate, counts) - bootstrap_auc(hard, intact, counts))
            for t in range(len(targets)):
                h, c, i = hard[:, [t]], candidate[:, [t]], intact[:, [t]]
                per_target_point[r, t] = bootstrap_auc(h, c, full)[0] - bootstrap_auc(h, i, full)[0]
                per_target_draws[r, t] = bootstrap_auc(h, c, counts) - bootstrap_auc(h, i, counts)
        mean_draws = np.mean(draws, axis=0)
        macro_rows.append(
            {
                "variant": variant,
                "studies": len(members[variant]),
                "delta_macro_x1000": round(1000 * float(np.mean(point)), 3),
                "ci_low_x1000": round(1000 * float(np.quantile(mean_draws, 0.025)), 3),
                "ci_high_x1000": round(1000 * float(np.quantile(mean_draws, 0.975)), 3),
                "p_delta_le_0": round(float((mean_draws <= 0).mean()), 4),
                **{f"{run}_x1000": round(1000 * v, 3) for run, v in zip(RUNS, point, strict=True)},
                "mean_abs_prob_delta": round(float(np.mean(shifts)), 5),
                "max_abs_prob_delta": round(float(np.max(shifts)), 5),
            }
        )
        for t, target in enumerate(targets):
            md = per_target_draws[:, t].mean(axis=0)
            target_rows.append(
                {
                    "variant": variant,
                    "target": target,
                    "positives_in_affected": int(hard[affected, t].sum()),
                    "delta_x1000": round(1000 * float(per_target_point[:, t].mean()), 3),
                    "ci_low_x1000": round(1000 * float(np.quantile(md, 0.025)), 3),
                    "ci_high_x1000": round(1000 * float(np.quantile(md, 0.975)), 3),
                    **{f"{run}_x1000": round(1000 * float(v), 3) for run, v in zip(RUNS, per_target_point[:, t], strict=True)},
                }
            )
    macro = pl.DataFrame(macro_rows)
    per_target = pl.DataFrame(target_rows)
    macro.write_csv(OUT / "macro_deltas.csv")
    per_target.write_csv(OUT / "per_target_deltas.csv")
    intact_macro = {run: float(bootstrap_auc(hard, item["predictions"][0], full)[0]) for run, item in zip(RUNS, data, strict=True)}
    return macro, per_target, {"robust20_vs_range_gated_max_abs_on_shared_studies": same_centre_max_abs, "intact_macro_auc": intact_macro, "bootstrap": {"unit": "report group", "groups": len(unique), "draws": DRAWS, "valid_draws": int(valid.sum()), "seed": SEED}}


def main():
    started = time.perf_counter()
    OUT.mkdir(parents=True, exist_ok=True)
    RAW.mkdir(parents=True, exist_ok=True)
    configure_cuda_allocator()
    torch.set_float32_matmul_precision("high")
    torch.backends.cudnn.benchmark = False
    config = load_config(QUEUE / RUNS[0] / "effective_config.json")
    index = build_series_index(config.paths.data_root, config.data, "train")
    rejected = rejected_fold0()
    studies_table, sets = rescue_sets(config, index, rejected)
    studies_table.write_parquet(OUT / "studies.parquet")
    census_robust = set(rejected.filter(pl.col("robust_reason") == "accepted")[STUDY])
    followup = pl.read_parquet(CENSUS / "followup/reference_after_drops.parquet")
    census_gated = set(followup.filter(pl.col(STUDY).is_in(rejected[STUDY]) & (pl.col("range_gated_reason") == "accepted"))[STUDY])
    check = {
        "rejected": len(rejected),
        "current_reasons": dict(Counter(studies_table["current_reason"])),
        "robust20": len(sets["robust20"]),
        "robust20_matches_census": set(sets["robust20"]) == census_robust,
        "range_gated": len(sets["range_gated"]),
        "range_gated_matches_followup": set(sets["range_gated"]) == census_gated,
        "range_gated_subset_of_robust20": set(sets["range_gated"]) <= set(sets["robust20"]),
    }
    print(json.dumps(check), flush=True)
    print(studies_table.drop(STUDY, "robust20_point", "range_gated_point"), flush=True)
    labels = load_fold_labels(config, 0)
    multiplier = prediction_image_multiplier(config, "validation")
    union = sorted(set().union(*sets.values()))
    frame = labels.validation.filter(pl.col(STUDY).is_in(union))
    rows, failures = sampling_census(KneeStudyDataset, config, frame, index, multiplier, None)
    for variant in VARIANTS:
        extra, extra_failures = sampling_census(RobustDataset, config, labels.validation.filter(pl.col(STUDY).is_in(sets[variant])), index, multiplier, variant)
        rows.extend(extra)
        failures.update({f"{variant}:{k}": v for k, v in extra_failures.items()})
    series = pl.DataFrame(rows, infer_schema_length=None)
    series.write_parquet(OUT / "series_sampling.parquet")
    paths = series.group_by("variant", "series_path").len().sort("variant", "series_path")
    paths.write_csv(OUT / "series_paths.csv")
    print(paths, flush=True)
    print("census seconds", round(time.perf_counter() - started), "failures", sum(map(len, failures.values())), flush=True)
    reports = infer(sets)
    macro, per_target, meta = analyze(sets, reports)
    pl.Config.set_tbl_cols(20)
    pl.Config.set_tbl_rows(40)
    pl.Config.set_tbl_width_chars(250)
    print(macro)
    print(per_target)
    (OUT / "results.json").write_text(
        json.dumps(
            {
                **meta,
                "checks": check,
                "sets": sets,
                "runs": reports,
                "failures": failures,
                "macro": macro.to_dicts(),
                "precision": "bf16 autocast, batch size 1",
                "patch": "RobustDataset._index_study_geometry replaces a rejected reference with the robust fit; dataset.build_pool wrapped to split sources by whether the robust centre projects inside the acquired slice range",
                "seconds": time.perf_counter() - started,
            },
            indent=1,
        )
        + "\n"
    )
    print(json.dumps(meta, indent=1))
    print("total seconds", round(time.perf_counter() - started))

    index = build_series_index(config.paths.data_root, config.data, "train")
    labels = load_fold_labels(config, 0)
    sets = json.loads((OUT / "results.json").read_text())["sets"]
    chosen = [uid for suffix in ("85971577", "04738314", "76714231") for uid in sets["robust20"] if uid.endswith(suffix)]
    multiplier = prediction_image_multiplier(config, "validation")
    dataset = RobustDataset(labels.validation.filter(pl.col(STUDY).is_in(chosen)), index, config.data, labels.targets, False, config.seed, multiplier)
    dataset.mode = "robust20"
    rows, studies = [], []
    for position in range(len(dataset)):
        item = dataset[position]
        reference = item["study_reference"]
        studies.append(
            {
                STUDY: item["study_uid"],
                "reference_type": type(reference).__name__,
                "rms_mm": reference.rms_mm,
                "condition": reference.condition,
                "dropped_for_centre": [[uid[-8:], round(r, 1)] for uid, r in reference.dropped],
                "failures": list(item["failures"]),
            }
        )
        for slot, (values, metadata) in enumerate(zip(item["series"], item["sampling"], strict=True)):
            if not metadata["assignments"]:
                rows.append({"study": item["study_uid"][-8:], "slot": slot, "path": metadata["path"], "series": len(values)})
                continue
            depths = [a["depth"] for a in metadata["assignments"]]
            supports = [a["support"] for a in metadata["assignments"]]
            rows.append(
                {
                    "study": item["study_uid"][-8:],
                    "slot": slot,
                    "path": metadata["path"],
                    "series": len(values),
                    "excluded": len(metadata["excluded_local_sources"]),
                    "triplets": metadata["triplets"],
                    "shortfall": metadata["shortfall"],
                    "target_min_mm": round(min(metadata["targets"]), 1),
                    "target_max_mm": round(max(metadata["targets"]), 1),
                    "depth_min_mm": round(min(depths), 1),
                    "depth_max_mm": round(max(depths), 1),
                    "support_lo_mm": round(min(lo for lo, _ in supports), 1),
                    "support_hi_mm": round(max(hi for _, hi in supports), 1),
                    "max_error_mm": round(max(a["error"] for a in metadata["assignments"]), 2),
                    "geometry_valid": all(bool(v.geometry_valid.all()) for v in values),
                    "positions_finite": all(bool(torch.isfinite(torch.as_tensor(v.positions)).all()) for v in values),
                    "reflection": sorted({v.reflection_reason for v in values}),
                }
            )
    table = pl.DataFrame(rows, infer_schema_length=None)
    table.with_columns(pl.col("reflection").list.join(";")).write_csv(OUT / "sanity_slots.csv")
    (OUT / "sanity_studies.json").write_text(json.dumps(studies, indent=1) + "\n")
    pl.Config.set_tbl_cols(20)
    pl.Config.set_tbl_rows(40)
    pl.Config.set_tbl_width_chars(250)
    print(table)
    print(json.dumps(studies, indent=1))
