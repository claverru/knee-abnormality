FLAGS = pl.read_parquet(OUT / "series_flags.parquet")
DROP = set(FLAGS.filter(pl.col("action") == "drop")["SeriesInstanceUID"])


def range_gated(rows):
    usable = sorted((r["uid"], r["plane"], np.asarray(r["center"]), np.asarray(r["normal"])) for r in rows)
    by_uid = {r["uid"]: r for r in rows}
    retained = list(usable)
    dropped = []
    while True:
        result = gs.fit(retained)
        if result["reason"] is not None:
            if dropped:
                result["reason"] += "_after_drop"
            break
        point = result["point"]
        candidates = []
        for uid, residual in result["residuals"].items():
            if residual <= gs.THRESHOLD:
                continue
            signals = gs.series_signals(by_uid[uid], point)
            if signals["outside_range_mm"] > 0 or signals["fov_outside_mm"] > 0:
                candidates.append((residual, uid))
        if not candidates:
            if result["rms"] > gs.MAXIMUM_RMS:
                result["reason"] = "excessive_rms"
            break
        dropped.append(max(candidates)[1])
        retained = [item for item in retained if item[0] != dropped[-1]]
    reason = result["reason"]
    strict = reason
    if reason is None and max(result["residuals"].values()) > gs.MAXIMUM_RESIDUAL:
        strict = "excessive_residual"
    return reason or "accepted", strict or "accepted", dropped


def study(uid):
    item = INDEX[uid]
    dataset = KneeStudyDataset(pl.DataFrame({STUDY: [uid]}), {uid: item}, CONFIG.data, (), False, CONFIG.seed)
    dataset._index_study_geometry(item)
    before = dataset._study_references[uid]
    candidates = {}
    series_rows = []
    for slot, values in enumerate(item.slots):
        plane = CONFIG.data.series_types[slot].plane
        for candidate in values:
            candidates.setdefault(candidate.uid, (candidate, plane))
            indexed = dataset._slice_index.get(candidate.directory)
            if indexed is None:
                series_rows.append({"StudyInstanceUID": uid, "SeriesInstanceUID": candidate.uid, "slot": slot, "reflection_eligible": False, "reflection_reason": "index_failure"})
                continue
            eligible, reason = reflection_eligibility(indexed.geometry, plane)
            series_rows.append(
                {
                    "StudyInstanceUID": uid,
                    "SeriesInstanceUID": candidate.uid,
                    "slot": slot,
                    "reflection_eligible": eligible,
                    "reflection_reason": reason,
                    "distinct_orientations": len(indexed.geometry.orientations),
                }
            )
    kept_eligible, gated_rows = [], []
    for series_uid, (candidate, plane) in sorted(candidates.items()):
        if series_uid in DROP:
            continue
        indexed = dataset._slice_index.get(candidate.directory)
        if indexed is None or len(indexed.records) < SLICES_PER_INPUT:
            continue
        kept_eligible.append((series_uid, plane, indexed.geometry))
        geometry = indexed.geometry
        middle = indexed.records[len(indexed.records) // 2]
        if geometry.reason is None and middle.pose is not None:
            layout = middle.pose.layout
            gated_rows.append(
                {
                    "uid": series_uid,
                    "plane": plane,
                    "center": geometry.center,
                    "normal": geometry.normal,
                    "bounds": geometry.bounds,
                    "middle_orientation": layout.orientation,
                    "middle_position": middle.pose.position,
                    "pixel_spacing": layout.spacing,
                    "rows": layout.rows,
                    "columns": layout.columns,
                }
            )
    changed = any(u in DROP for u in candidates)
    after = estimate_reference(tuple(kept_eligible)) if changed else before
    gated, gated_strict, gated_dropped = range_gated(gated_rows)
    summary = {
        "StudyInstanceUID": uid,
        "series_set_changed": changed,
        "dropped_series": sum(u in DROP for u in candidates),
        "reason_before": before.reason or "accepted",
        "rms_before": before.rms_mm,
        "reason_after": after.reason or "accepted",
        "rms_after": after.rms_mm,
        "range_gated_reason": gated,
        "range_gated_strict_reason": gated_strict,
        "range_gated_dropped": len(gated_dropped),
        "range_gated_dropped_uids": gated_dropped,
    }
    return summary, series_rows


def main():
    FOLLOW.mkdir(parents=True, exist_ok=True)
    started = time.perf_counter()
    studies, series = [], []
    with ProcessPoolExecutor(28) as pool:
        for summary, rows in pool.map(study, sorted(INDEX), chunksize=4):
            studies.append(summary)
            series.extend(rows)
    pl.DataFrame(studies, infer_schema_length=None).write_parquet(FOLLOW / "reference_after_drops.parquet")
    pl.DataFrame(series, infer_schema_length=None).write_parquet(FOLLOW / "reflection_eligibility_series.parquet")
    print("done", len(studies), len(series), round(time.perf_counter() - started))


if __name__ == "__main__":
    main()
