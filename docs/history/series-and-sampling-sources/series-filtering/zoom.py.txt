import sys
from pathlib import Path

import polars as pl
from PIL import Image, ImageDraw

sys.path.insert(0, str(Path(__file__).parent))
import sheets
from common import OUT, load

sheets.TILE = 300
frame = load()


def main(keys, name):
    blocks = []
    for key in keys:
        row = frame.filter(pl.col("series8") == key).row(0, named=True)
        same = frame.filter((pl.col("StudyInstanceUID") == row["StudyInstanceUID"]) & (pl.col("plane") == row["plane"])).sort(pl.col("SeriesInstanceUID") != row["SeriesInstanceUID"], "slot")
        tiles = []
        for other in same.head(4).iter_rows(named=True):
            image = sheets.tile(other["sample_paths"][1])
            draw = ImageDraw.Draw(image)
            label = f"{other['series8']} {other['slot_name']} TE{other['te']} {other['desc'][:22]}"
            draw.text((3, 3), label, fill=(255, 60, 60) if other["series8"] == key else (255, 255, 90), font=sheets.FONT, stroke_width=1, stroke_fill=(0, 0, 0))
            tiles.append(image)
        block = Image.new("RGB", (4 * 300, 300))
        for index, image in enumerate(tiles):
            block.paste(image, (index * 300, 0))
        blocks.append(block)
    canvas = Image.new("RGB", (1200, 300 * len(blocks)))
    for index, block in enumerate(blocks):
        canvas.paste(block, (0, index * 300))
    path = OUT / "sheets" / f"zoom-{name}.png"
    canvas.save(path)
    print(path)


if __name__ == "__main__":
    main(sys.argv[2].split(","), sys.argv[1])
