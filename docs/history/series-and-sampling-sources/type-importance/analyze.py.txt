ROOT = Path("/home/claudio/src/knee-abnormality")
QUEUE = ROOT / "outputs/training/overnight-20260916T214520Z-t8k_kw2z/runs"
RUNS = ("00-control", "01-control-seed-43", "02-control-seed-44")
OUT = ROOT / "outputs/analysis/type-importance-20260917"

RESAMPLES = 1000

def weighted_auc(labels, scores, weights):
    order = np.argsort(scores, kind="mergesort")
    ordered = scores[order]
    positive = labels[order]
    w = weights[:, order]
    negative_w = w * ~positive
    cumulative = np.concatenate(
        (np.zeros((len(w), 1)), np.cumsum(negative_w, axis=1)), axis=1
    )
    left = np.searchsorted(ordered, ordered, "left")
    right = np.searchsorted(ordered, ordered, "right")
    index = np.nonzero(positive)[0]
    below = cumulative[:, left[index]]
    equal = cumulative[:, right[index]] - below
    positive_w = w[:, index]
    numerator = (positive_w * (below + 0.5 * equal)).sum(1)
    total_positive = positive_w.sum(1)
    total_negative = negative_w.sum(1)
    with np.errstate(invalid="ignore", divide="ignore"):
        auc = numerator / (total_positive * total_negative)
    auc[(total_positive == 0) | (total_negative == 0)] = np.nan
    return auc


def auc_matrix(hard, probabilities, weights):
    return np.stack(
        [
            weighted_auc(hard[:, t], probabilities[:, t].astype(np.float64), weights)
            for t in range(hard.shape[1])
        ],
        axis=1,
    )


def draw_weights(n):
    rng = np.random.default_rng(SEED)
    sampled = rng.integers(0, n, size=(RESAMPLES, n))
    weights = np.zeros((RESAMPLES, n))
    np.add.at(weights, (np.repeat(np.arange(RESAMPLES), n), sampled.ravel()), 1.0)
    return np.concatenate((np.ones((1, n)), weights))


def with_macro(values):
    macro = values.mean(1, keepdims=True)
    return np.concatenate((values, macro), axis=1)


def percentile(values):
    finite = values[np.isfinite(values)]
    if not len(finite):
        return None, None, 0
    return float(np.quantile(finite, 0.025)), float(np.quantile(finite, 0.975)), len(finite)


def subsets_for(variant, keeps, presence, defined):
    kept = [TYPE_NAMES.index(name) for name in keeps[variant]]
    if variant.startswith("without_"):
        removed = TYPE_NAMES.index(variant.removeprefix("without_"))
        return {
            "all_defined": defined,
            "has_removed_type": defined & presence[:, removed],
        }
    return {
        "all_defined": defined,
        "has_all_kept_types": defined & presence[:, kept].all(1),
    }


def main():
    started = time.perf_counter()
    inference = json.loads((RAW / "inference.json").read_text())
    keeps = inference["runs"][0]["variant_keeps"]
    data = [np.load(RAW / f"{run}.npz") for run in RUNS]
    study = data[0]["study"]
    targets = tuple(data[0]["targets"].tolist())
    variants = data[0]["variants"].tolist()
    for item in data[1:]:
        if not (np.array_equal(item["study"], study) and item["variants"].tolist() == variants):
            raise ValueError("Run artifacts differ in membership")
    presences = [item["presence"] for item in data]
    same_presence = all(np.array_equal(presences[0], p) for p in presences[1:])
    presence = presences[0]
    config = load_config(QUEUE / RUNS[0] / "effective_config.json")
    labels = load_fold_labels(config, 0)
    validation = labels.validation
    if validation[STUDY].to_list() != study.tolist():
        raise ValueError("Label order differs")
    soft = validation.select([f"target_{t}" for t in targets]).to_numpy()
    hard = soft >= TARGET_THRESHOLD
    predictions = {
        variant: np.stack([item["predictions"][variants.index(variant)] for item in data])
        for variant in variants
    }
    intact = predictions["intact"]
    check = roc_auc_score(hard[:, 0], intact[0][:, 0])
    mine = weighted_auc(hard[:, 0], intact[0][:, 0].astype(np.float64), np.ones((1, len(study))))[0]
    rng = np.random.default_rng(1)
    sampled = rng.integers(0, len(study), len(study))
    weights = np.bincount(sampled, minlength=len(study))[None].astype(float)
    check_boot = roc_auc_score(hard[sampled, 3], intact[1][sampled, 3])
    mine_boot = weighted_auc(hard[:, 3], intact[1][:, 3].astype(np.float64), weights)[0]
    auc_validation = {
        "full": [float(check), float(mine)],
        "resampled": [float(check_boot), float(mine_boot)],
    }
    columns = [*targets, "macro"]
    rows = []
    cache = {}
    for variant in variants:
        if variant == "intact":
            continue
        defined = np.isfinite(predictions[variant]).all(axis=(0, 2))
        for subset, mask in subsets_for(variant, keeps, presence, defined).items():
            n = int(mask.sum())
            key = mask.tobytes()
            if key not in cache:
                weights = draw_weights(n)
                cache[key] = (
                    weights,
                    np.stack([with_macro(auc_matrix(hard[mask], intact[r][mask], weights)) for r in range(3)]),
                )
            weights, intact_auc = cache[key]
            variant_auc = np.stack(
                [with_macro(auc_matrix(hard[mask], predictions[variant][r][mask], weights)) for r in range(3)]
            )
            delta = variant_auc - intact_auc
            mean_delta = delta.mean(0)
            change = np.abs(predictions[variant][:, mask] - intact[:, mask]).mean(1)
            for c, column in enumerate(columns):
                point = delta[:, 0, c]
                mean_point = float(point.mean())
                sign = np.sign(mean_point)
                row = {
                    "variant": variant,
                    "subset": subset,
                    "studies": n,
                    "positives": int(hard[mask][:, c].sum()) if c < len(targets) else None,
                    "target": column,
                    "intact_auc_mean": float(intact_auc[:, 0, c].mean()),
                    "variant_auc_mean": float(variant_auc[:, 0, c].mean()),
                    "delta_mean": mean_point,
                    "delta_min": float(point.min()),
                    "delta_max": float(point.max()),
                    "delta_spread": float(point.max() - point.min()),
                    "sign_agreement": int((np.sign(point) == sign).sum()) if sign != 0 else 0,
                }
                low, high, valid = percentile(mean_delta[1:, c])
                row.update({"ci_low": low, "ci_high": high, "valid_draws": valid})
                row["ci_excludes_zero"] = low is not None and (low > 0 or high < 0)
                for r, run in enumerate(RUNS):
                    row[f"delta_{run}"] = float(point[r])
                    run_low, run_high, _ = percentile(delta[r, 1:, c])
                    row[f"ci_low_{run}"] = run_low
                    row[f"ci_high_{run}"] = run_high
                row["mean_abs_probability_change"] = (
                    float(change[:, c].mean()) if c < len(targets) else float(change.mean())
                )
                rows.append(row)
    deltas = pl.DataFrame(rows)
    deltas.write_csv(OUT / "deltas.csv")

    intact_rows = []
    for c, column in enumerate(columns):
        full = [with_macro(auc_matrix(hard, intact[r], np.ones((1, len(study)))))[0, c] for r in range(3)]
        intact_rows.append({"target": column, **{run: float(v) for run, v in zip(RUNS, full)}, "mean": float(np.mean(full))})
    intact_frame = pl.DataFrame(intact_rows)
    intact_frame.write_csv(OUT / "intact_auc.csv")

    subgroup_rows = []
    for k, name in enumerate(TYPE_NAMES):
        for has in (True, False):
            mask = presence[:, k] == has
            if mask.sum() < 20:
                continue
            values = np.stack(
                [with_macro(auc_matrix(hard[mask], intact[r][mask], np.ones((1, int(mask.sum())))))[0] for r in range(3)]
            ).mean(0)
            subgroup_rows.append(
                {"type": name, "has_type": has, "studies": int(mask.sum()), **dict(zip(columns, map(float, values)))}
            )
    pl.DataFrame(subgroup_rows).write_csv(OUT / "intact_auc_by_type_presence.csv")

    shares = np.stack([item["attention_share"] for item in data])
    tokens = np.stack([item["token_counts"] for item in data])
    token_share = tokens / tokens.sum(2, keepdims=True)
    attention_rows = []
    groups = {
        "all_six_types": presence.all(1),
        "five_types_without_ax_t1pd": presence[:, 1:].all(1) & ~presence[:, 0],
    }
    share_long = []
    for group, mask in groups.items():
        for r, run in enumerate([*RUNS, "mean"]):
            s = shares[:, mask].mean(0) if run == "mean" else shares[r, mask]
            ts = token_share[:, mask].mean(0) if run == "mean" else token_share[r, mask]
            for c, target in enumerate(targets):
                for k, name in enumerate(TYPE_NAMES):
                    if group != "all_six_types" and k == 0:
                        continue
                    attention_mean = float(s[:, k, c].mean())
                    token_mean = float(ts[:, k].mean())
                    attention_rows.append(
                        {
                            "group": group,
                            "studies": int(mask.sum()),
                            "run": run,
                            "target": target,
                            "type": name,
                            "attention_share": attention_mean,
                            "token_share": token_mean,
                            "enrichment": attention_mean / token_mean,
                        }
                    )
    attention = pl.DataFrame(attention_rows)
    attention.write_csv(OUT / "attention_share.csv")
    per_study = []
    for r, run in enumerate(RUNS):
        for k, name in enumerate(TYPE_NAMES):
            frame = pl.DataFrame(
                {
                    STUDY: study,
                    "run": run,
                    "type": name,
                    "present": presence[:, k],
                    "valid_tokens": tokens[r][:, k],
                    **{target: shares[r][:, k, c] for c, target in enumerate(targets)},
                }
            )
            per_study.append(frame)
    pl.concat(per_study).write_csv(OUT / "attention_share_per_study.csv")

    for r, run in enumerate(RUNS):
        frames = []
        for variant in variants:
            values = predictions[variant][r]
            keep = np.isfinite(values).all(1)
            frames.append(
                pl.DataFrame(
                    {STUDY: study[keep], "variant": variant, **{t: values[keep, c] for c, t in enumerate(targets)}}
                )
            )
        pl.concat(frames).write_csv(OUT / f"predictions_{run}.csv")
    pl.DataFrame(
        {STUDY: study, **{name: presence[:, k] for k, name in enumerate(TYPE_NAMES)}}
    ).write_csv(OUT / "validation_presence.csv")

    index = build_series_index(config.paths.data_root, config.data, "train")
    training = labels.training
    everything = pl.concat(
        [
            training.select(STUDY, *[f"target_{t}" for t in targets]),
            validation.select(STUDY, *[f"target_{t}" for t in targets]),
        ]
    )
    available = np.array([[bool(index[uid].slots[k]) for k in range(6)] for uid in everything[STUDY]])
    all_hard = everything.select([f"target_{t}" for t in targets]).to_numpy() >= TARGET_THRESHOLD
    availability_rows = []
    for k, name in enumerate(TYPE_NAMES):
        has = available[:, k]
        for c, target in enumerate(targets):
            a, b = int(all_hard[has, c].sum()), int((~all_hard[has, c]).sum())
            d, e = int(all_hard[~has, c].sum()), int((~all_hard[~has, c]).sum())
            odds, p = fisher_exact([[a, b], [d, e]])
            availability_rows.append(
                {
                    "type": name,
                    "target": target,
                    "studies_with": int(has.sum()),
                    "studies_without": int((~has).sum()),
                    "prevalence_with": a / max(a + b, 1),
                    "prevalence_without": d / max(d + e, 1),
                    "prevalence_ratio": (a / max(a + b, 1)) / (d / max(d + e, 1)) if d else None,
                    "odds_ratio": float(odds),
                    "fisher_p": float(p),
                }
            )
    availability = pl.DataFrame(availability_rows)
    availability.write_csv(OUT / "availability_vs_prevalence.csv")
    cooccurrence = pl.DataFrame(
        {
            "type": TYPE_NAMES,
            "studies": available.sum(0),
            **{f"also_{name}": [float(available[available[:, i], j].mean()) for i in range(6)] for j, name in enumerate(TYPE_NAMES)},
        }
    )
    cooccurrence.write_csv(OUT / "availability_cooccurrence.csv")
    patterns = (
        pl.DataFrame({"pattern": ["+".join(n for n, v in zip(TYPE_NAMES, row) if v) for row in available]})
        .group_by("pattern")
        .len()
        .sort("len", descending=True)
    )
    patterns.write_csv(OUT / "availability_patterns.csv")
    group_counts = {
        "studies": int(len(available)),
        "all_six": int(available.all(1).sum()),
        "missing_any": int((~available.all(1)).sum()),
        "types_present_distribution": {int(k): int(v) for k, v in zip(*np.unique(available.sum(1), return_counts=True))},
        "validation_presence_vs_metadata_mismatch": None,
    }
    validation_available = np.array([[bool(index[uid].slots[k]) for k in range(6)] for uid in study])
    group_counts["validation_presence_vs_metadata_mismatch"] = int((validation_available != presence).any(1).sum())

    summary = {
        "created": time.strftime("%Y-%m-%dT%H:%M:%S"),
        "inference": inference,
        "presence_identical_across_runs": same_presence,
        "validation_presence_counts": dict(zip(TYPE_NAMES, map(int, presence.sum(0)))),
        "auc_implementation_check": auc_validation,
        "bootstrap": {
            "resamples": RESAMPLES,
            "seed": SEED,
            "unit": "study, paired across intact/variant and across the three checkpoints",
            "interval": "percentile 95% of the three-checkpoint mean delta; per-run intervals in deltas.csv",
        },
        "subsets": {
            "all_defined": "studies with at least one visually valid token after removal",
            "has_removed_type": "leave-one-type-out restricted to studies whose batch had that type",
            "has_all_kept_types": "keep-only variants restricted to studies having every kept type",
        },
        "intact_auc": intact_frame.to_dicts(),
        "availability_counts": group_counts,
        "attention_groups": {g: int(m.sum()) for g, m in groups.items()},
        "numerics_checks": {
            name: json.loads((RAW / f"{name}.json").read_text())
            for name in ("numerics_check", "numerics_check_no_tf32")
        },
        "triplet_capacity": pl.read_csv(OUT / "validation_triplet_capacity_summary.csv").to_dicts(),
        "analysis_seconds": time.perf_counter() - started,
        "interpretation": "Removal effects measure reliance of models trained with the current budget and series dropout 0.2; they do not measure the value of adding slices. Removal happens after preparation, so the study reference geometry still uses every series. Attention share is descriptive, not contribution.",
    }
    (OUT / "results.json").write_text(json.dumps(summary, indent=2, default=str) + "\n")
    print(json.dumps({k: summary[k] for k in ("presence_identical_across_runs", "validation_presence_counts", "auc_implementation_check", "availability_counts", "attention_groups", "analysis_seconds")}, indent=2))


if __name__ == "__main__":
    main()
