import json
from pathlib import Path

import numpy as np
import polars as pl

from knee_abnormality.config import load_config
from knee_abnormality.data.dataset import build_data_loader
from knee_abnormality.data.labels import load_fold_labels
from knee_abnormality.inference.engine import prediction_dataset

ROOT = Path("/home/claudio/src/knee-abnormality")
OUT = ROOT / "outputs/analysis/type-importance-20260917"
TYPE_NAMES = ("ax_t1pd", "ax_fluid", "cor_t1pd", "cor_fluid", "sag_t1pd", "sag_fluid")
config = load_config(ROOT / "outputs/training/overnight-20260916T214520Z-t8k_kw2z/runs/00-control/effective_config.json")
labels = load_fold_labels(config, 0)
dataset = prediction_dataset(config, labels.targets, labels.validation, "train", 2)
loader = build_data_loader(dataset, config.trainer, 8, num_workers=12)
rows = []
for batch in loader:
    for uid, sampling in zip(batch["study_uid"], batch["sampling"]):
        for slot, meta in enumerate(sampling):
            if meta["capacity"] == 0 and meta["triplets"] == 0 and not batch["series_presence"][batch["study_uid"].index(uid), slot]:
                continue
            rows.append({"study": uid, "type": TYPE_NAMES[slot], "requested": meta["requested_triplets"], "capacity": meta["capacity"], "triplets": meta["triplets"], "shortfall": meta["shortfall"]})
frame = pl.DataFrame(rows)
frame.write_csv(OUT / "validation_triplet_capacity.csv")
summary = frame.group_by("type").agg(
    pl.len().alias("series_slots"),
    pl.col("requested").first(),
    pl.col("capacity").quantile(0.1).alias("cap_p10"),
    pl.col("capacity").median().alias("cap_median"),
    pl.col("capacity").quantile(0.9).alias("cap_p90"),
    (pl.col("shortfall") > 0).mean().alias("share_short"),
    *[(pl.col("capacity") >= k).mean().alias(f"cap_ge_{k}") for k in (6, 8, 12, 14, 16)],
).sort("type")
pl.Config.set_tbl_cols(20); pl.Config.set_tbl_width_chars(250)
print(summary)
summary.write_csv(OUT / "validation_triplet_capacity_summary.csv")
