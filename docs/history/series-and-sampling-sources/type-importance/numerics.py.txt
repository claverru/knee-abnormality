import json
import sys
import time
import warnings
from pathlib import Path

import numpy as np
import torch

sys.path.insert(0, str(Path(__file__).parent))
import infer
from knee_abnormality.config import load_config
from knee_abnormality.data.dataset import collate_studies
from knee_abnormality.data.labels import load_fold_labels
from knee_abnormality.inference.checkpoint import load_checkpoint_model
from knee_abnormality.inference.engine import move_batch, prediction_dataset

warnings.filterwarnings("ignore")
STUDIES = 16


def token_pool(model, dataset, keeps, dtype, device, batch_size):
    items = [dataset[i] for i in range(STUDIES)]
    collected = {k: [] for k in keeps}
    with torch.inference_mode():
        for start in range(0, STUDIES, batch_size):
            batch = collate_studies(items[start : start + batch_size])
            types, _ = infer.image_types(batch)
            with torch.autocast("cuda", dtype=dtype, enabled=dtype is not torch.float32):
                encoded = model.encode_batch(move_batch(batch, device))
            results, _, _, _ = infer.pool_variants(model, encoded, types.to(device), keeps, dtype)
            for k, v in results.items():
                collected[k].append(v)
    return {k: torch.cat(v).numpy() for k, v in collected.items()}, items


def real_removal(model, items, keeps, dtype, device):
    out = {}
    for name, kept in keeps.items():
        modified, rows = [], []
        for index, item in enumerate(items):
            series = tuple(v if s in kept else () for s, v in enumerate(item["series"]))
            if any(series):
                modified.append({**item, "series": series})
                rows.append(index)
        values = np.full((len(items), 12), np.nan, dtype=np.float32)
        with torch.inference_mode():
            for start in range(0, len(modified), 8):
                batch = move_batch(collate_studies(modified[start : start + 8]), device)
                with torch.autocast("cuda", dtype=dtype, enabled=dtype is not torch.float32):
                    values[rows[start : start + 8]] = torch.sigmoid(model(batch).float()).cpu().numpy()
        out[name] = values
    return out


def main():
    started = time.perf_counter()
    torch.set_float32_matmul_precision("highest")
    run_dir = infer.QUEUE / infer.RUNS[0]
    config = load_config(run_dir / "effective_config.json")
    labels = load_fold_labels(config, 0)
    selected = json.loads((run_dir / "fold_0/selected_checkpoint.json").read_text())
    device = torch.device("cuda")
    model = load_checkpoint_model(
        config, labels.targets, run_dir / "fold_0" / Path(selected["path"]).name, device, expected_fold=0
    )
    dataset = prediction_dataset(config, labels.targets, labels.validation.head(STUDIES), "train", 2)
    keeps = infer.variant_keeps(config.data.series_types)
    report = {}
    fp32_pool, items = token_pool(model, dataset, keeps, torch.float32, device, 8)
    fp32_real = real_removal(model, items, keeps, torch.float32, device)
    report["fp32_real_removal_vs_token_pool_max_abs"] = {
        k: float(np.nanmax(np.abs(fp32_real[k] - fp32_pool[k]))) for k in keeps
    }
    bf8, _ = token_pool(model, dataset, keeps, torch.bfloat16, device, 8)
    bf5, _ = token_pool(model, dataset, keeps, torch.bfloat16, device, 5)
    report["bf16_batch8_vs_batch5_token_pool_max_abs"] = {
        k: float(np.nanmax(np.abs(bf8[k] - bf5[k]))) for k in keeps
    }
    report["bf16_vs_fp32_token_pool_max_abs"] = {
        k: float(np.nanmax(np.abs(bf8[k] - fp32_pool[k]))) for k in keeps
    }
    report["seconds"] = time.perf_counter() - started
    (infer.RAW / "numerics_check.json").write_text(json.dumps(report, indent=2) + "\n")
    print(json.dumps(report, indent=2))


if __name__ == "__main__":
    main()
