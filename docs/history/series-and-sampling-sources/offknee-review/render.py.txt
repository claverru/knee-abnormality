from pathlib import Path
import numpy as np
import polars as pl
import pydicom
import matplotlib
matplotlib.use("Agg")
import matplotlib.pyplot as plt

root = Path("data/raw/train_series")
flags = pl.read_parquet("outputs/analysis/series-filtering-20260917/series_flags_with_triplets.parquet")
targets = [("89344702", "calf axial STIR, Baker's cyst 'con fuga'"), ("33830134", "thigh axial T1, ascending Baker's cyst"), ("94280767", "thigh axial T2 FS, effusion/synovitis")]


def load(study, series):
    files = [pydicom.dcmread(p) for p in (root / study / series).glob("*.dcm")]
    normal = np.cross(*np.array(files[0].ImageOrientationPatient, float).reshape(2, 3))
    files.sort(key=lambda d: float(np.dot(normal, np.array(d.ImagePositionPatient, float))))
    return files


def norm(a):
    lo, hi = np.percentile(a, [1, 99.5])
    return np.clip((a - lo) / max(hi - lo, 1e-6), 0, 1)


fig, axes = plt.subplots(len(targets), 9, figsize=(27, 3.4 * len(targets)))
for row, (s8, title) in enumerate(targets):
    r = flags.filter(pl.col("series8") == s8).row(0, named=True)
    study, series = r["StudyInstanceUID"], r["SeriesInstanceUID"]
    ax_files = load(study, series)
    zs = [float(d.ImagePositionPatient[2]) for d in ax_files]
    idx = np.linspace(0, len(ax_files) - 1, 6).round().astype(int)
    for k, i in enumerate(idx):
        axes[row, k].imshow(norm(ax_files[i].pixel_array.astype(float)), cmap="gray")
        axes[row, k].set_title(f"{s8} z={zs[i]:.0f}", fontsize=9)
    sag = flags.filter((pl.col("StudyInstanceUID") == study) & (pl.col("slot_name") == "sag_fluid") & (pl.col("action") == "keep"))
    sag = sag if sag.height else flags.filter((pl.col("StudyInstanceUID") == study) & pl.col("slot_name").str.starts_with("sag"))
    sag_files = load(study, sag["SeriesInstanceUID"][0])
    for k, frac in enumerate([0.3, 0.5, 0.7]):
        d = sag_files[int(frac * (len(sag_files) - 1))]
        a = axes[row, 6 + k]
        a.imshow(norm(d.pixel_array.astype(float)), cmap="gray")
        z0 = float(d.ImagePositionPatient[2])
        rowdir = np.array(d.ImageOrientationPatient[3:], float)
        dz = float(d.PixelSpacing[0]) * rowdir[2]
        for z in (min(zs), max(zs)):
            if dz:
                y = (z - z0) / dz
                if 0 <= y < d.Rows:
                    a.axhline(y, color="red", lw=1)
        a.set_title(f"sag {sag['desc'][0]} {int(frac*100)}% (red = axial slab)", fontsize=8)
    axes[row, 0].set_ylabel(f"{r['study8']}\n{title}", fontsize=10)
for a in axes.ravel():
    a.set_xticks([]); a.set_yticks([])
out = Path("outputs/analysis/series-filtering-20260917/offknee-review.png")
fig.tight_layout(); fig.savefig(out, dpi=55)
print(out)
