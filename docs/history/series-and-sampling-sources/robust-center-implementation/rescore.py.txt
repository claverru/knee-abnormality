import json
import sys
import time
from pathlib import Path

import numpy as np
import polars as pl
import torch

from knee_abnormality.config import ExperimentConfig
from knee_abnormality.data.dataset import KneeStudyDataset, build_data_loader
from knee_abnormality.data.labels import STUDY, load_fold_labels
from knee_abnormality.data.sources import build_series_index
from knee_abnormality.evaluation.metrics import validation_metrics
from knee_abnormality.inference.checkpoint import checkpoint_model, read_checkpoint
from knee_abnormality.inference.engine import move_batch, prediction_image_multiplier

S = Path(sys.argv[1])
RUN = Path("/home/claudio/src/knee-abnormality/outputs/training/ranked-filtered-seed42")
CHECKPOINT = RUN / "fold_0/best-epoch=09-val_macro_auc=0.88957.ckpt"
started = time.perf_counter()
torch.backends.cudnn.benchmark = False
raw = json.loads((RUN / "effective_config.json").read_text())
raw = {k: v for k, v in raw.items() if k in ExperimentConfig.model_fields}


def config_with(option):
    values = json.loads(json.dumps(raw))
    values["data"]["robust_center_outlier_mm"] = option
    return ExperimentConfig.model_validate(values)


base = config_with(None)
labels = load_fold_labels(base, 0)
targets, validation = labels.targets, labels.validation
rescued = json.loads((S / "report.json").read_text())["20.0_sets"]["fold0"]
checkpoint = read_checkpoint(CHECKPOINT)
assert checkpoint["hyper_parameters"]["preprocessing_identity"]["algorithm"].endswith("_v10")
device = torch.device("cuda")
model = checkpoint_model(checkpoint, base, targets, device)
index = build_series_index(base.paths.data_root, base.data, "train")
multiplier = prediction_image_multiplier(base, "validation")
saved = pl.read_csv(RUN / "fold_0/validation_predictions.csv")
assert saved[STUDY].to_list() == validation[STUDY].to_list()
order = {uid: i for i, uid in enumerate(validation[STUDY].to_list())}
y = validation.select([f"target_{t}" for t in targets]).to_numpy()
gold = validation["is_gold"].to_numpy()
frame = validation.filter(pl.col(STUDY).is_in(rescued))


def predict(config):
    dataset = KneeStudyDataset(frame, index, config.data, targets, None, multiplier)
    loader = build_data_loader(dataset, config.trainer, 1, num_workers=12)
    ids, outputs, paths = [], [], []
    with torch.inference_mode():
        for batch in loader:
            ids.extend(batch["study_uid"])
            paths.append(sum(s["path"] == "physical" for s in batch["sampling"][0]))
            with torch.autocast("cuda", dtype=torch.bfloat16):
                outputs.append(torch.sigmoid(model(move_batch(batch, device)).float()).cpu())
    return ids, torch.cat(outputs).numpy(), paths


def macro(values):
    return validation_metrics(y, values, targets, gold)["macro_auc"]


saved_values = saved.select(targets).to_numpy()
result = {"saved_macro": macro(saved_values), "rescued_studies": len(rescued)}
intact_ids, intact, intact_paths = predict(base)
robust_ids, robust, robust_paths = predict(config_with(20.0))
assert intact_ids == robust_ids
rows = [order[uid] for uid in intact_ids]
intact_values = saved_values.copy()
intact_values[rows] = intact
robust_values = saved_values.copy()
robust_values[rows] = robust
robust_on_saved = saved_values.copy()
robust_on_saved[rows] = robust
result.update(
    intact_recompute_max_abs_vs_saved=float(np.abs(intact - saved_values[rows]).max()),
    intact_macro=macro(intact_values),
    robust_macro=macro(robust_values),
    delta_vs_intact_batch1=macro(robust_values) - macro(intact_values),
    delta_vs_saved=macro(robust_on_saved) - result["saved_macro"],
    mean_abs_prob_delta=float(np.abs(robust - intact).mean()),
    physical_slots_intact=int(sum(intact_paths)),
    physical_slots_robust=int(sum(robust_paths)),
    seconds=time.perf_counter() - started,
)
print(json.dumps(result, indent=1))
(S / "rescore.json").write_text(json.dumps(result, indent=1))
