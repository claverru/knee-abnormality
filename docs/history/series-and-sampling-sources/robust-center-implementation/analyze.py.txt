import json
import pickle
import sys
from collections import Counter
from pathlib import Path

import polars as pl

S = Path(sys.argv[1])
ROOT = Path("/home/claudio/src/knee-abnormality")
head = {r["study"]: r for r in pickle.loads((S / "head.pkl").read_bytes())}
new = {r["study"]: r for r in pickle.loads((S / "new.pkl").read_bytes())}
assert set(head) == set(new)
report = {}

mismatch_ref, mismatch_eval, fold0 = [], [], 0
for uid, r in new.items():
    h = head[uid]["options"]["absent"]
    n = r["options"]["None"]
    if n["reference"][:7] != h["reference"] or n["reference"][7] is not None or len(n["reference"]) != 8:
        mismatch_ref.append(uid)
    if r["fold"] == 0:
        fold0 += 1
        if n["eval"] != h["eval"]:
            mismatch_eval.append(uid)
report["null_vs_head"] = {
    "studies": len(new),
    "reference_mismatches": len(mismatch_ref),
    "fold0_studies": fold0,
    "fold0_selection_mismatches": len(mismatch_eval),
    "fold0_triplets": sum(len(s["selected"]) for r in new.values() if r["fold"] == 0 for s in r["options"]["None"]["eval"]),
}

for option in ("20.0", "10.0"):
    stats = Counter()
    changed_accepted = []
    rescued_sets = {"train": [], "fold0": []}
    for uid, r in new.items():
        base = r["options"]["None"]
        o = r["options"][option]
        if base["reason"] is None:
            stats["accepted"] += 1
            if o["reference"] != base["reference"] or o.get("eval") != base.get("eval"):
                changed_accepted.append(uid)
            continue
        stats["rejected"] += 1
        stats["rejected_fold0"] += r["fold"] == 0
        if o["point"] is None:
            if o["reference"] != base["reference"] or o["eval"] != base["eval"]:
                changed_accepted.append(("still_rejected_changed", uid))
            stats["still_rejected_" + base["reason"]] += 1
            continue
        scope = ["train"] + (["fold0"] if r["fold"] == 0 else [])
        for sc in scope:
            p = "" if sc == "train" else "fold0_"
            rescued_sets[sc].append(uid)
            stats[p + "rescued"] += 1
            stats[p + "fit_outliers"] += len(o["outliers"])
            stats[p + "rescued_from_" + base["reason"]] += 1
            for bs, os_ in zip(base["eval"], o["eval"]):
                left = set(bs["pool_uids"]) - set(os_["pool_uids"])
                added = set(os_["pool_uids"]) - set(bs["pool_uids"])
                stats[p + "series_in_input_before"] += len(bs["pool_uids"])
                stats[p + "series_leave_input"] += len(left)
                stats[p + "series_added"] += len(added)
                stats[p + "outside_range_series"] += sum(v == "outside_acquired_range" for v in os_["sources"].values())
                stats[p + "outliers_leaving_input"] += len(left & set(o["outliers"]))
                if os_["pool_uids"]:
                    stats[p + ("slots_physical" if os_["physical"] else "slots_local")] += 1
                    if os_["physical"]:
                        stats[p + "series_physical"] += len(os_["pool_uids"])
                    else:
                        stats[p + "series_local"] += len(os_["pool_uids"])
                stats[p + "triplets_before"] += len(bs["selected"])
                stats[p + "triplets_after"] += len(os_["selected"])
                stats[p + "shortfall_increase_slots"] += len(os_["selected"]) < len(bs["selected"])
    report[option] = {**dict(sorted(stats.items())), "accepted_or_unrescued_changed": len(changed_accepted)}
    report[option + "_sets"] = {k: sorted(v) for k, v in rescued_sets.items()}

census = pl.read_parquet(ROOT / "outputs/analysis/series-filtering-20260917/robust_studies.parquet")
patch = json.loads((ROOT / "outputs/analysis/series-filtering-20260917/robust-centre-rescore/results.json").read_text())
census_set = set(census.filter(pl.col("robust_reason") == "accepted")["StudyInstanceUID"]) if "robust_reason" in census.columns else None
mine = set(report["20.0_sets"]["train"])
report["compare"] = {
    "census_columns": census.columns,
    "census_rescued": len(census_set) if census_set is not None else None,
    "only_census": sorted(census_set - mine) if census_set is not None else None,
    "only_mine": sorted(mine - census_set) if census_set is not None else None,
    "patch_fold0_equal": set(patch["sets"]["robust20"]) == set(report["20.0_sets"]["fold0"]),
}
pstudies = pl.read_parquet(ROOT / "outputs/analysis/series-filtering-20260917/robust-centre-rescore/studies.parquet")
diff = 0.0
for uid, point in pstudies.select("StudyInstanceUID", "robust20_point").iter_rows():
    if point is None:
        continue
    mp = new[uid]["options"]["20.0"]["point"]
    diff = max(diff, max(abs(a - b) for a, b in zip(point, mp)))
report["compare"]["patch_fold0_point_max_abs_diff"] = diff
series = pl.read_parquet(ROOT / "outputs/analysis/series-filtering-20260917/robust-centre-rescore/series_sampling.parquet")
report["compare"]["patch_series_paths"] = series.filter(pl.col("variant") == "robust20").group_by("series_path").len().to_dicts()
leave = []
for uid in report["20.0_sets"]["fold0"]:
    o, b = new[uid]["options"]["20.0"], new[uid]["options"]["None"]
    for slot, (bs, os_) in enumerate(zip(b["eval"], o["eval"])):
        leave += [(uid, slot, s) for s in set(bs["pool_uids"]) - set(os_["pool_uids"])]
patch_excluded = set(series.filter((pl.col("variant") == "robust20") & (pl.col("series_path") == "excluded")).select("StudyInstanceUID", "slot", "SeriesInstanceUID").iter_rows())
report["compare"]["fold0_leave_equal_patch_excluded"] = set(leave) == patch_excluded
report["compare"]["patch_excluded_count"] = len(patch_excluded)
print(json.dumps({k: v for k, v in report.items() if not k.endswith("_sets")}, indent=1))
(S / "report.json").write_text(json.dumps(report, indent=1))
