from pathlib import Path

import numpy as np
import polars as pl
import torch

from knee_abnormality.config import load_config
from knee_abnormality.data.dataset import build_data_loader, collate_studies
from knee_abnormality.data.labels import STUDY, TARGET_THRESHOLD, load_fold_labels
from knee_abnormality.inference.checkpoint import load_checkpoint_model

    move_batch,
    prediction_dataset,
    prediction_image_multiplier,
)

ROOT = Path("/home/claudio/src/knee-abnormality")
RUNS = ("ranked-filtered-seed42", "ranked-filtered-seed43", "ranked-filtered-seed44")
OUT = ROOT / "outputs/analysis/type-contribution-20260917"
RAW = OUT / "raw"
TYPE_NAMES = ("ax_t1pd", "ax_fluid", "cor_t1pd", "cor_fluid", "sag_t1pd", "sag_fluid")
WORKERS = int(sys.argv[1]) if len(sys.argv) > 1 else 12


def image_types(batch):
    presence = batch["series_presence"]
    keep = presence[batch["series_study_index"], batch["series_type_index"]]
    if not bool(keep.all()):
        raise ValueError("Collated series without presence")
    acquisition = batch["slice_mask"].nonzero(as_tuple=True)[0]
    return batch["series_type_index"][acquisition], batch["series_study_index"][acquisition]


def pool_intact(model, encoded, types, dtype):
    with torch.autocast("cuda", dtype=dtype):
        tokens = (
            encoded["features"]
            + encoded["position_1d"][:, None]
            + encoded["position_2d"][None]
            + encoded["plane"][:, None]
            + encoded["contrast"][:, None]
        ).flatten(0, 1)
        normed = model.mil_norm(tokens)
        scores = model.attention(normed)
    valid = encoded["visual_valid"]
    token_study = encoded["study"][:, None].expand_as(valid).flatten()
    token_type = types[:, None].expand_as(valid).flatten()
    valid = valid.flatten()
    batch_size = encoded["batch_size"]
    scores = scores.float()
    contribution = normed.float() @ model.head.weight.float().T
    bias = model.head.bias.float()
    member = token_study[None] == torch.arange(batch_size, device=scores.device)[:, None]
    onehot = torch.nn.functional.one_hot(token_type, len(TYPE_NAMES)).float()
    mask = member & valid[None]
    if not mask.any(1).all():
        raise ValueError("A study has no visually valid spatial tokens")
    logits_input = scores[None].expand(batch_size, -1, -1).masked_fill(~mask[:, :, None], -torch.inf)
    attention = torch.nan_to_num(logits_input.softmax(1), nan=0.0)
    weighted = attention * contribution[None]
    logits = weighted.sum(1) + bias
    probabilities = torch.sigmoid(logits)
    shares = torch.einsum("bnk,nj->bjk", attention, onehot).cpu()
    per_type_contribution = torch.einsum("bnk,nj->bjk", weighted, onehot).cpu()
    token_counts = (member & valid[None]).float() @ onehot
    token_counts = token_counts.cpu()
    return probabilities.cpu(), logits.cpu(), bias.cpu(), shares, per_type_contribution, token_counts


def run(name):
    started = time.perf_counter()
    run_dir = ROOT / "outputs/training" / name
    config = load_config(run_dir / "effective_config.json")
    fold_labels = load_fold_labels(config, 0)
    targets = fold_labels.targets
    studies = fold_labels.validation
    target_columns = [f"target_{target}" for target in targets]
    if not studies.select(f"observed_{target}" for target in targets).to_numpy().all():
        raise ValueError("Validation targets contain unobserved entries")
    target_values = studies.select(target_columns).to_numpy().astype(np.float32)
    selected = json.loads((run_dir / "fold_0/selected_checkpoint.json").read_text())
    checkpoint = run_dir / "fold_0" / Path(selected["path"]).name
    device = torch.device("cuda")
    dtype = torch.bfloat16
    model = load_checkpoint_model(config, targets, checkpoint, device, expected_fold=0)
    multiplier = prediction_image_multiplier(config, "validation")
    dataset = prediction_dataset(config, targets, studies, "train", multiplier)
    loader = build_data_loader(
        dataset, config.trainer, config.trainer.batch_size, num_workers=WORKERS
    )
    model_path = []
    ids = []
    presence_rows = []
    share_rows = []
    contribution_rows = []
    token_rows = []
    logits_rows = []
    bias_value = None
    conditions = 0
    max_pool_vs_model_diff = 0.0
    with torch.inference_mode():
        for batch in loader:
            ids.extend(batch["study_uid"])
            conditions += sum(len(values) for values in batch["failures"])
            types, study_of_image = image_types(batch)
            presence_rows.append(batch["series_presence"].numpy())
            device_batch = move_batch(batch, device)
            with torch.autocast("cuda", dtype=dtype):
                encoded = model.encode_batch(device_batch)
                model_probabilities = torch.sigmoid(
                    model.pool_encoded(encoded).to(torch.float32)
                ).cpu()
            model_path.append(model_probabilities)
            if not torch.equal(encoded["study"].cpu(), study_of_image):
                raise ValueError("Token study order differs from collated order")
            if not torch.equal(
                encoded["image_plane"].cpu(),
                model.spatial_tokens.plane_lookup.cpu()[types],
            ):
                raise ValueError("Token plane order differs from collated types")
            probabilities, logits, bias, shares, contributions, counts = pool_intact(
                model, encoded, types.to(device), dtype
            )
            bias_value = bias.numpy()
            diff = (probabilities - model_probabilities).abs().max().item()
            max_pool_vs_model_diff = max(max_pool_vs_model_diff, diff)
            share_rows.append(shares.numpy())
            contribution_rows.append(contributions.numpy())
            token_rows.append(counts.numpy())
            logits_rows.append(logits.numpy())
            if len(ids) % 160 == 0:
                print(f"{name}: {len(ids)}/{len(dataset)}", flush=True)
    inference_seconds = time.perf_counter() - started
    if ids != studies[STUDY].to_list():
        raise ValueError("Inference order differs from validation membership")
    model_probabilities = torch.cat(model_path).numpy()
    saved = pl.read_csv(run_dir / "fold_0/validation_predictions.csv")
    if saved[STUDY].to_list() != ids:
        raise ValueError("Saved prediction order differs")
    saved_values = saved.select(targets).to_numpy()
    presence = np.concatenate(presence_rows)
    attention_share = np.concatenate(share_rows)
    contributions = np.concatenate(contribution_rows)
    token_counts = np.concatenate(token_rows)
    logits = np.concatenate(logits_rows)
    reconstructed_logit = contributions.sum(axis=1) + bias_value[None]
    reconstruction_max_abs = float(np.abs(reconstructed_logit - logits).max())
    reproduction = {
        "pool_intact_vs_model_path_max_abs": float(max_pool_vs_model_diff),
        "model_path_vs_saved_max_abs": float(np.abs(model_probabilities - saved_values).max()),
        "model_path_vs_saved_mean_abs": float(np.abs(model_probabilities - saved_values).mean()),
        "sum_type_contribution_plus_bias_vs_pool_logits_max_abs": reconstruction_max_abs,
    }
    print(name, json.dumps(reproduction), flush=True)
    RAW.mkdir(parents=True, exist_ok=True)
    np.savez_compressed(
        RAW / f"{name}.npz",
        study=np.asarray(ids),
        targets=np.asarray(targets),
        target_values=target_values,
        presence=presence,
        contribution=contributions,
        bias=bias_value,
        logits=logits,
        attention_share=attention_share,
        token_counts=token_counts,
        saved=saved_values,
        model_path_probabilities=model_probabilities,
    )
    from sklearn.metrics import roc_auc_score

    binary_targets = (target_values >= TARGET_THRESHOLD).astype(np.int32)
    aucs_model = []
    aucs_saved = []
    for k in range(len(targets)):
        aucs_model.append(roc_auc_score(binary_targets[:, k], model_probabilities[:, k]))
        aucs_saved.append(roc_auc_score(binary_targets[:, k], saved_values[:, k]))
    macro_auc_model = float(np.mean(aucs_model))
    macro_auc_saved = float(np.mean(aucs_saved))
    return {
        "run": name,
        "checkpoint": str(checkpoint),
        "checkpoint_score": selected["score"],
        "seed": config.seed,
        "validation_image_multiplier": multiplier,
        "series_types": [kind.model_dump() for kind in config.data.series_types],
        "targets": list(targets),
        "preparation_conditions": conditions,
        "reproduction": reproduction,
        "macro_auc_model_path": macro_auc_model,
        "macro_auc_saved": macro_auc_saved,
        "inference_seconds": inference_seconds,
        "peak_cuda_allocated_bytes": torch.cuda.max_memory_allocated(),
    }


def main():
    configure_cuda_allocator()
    torch.set_float32_matmul_precision("high")
    torch.backends.cudnn.benchmark = False
    started = time.perf_counter()
    reports = [run(name) for name in RUNS]
    RAW.mkdir(parents=True, exist_ok=True)
    (RAW / "inference.json").write_text(
        json.dumps(
            {
                "runs": reports,
                "workers": WORKERS,
                "batch_size": 8,
                "precision": "bf16 autocast",
                "total_seconds": time.perf_counter() - started,
                "torch": torch.__version__,
                "gpu": torch.cuda.get_device_name(0),
            },
            indent=2,
        )
        + "\n"
    )


if __name__ == "__main__":
    main()
