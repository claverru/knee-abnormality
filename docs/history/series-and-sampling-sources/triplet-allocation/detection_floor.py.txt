from knee_abnormality.config import load_config

from knee_abnormality.evaluation.metrics import aucs, bootstrap_auc

ROOT = Path("/home/claudio/src/knee-abnormality")
OUT = ROOT / "outputs/analysis/detection-floor-20260917"
OUT.mkdir(parents=True, exist_ok=True)
RAW_TYPE_IMPORTANCE = ROOT / "outputs/analysis/type-importance-20260917/raw"
RESCORE = ROOT / "outputs/analysis/deterministic-selection-20260917/rescore"
QUEUE = ROOT / "outputs/training/overnight-20260916T214520Z-t8k_kw2z/runs"
RUNS = ("00-control", "01-control-seed-43", "02-control-seed-44")
TYPE_NAMES = ("ax_t1pd", "ax_fluid", "cor_t1pd", "cor_fluid", "sag_t1pd", "sag_fluid")

N_BOOT = 4000
GLOBAL_SEED = 20260917
HALF_SEED = 4045391

config = load_config(ROOT / "configs/training.yaml")
labels = load_fold_labels(config, 0)
targets = list(labels.targets)
validation = labels.validation.sort(STUDY)
study_order = validation[STUDY].to_list()
soft = validation.select(f"target_{t}" for t in targets).to_numpy().astype(np.float64)
hard = (soft >= TARGET_THRESHOLD).astype(np.int8)
n_studies = len(study_order)
assert n_studies == 904

rng_half = np.random.default_rng(HALF_SEED)
half_index = np.sort(rng_half.choice(n_studies, size=n_studies // 2, replace=False))
half_mask_full = np.zeros(n_studies, dtype=bool)
half_mask_full[half_index] = True

type_data = [np.load(RAW_TYPE_IMPORTANCE / f"{run}.npz", allow_pickle=True) for run in RUNS]
for item in type_data:
    assert item["study"].tolist() == study_order
    assert item["targets"].tolist() == targets
variants = type_data[0]["variants"].tolist()
presence = type_data[0]["presence"]
assert all(np.array_equal(presence, item["presence"]) for item in type_data[1:])
inference_meta = json.loads((RAW_TYPE_IMPORTANCE / "inference.json").read_text())
variant_keeps = inference_meta["runs"][0]["variant_keeps"]


def variant_mask(variant):
    if variant.startswith("without_"):
        removed = TYPE_NAMES.index(variant.removeprefix("without_"))
        return presence[:, removed]
    kept = [TYPE_NAMES.index(name) for name in variant_keeps[variant]]
    return presence[:, kept].all(axis=1)


def logit(p):
    p = np.clip(p, 1e-6, 1 - 1e-6)
    return np.log(p) - np.log1p(-p)


def macro_auc(mask, pred):
    return aucs(hard[mask], pred[mask], targets, strict=False)["macro_auc"]


def perturbation_stats(mask, pred_a, pred_b):
    a, b = pred_a[mask], pred_b[mask]
    per_target_spearman = [
        spearmanr(a[:, i], b[:, i]).statistic for i in range(len(targets))
    ]
    return {
        "mean_abs_prob_diff": float(np.abs(a - b).mean()),
        "rms_logit_diff": float(np.sqrt(((logit(a) - logit(b)) ** 2).mean())),
        "mean_target_spearman": float(np.mean(per_target_spearman)),
        "rank_change_proxy": float(1 - np.mean(per_target_spearman)),
    }


def bootstrap_delta(mask, pred_a, pred_b, seed, n_boot=N_BOOT):
    n = int(mask.sum())
    rng = np.random.default_rng(seed)
    counts = rng.multinomial(n, np.full(n, 1.0 / n), size=n_boot)
    boot_a = bootstrap_auc(hard[mask], pred_a[mask], counts)
    boot_b = bootstrap_auc(hard[mask], pred_b[mask], counts)
    delta = boot_a - boot_b
    finite = np.isfinite(delta)
    return delta[finite], n, seed, int((~finite).sum())


def summarize_boot(delta):
    return {
        "se": float(delta.std(ddof=1)),
        "ci_low": float(np.quantile(delta, 0.025)),
        "ci_high": float(np.quantile(delta, 0.975)),
        "n_boot_valid": int(len(delta)),
    }


def build_pairs():
    pairs = []
    for run_name, item in zip(RUNS, type_data):
        preds = {v: item["predictions"][variants.index(v)] for v in variants}
        intact = preds["intact"]
        for variant in variants:
            if variant == "intact":
                continue
            mask = variant_mask(variant)
            pairs.append(
                {
                    "name": f"{run_name}__intact_vs_{variant}",
                    "checkpoint": run_name,
                    "kind": "type-perturbation",
                    "mask": mask,
                    "pred_a": intact,
                    "pred_b": preds[variant],
                }
            )
    for run_name in RUNS:
        new_frame = pl.read_csv(RESCORE / f"{run_name}_validation_predictions.csv").sort(STUDY)
        old_frame = pl.read_csv(QUEUE / run_name / "fold_0/validation_predictions.csv").sort(STUDY)
        assert new_frame[STUDY].to_list() == study_order
        assert old_frame[STUDY].to_list() == study_order
        pred_new = new_frame.select(targets).to_numpy().astype(np.float64)
        pred_old = old_frame.select(targets).to_numpy().astype(np.float64)
        pairs.append(
            {
                "name": f"{run_name}__new_vs_old_eval",
                "checkpoint": run_name,
                "kind": "tiny-perturbation",
                "mask": np.ones(n_studies, dtype=bool),
                "pred_a": pred_new,
                "pred_b": pred_old,
            }
        )
    return pairs


pairs = build_pairs()
rows = []
tiny_pair_arrays = {}
for i, p in enumerate(pairs):
    mask = p["mask"]
    pred_a, pred_b = p["pred_a"], p["pred_b"]
    n = int(mask.sum())
    delta_point = macro_auc(mask, pred_a) - macro_auc(mask, pred_b)
    pert = perturbation_stats(mask, pred_a, pred_b)
    boot_full, n_full, seed_full, dropped_full = bootstrap_delta(
        mask, pred_a, pred_b, seed=GLOBAL_SEED + i
    )
    full_summary = summarize_boot(boot_full)

    half_mask = mask & half_mask_full
    n_half = int(half_mask.sum())
    delta_half_point = macro_auc(half_mask, pred_a) - macro_auc(half_mask, pred_b)
    boot_half, n_half_b, seed_half, dropped_half = bootstrap_delta(
        half_mask, pred_a, pred_b, seed=GLOBAL_SEED + 10_000 + i
    )
    half_summary = summarize_boot(boot_half)

    row = {
        "name": p["name"],
        "checkpoint": p["checkpoint"],
        "kind": p["kind"],
        "n_full": n,
        "delta_full": delta_point,
        "se_full": full_summary["se"],
        "ci_low_full": full_summary["ci_low"],
        "ci_high_full": full_summary["ci_high"],
        "dropped_resamples_full": dropped_full,
        "n_half": n_half,
        "delta_half": delta_half_point,
        "se_half": half_summary["se"],
        "ci_low_half": half_summary["ci_low"],
        "ci_high_half": half_summary["ci_high"],
        "dropped_resamples_half": dropped_half,
        "se_half_over_full": half_summary["se"] / full_summary["se"] if full_summary["se"] else None,
        **pert,
    }
    rows.append(row)
    if p["kind"] == "tiny-perturbation":
        tiny_pair_arrays[p["checkpoint"]] = (mask, pred_a, pred_b)

pairs_frame = pl.DataFrame(rows)
pairs_frame.write_csv(OUT / "pairs.csv")

def joint_bootstrap_group(mask, preds_by_run, seed):
    n = int(mask.sum())
    rng = np.random.default_rng(seed)
    counts = rng.multinomial(n, np.full(n, 1.0 / n), size=N_BOOT)
    per_checkpoint_boot = []
    for pred_a, pred_b in preds_by_run:
        boot_a = bootstrap_auc(hard[mask], pred_a[mask], counts)
        boot_b = bootstrap_auc(hard[mask], pred_b[mask], counts)
        per_checkpoint_boot.append(boot_a - boot_b)
    per_checkpoint_boot = np.stack(per_checkpoint_boot, axis=0)
    joint_mean_delta = per_checkpoint_boot.mean(axis=0)
    finite_joint = np.isfinite(joint_mean_delta)
    joint_mean_delta = joint_mean_delta[finite_joint]
    individual_se = [
        float(per_checkpoint_boot[k][np.isfinite(per_checkpoint_boot[k])].std(ddof=1))
        for k in range(len(preds_by_run))
    ]
    n_runs = len(preds_by_run)
    pairwise_corr = {
        f"{a}~{b}": float(
            np.corrcoef(
                per_checkpoint_boot[a][np.isfinite(per_checkpoint_boot[a]) & np.isfinite(per_checkpoint_boot[b])],
                per_checkpoint_boot[b][np.isfinite(per_checkpoint_boot[a]) & np.isfinite(per_checkpoint_boot[b])],
            )[0, 1]
        )
        for a in range(n_runs)
        for b in range(a + 1, n_runs)
    }
    return {
        "seed": seed,
        "n_boot": N_BOOT,
        "n_studies": n,
        "individual_se": individual_se,
        "mean_of_individual_se": float(np.mean(individual_se)),
        "naive_se_over_sqrtN": float(np.mean(individual_se) / np.sqrt(n_runs)),
        "joint_se": float(joint_mean_delta.std(ddof=1)),
        "joint_ci_low": float(np.quantile(joint_mean_delta, 0.025)),
        "joint_ci_high": float(np.quantile(joint_mean_delta, 0.975)),
        "n_boot_valid": int(len(joint_mean_delta)),
        "pairwise_correlation": pairwise_corr,
        "joint_over_mean_individual": float(
            joint_mean_delta.std(ddof=1) / np.mean(individual_se)
        ),
    }

def fit_loglog(x, y):
    fit = linregress(np.log(x), np.log(y))
    return fit, (lambda p, fit=fit: float(np.exp(fit.intercept) * p ** fit.slope))


fit_data = pairs_frame.filter(pl.col("se_full") > 0)

def linear_se(perturbation):
    return float(linear_fit.intercept + linear_fit.slope * perturbation)
