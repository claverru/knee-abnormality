import sys
import json
import numpy as np

sys.path.insert(0, "outputs/analysis/token-cache-20260917")
import cache_lib

cache = cache_lib.load()
SLOTS = list(cache_lib.SLOTS)
DEFAULT = tuple(cache_lib.DEFAULT_COUNTS)
CANDIDATE = (0, 9, 6, 15, 9, 15)

results = {}
for half in ("search", "confirm"):
    d = cache.score_rule(DEFAULT, half=half)
    c = cache.score_rule(CANDIDATE, half=half)
    results[half] = {
        "default_macro": d["macro_auc_mean"],
        "default_mean_triplets": d["mean_triplets"],
        "candidate_macro": c["macro_auc_mean"],
        "candidate_mean_triplets": c["mean_triplets"],
        "delta": c["macro_auc_mean"] - d["macro_auc_mean"],
        "default_per_run": {r: d["runs"][r]["macro_auc"] for r in cache.runs},
        "candidate_per_run": {r: c["runs"][r]["macro_auc"] for r in cache.runs},
        "default_per_label": {t: float(np.mean([d["runs"][r]["per_target_auc"][t] for r in cache.runs])) for t in cache.targets},
        "candidate_per_label": {t: float(np.mean([c["runs"][r]["per_target_auc"][t] for r in cache.runs])) for t in cache.targets},
    }
    # p = mean abs probability change, averaged over runs, on evaluated studies of this half
    mask = cache.half_mask(half)
    probs_d = np.stack([d["runs"][r]["probabilities"] for r in cache.runs])
    probs_c = np.stack([c["runs"][r]["probabilities"] for r in cache.runs])
    p = float(np.nanmean(np.abs(probs_c - probs_d)))
    results[half]["p_mean_abs_prob_change"] = p

    # paired bootstrap over studies, 500 resamples, seed 0
    rng = np.random.default_rng(0)
    idx = np.where(mask)[0]
    n = len(idx)
    deltas = []
    hard = cache.hard[idx]
    for _ in range(500):
        sample = rng.integers(0, n, n)
        s_idx = idx[sample]
        pd_ = probs_d[:, s_idx, :]
        pc_ = probs_c[:, s_idx, :]
        hd = cache.hard[s_idx]
        macro_d = np.mean([cache_lib.aucs(hd, pd_[i], cache.targets)["macro_auc"] for i in range(3)])
        macro_c = np.mean([cache_lib.aucs(hd, pc_[i], cache.targets)["macro_auc"] for i in range(3)])
        deltas.append(macro_c - macro_d)
    deltas = np.asarray(deltas)
    results[half]["bootstrap_delta_mean"] = float(deltas.mean())
    results[half]["bootstrap_delta_ci95"] = [float(np.percentile(deltas, 2.5)), float(np.percentile(deltas, 97.5))]

with open("outputs/analysis/allocation-search-theory-20260917/confirm_results.json", "w") as f:
    json.dump(results, f, indent=2)

print(json.dumps(results, indent=2))
