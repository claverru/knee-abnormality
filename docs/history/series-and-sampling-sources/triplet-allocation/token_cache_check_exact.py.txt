import json
import sys
import time
from pathlib import Path

import numpy as np
import polars as pl
import torch

HERE = Path(__file__).resolve().parent
sys.path.insert(0, str(HERE))

import cache_lib  # noqa: E402
import exact  # noqa: E402
import knee_abnormality  # noqa: E402

ROOT = Path("/home/claudio/src/knee-abnormality")
RULES = {
    "default": cache_lib.DEFAULT_COUNTS,
    "training": cache_lib.TRAINING_COUNTS,
    "default_x1.5": (18, 18, 15, 15, 15, 15),
    "budget_a_eval": (6, 12, 10, 14, 10, 12),
    "budget_b_eval": (6, 12, 8, 14, 8, 10),
}


def main():
    print(knee_abnormality.__file__, flush=True)
    started = time.perf_counter()
    workers = int(sys.argv[1]) if len(sys.argv) > 1 else 24
    cache = cache_lib.TokenCache()
    models = exact.load_models(cache.runs, torch.device("cuda"))
    rows, results = [], {}
    for name, rule in RULES.items():
        cached = cache.score_rule(rule)
        precise = exact.exact_score(cache, rule=rule, workers=workers, models=models)
        results[name] = (cached, precise)
        row = {
            "rule": name,
            "counts": "/".join(map(str, rule)),
            "mean_triplets": cached["mean_triplets"],
            "exact_seconds": precise["seconds"],
        }
        for run in cache.runs:
            a = cached["runs"][run]
            b = precise["runs"][run]
            short = run.removeprefix("ranked-filtered-")
            row[f"{short}_cached"] = a["macro_auc"]
            row[f"{short}_exact"] = b["macro_auc"]
            row[f"{short}_prob_max_abs"] = float(np.abs(a["probabilities"] - b["probabilities"]).max())
            row[f"{short}_prob_mean_abs"] = float(np.abs(a["probabilities"] - b["probabilities"]).mean())
            row[f"{short}_logit_mean_abs"] = float(np.abs(a["logits"] - b["logits"]).mean())
            row[f"{short}_max_label_auc_abs_diff"] = max(
                abs(a["per_target_auc"][t] - b["per_target_auc"][t]) for t in cache.targets
            )
            if name == "default":
                saved = pl.read_csv(ROOT / "outputs/training" / run / "fold_0/validation_predictions.csv")
                values = saved.select(cache.targets).to_numpy()
                row[f"{short}_exact_vs_saved_prob_max_abs"] = float(np.abs(b["probabilities"] - values).max())
                row[f"{short}_exact_vs_saved_prob_mean_abs"] = float(np.abs(b["probabilities"] - values).mean())
                row[f"{short}_saved"] = cache.evaluate(values)["macro_auc"]
        row["cached_mean"] = cached["macro_auc_mean"]
        row["exact_mean"] = precise["macro_auc_mean"]
        rows.append(row)
        print(json.dumps(row), flush=True)
    frame = pl.DataFrame(rows)
    base = frame.filter(pl.col("rule") == "default").row(0, named=True)
    deltas = []
    for row in frame.iter_rows(named=True):
        entry = {"rule": row["rule"]}
        for run in cache.runs:
            short = run.removeprefix("ranked-filtered-")
            entry[f"{short}_delta_cached"] = row[f"{short}_cached"] - base[f"{short}_cached"]
            entry[f"{short}_delta_exact"] = row[f"{short}_exact"] - base[f"{short}_exact"]
        entry["mean_delta_cached"] = row["cached_mean"] - base["cached_mean"]
        entry["mean_delta_exact"] = row["exact_mean"] - base["exact_mean"]
        entry["mean_delta_error"] = entry["mean_delta_cached"] - entry["mean_delta_exact"]
        deltas.append(entry)
    frame.write_csv(HERE / "checks/exact_vs_cached.csv")
    pl.DataFrame(deltas).write_csv(HERE / "checks/exact_vs_cached_deltas.csv")
    (HERE / "checks/exact_vs_cached.json").write_text(
        json.dumps({"rules": rows, "deltas": deltas, "seconds": time.perf_counter() - started}, indent=2) + "\n"
    )
    print(pl.DataFrame(deltas))


if __name__ == "__main__":
    main()
