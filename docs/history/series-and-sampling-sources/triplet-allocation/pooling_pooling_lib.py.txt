import importlib.util
import sys
from pathlib import Path

import numpy as np
import polars as pl

ROOT = Path(__file__).resolve().parent
CACHE1_DIR = ROOT.parent / "token-cache-20260917"
CACHE2_DIR = ROOT.parent / "token-cache-allocation-20260917"


def _load_cache_module(cache_dir, module_name):
    spec = importlib.util.spec_from_file_location(module_name, cache_dir / "cache_lib.py")
    module = importlib.util.module_from_spec(spec)
    sys.modules[module_name] = module
    spec.loader.exec_module(module)
    return module


def load_cache(which):
    cache_dir = CACHE1_DIR if which == "main" else CACHE2_DIR
    lib = _load_cache_module(cache_dir, f"cache_lib_{which}")
    return lib.load()


def series_codes(cache):
    codes = cache.triplets["series"].cast(pl.Categorical).to_physical()
    return codes.to_numpy().astype(np.int64)


def token_logits(cache, run, ids, per_study, tau=1.0, count_norm=False, series_code=None):
    prep = prepare_allocation(cache, ids, per_study, series_code=series_code, need_count_norm=count_norm)
    return pool_from_prep(cache, run, prep, tau=tau, count_norm=count_norm)


def prepare_allocation(cache, ids, per_study, series_code=None, need_count_norm=False):
    n_studies = len(cache.studies)
    per_study = np.asarray(per_study, dtype=np.int64)
    present = per_study > 0
    present_idx = np.nonzero(present)[0]
    n_present = present_idx.size
    if n_present == 0:
        return {"n_studies": n_studies, "present_idx": present_idx, "n_present": 0}

    triplet_starts = (np.cumsum(per_study) - per_study)[present]
    triplet_segment = np.repeat(np.arange(n_present), per_study[present])
    valid = np.asarray(cache.visual_valid[ids])
    triplet_valid_counts = valid.sum(axis=1)
    token_segment = np.repeat(triplet_segment, triplet_valid_counts)
    seg_valid_counts = np.add.reduceat(triplet_valid_counts, triplet_starts)
    tokenstarts = np.cumsum(seg_valid_counts) - seg_valid_counts

    token_series_total = None
    if need_count_norm:
        triplet_series = series_code[ids]
        key_df = pl.DataFrame(
            {
                "segment": triplet_segment,
                "series": triplet_series,
                "valid_count": triplet_valid_counts,
            }
        )
        totals = key_df.group_by(["segment", "series"]).agg(
            pl.col("valid_count").sum().alias("series_total")
        )
        key_df = key_df.join(totals, on=["segment", "series"], how="left")
        triplet_series_total = key_df["series_total"].to_numpy()
        token_series_total = np.repeat(triplet_series_total, triplet_valid_counts)

    return {
        "n_studies": n_studies,
        "ids": ids,
        "valid": valid,
        "present_idx": present_idx,
        "n_present": n_present,
        "tokenstarts": tokenstarts,
        "seg_valid_counts": seg_valid_counts,
        "token_series_total": token_series_total,
    }


def pool_from_prep(cache, run, prep, tau=1.0, count_norm=False):
    full = np.full((prep["n_studies"], 12), np.nan)
    if prep["n_present"] == 0:
        return full

    scores = np.asarray(cache.scores[run][prep["ids"]], dtype=np.float32)
    contribution = np.asarray(cache.contribution[run][prep["ids"]], dtype=np.float32)
    valid = prep["valid"]
    token_scores = scores[valid].astype(np.float64)
    token_contrib = contribution[valid].astype(np.float64)

    if count_norm:
        adjusted = token_scores - np.log(prep["token_series_total"])[:, None]
    else:
        adjusted = token_scores / tau

    tokenstarts = prep["tokenstarts"]
    seg_valid_counts = prep["seg_valid_counts"]

    peak = np.maximum.reduceat(adjusted, tokenstarts, axis=0)
    peak_expanded = np.repeat(peak, seg_valid_counts, axis=0)
    weights = np.exp(adjusted - peak_expanded)
    denom = np.add.reduceat(weights, tokenstarts, axis=0)
    denom_expanded = np.repeat(denom, seg_valid_counts, axis=0)
    weights_norm = weights / denom_expanded
    pooled = np.add.reduceat(weights_norm * token_contrib, tokenstarts, axis=0)
    logits_present = pooled + np.asarray(cache.bias[run], dtype=np.float64)[None, :]

    full[prep["present_idx"]] = logits_present
    return full


def score_variant(cache, rule, tau=1.0, count_norm=False, half=None, mask=None, series_code=None, runs=None):
    evaluated = cache.half_mask(half) if mask is None else np.asarray(mask)
    ids, per_study = cache.allocation_ids(rule, evaluated)
    result = {"studies": int(evaluated.sum()), "mean_triplets": float(np.asarray(per_study)[evaluated].mean())}
    run_results = {}
    for run in runs or cache.runs:
        logits = token_logits(cache, run, ids, per_study, tau=tau, count_norm=count_norm, series_code=series_code)
        probabilities = 1 / (1 + np.exp(-logits))
        metrics = cache.evaluate(probabilities, evaluated)
        run_results[run] = {"probabilities": probabilities, "logits": logits, **metrics}
    result["runs"] = run_results
    result["macro_auc_mean"] = float(np.mean([v["macro_auc"] for v in run_results.values()]))
    result["counts"] = cache.counts_for(rule)
    return result
