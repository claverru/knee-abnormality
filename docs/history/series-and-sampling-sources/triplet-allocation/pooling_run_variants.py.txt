import json
import sys
import time

import numpy as np
import pooling_lib as pl_

WHICH = sys.argv[1] if len(sys.argv) > 1 else "main"

ALLOCATIONS = [
    ("uniform_1", (1, 1, 1, 1, 1, 1)),
    ("uniform_2", (2, 2, 2, 2, 2, 2)),
    ("uniform_4", (4, 4, 4, 4, 4, 4)),
    ("uniform_6", (6, 6, 6, 6, 6, 6)),
    ("uniform_8", (8, 8, 8, 8, 8, 8)),
    ("uniform_12", (12, 12, 12, 12, 12, 12)),
    ("uniform_16", (16, 16, 16, 16, 16, 16)),
    ("uniform_24", (24, 24, 24, 24, 24, 24)),
    ("default", (12, 12, 10, 10, 10, 10)),
    ("all", (1000, 1000, 1000, 1000, 1000, 1000)),
]

VARIANTS = [
    ("tau_0.75", dict(tau=0.75, count_norm=False)),
    ("tau_1.0", dict(tau=1.0, count_norm=False)),
    ("tau_1.5", dict(tau=1.5, count_norm=False)),
    ("count_norm", dict(tau=1.0, count_norm=True)),
]


def main():
    cache = pl_.load_cache(WHICH)
    series_code = pl_.series_codes(cache)
    full_mask = np.ones(len(cache.studies), dtype=bool)
    halves = ["full"] + (["search", "confirm"] if cache.split is not None else [])
    half_masks = {"full": full_mask}
    if cache.split is not None:
        half_masks["search"] = cache.half_mask("search")
        half_masks["confirm"] = cache.half_mask("confirm")

    rows = []
    t0 = time.time()
    for alloc_name, rule in ALLOCATIONS:
        ids, per_study = cache.allocation_ids(rule, full_mask)
        per_study = np.asarray(per_study)
        need_count_norm = any(kwargs.get("count_norm") for _, kwargs in VARIANTS)
        prep = pl_.prepare_allocation(cache, ids, per_study, series_code=series_code, need_count_norm=need_count_norm)

        for variant_name, kwargs in VARIANTS:
            probs_by_run = {}
            for run in cache.runs:
                logits = pl_.pool_from_prep(cache, run, prep, **kwargs)
                probs_by_run[run] = 1 / (1 + np.exp(-logits))

            for half in halves:
                mask = half_masks[half]
                run_aucs = {}
                for run in cache.runs:
                    metrics = cache.evaluate(probs_by_run[run], mask)
                    run_aucs[run] = metrics["macro_auc"]
                mean_triplets = float(per_study[mask].mean())
                macro_auc_mean = float(np.mean(list(run_aucs.values())))
                rows.append(
                    {
                        "cache": WHICH,
                        "allocation": alloc_name,
                        "half": half,
                        "variant": variant_name,
                        "mean_triplets": mean_triplets,
                        "macro_auc_mean": macro_auc_mean,
                        "run_aucs": {k: float(v) for k, v in run_aucs.items()},
                    }
                )
            print(
                WHICH,
                alloc_name,
                variant_name,
                round(mean_triplets, 1),
                round(macro_auc_mean, 5),
                f"{time.time()-t0:.0f}s",
                flush=True,
            )
    out_path = f"results_{WHICH}.json"
    with open(out_path, "w") as f:
        json.dump(rows, f, indent=2)
    print("wrote", out_path)


if __name__ == "__main__":
    main()
