def apply_shares(counts, idxs, shares, budget):
    counts = list(counts)
    floor_shares = [int(np.floor(s)) for s in shares]
    remainder = budget - sum(floor_shares)
    order = sorted(range(len(idxs)), key=lambda i: -(shares[i] - floor_shares[i]))
    add = list(floor_shares)
    i = 0
    while remainder > 0 and order:
        add[order[i % len(order)]] += 1
        remainder -= 1
        i += 1
    for idx, a in zip(idxs, add):
        counts[idx] += a
    return counts


def proportional_rule(available, capacity):
    counts = list(DEFAULT)
    missing = [i for i in range(6) if not available[i]]
    present = [i for i in range(6) if available[i]]
    if not missing or not present:
        return counts
    total_budget = sum(counts[i] for i in missing)
    for i in missing:
        counts[i] = 0
    wsum = sum(DEFAULT[i] for i in present)
    if wsum <= 0:
        return counts
    shares = [total_budget * DEFAULT[i] / wsum for i in present]
    return apply_shares(counts, present, shares, total_budget)


PLANE_PARTNER = {0: 1, 1: 0, 2: 3, 3: 2, 4: 5, 5: 4}
COR_T1PD = 2


def hybrid_rule(available, capacity):
    counts = list(DEFAULT)
    missing = [i for i in range(6) if not available[i]]
    if not missing:
        return counts
    leftover_types, leftover_budget = [], 0
    for t in missing:
        budget = counts[t]
        counts[t] = 0
        if t == COR_T1PD and available[PLANE_PARTNER[t]]:
            counts[PLANE_PARTNER[t]] += budget
        else:
            leftover_budget += budget
            leftover_types.append(t)
    if leftover_budget > 0:
        present = [i for i in range(6) if available[i] and i not in leftover_types]
        wsum = sum(DEFAULT[i] for i in present)
        if wsum > 0:
            shares = [leftover_budget * DEFAULT[i] / wsum for i in present]
            counts = apply_shares(counts, present, shares, leftover_budget)
    return counts


def score(rule_or_counts, mask, runs=None):
    runs = runs or cache.runs
    ids, per_study = cache.allocation_ids(rule_or_counts, mask)
    logits = {run: cache.logits_from_ids(run, ids, per_study) for run in runs}
    result = {"studies": int(mask.sum()), "mean_triplets": float(per_study[mask].mean())}
    result["runs"] = {}
    for run in runs:
        probs = 1 / (1 + np.exp(-logits[run]))
        m = cache_lib.aucs(cache.hard[mask], probs[mask], cache.targets, strict=False)
        result["runs"][run] = {"probabilities": probs, **m}
    macro_vals = [v["macro_auc"] for v in result["runs"].values() if v["macro_auc"] is not None]
    result["macro_auc_mean"] = float(np.mean(macro_vals)) if macro_vals else None
    result["per_run_macro"] = {run: result["runs"][run]["macro_auc"] for run in runs}
    return result


def mean_abs_prob_delta(res_a, res_b, mask):
    diffs = []
    for run in cache.runs:
        pa = res_a["runs"][run]["probabilities"][mask]
        pb = res_b["runs"][run]["probabilities"][mask]
        diffs.append(float(np.nanmean(np.abs(pa - pb))))
    return float(np.mean(diffs)), diffs


def paired_bootstrap_macro(res_a, res_b, mask, n_boot=1000, seed=0):
    rng = np.random.default_rng(seed)
    idx = np.where(mask)[0]
    n = len(idx)
    per_run = []
    for run in cache.runs:
        pa = res_a["runs"][run]["probabilities"]
        pb = res_b["runs"][run]["probabilities"]
        y = cache.hard
        boots = np.empty(n_boot)
        for b in range(n_boot):
            sample = rng.choice(idx, size=n, replace=True)
            ma = cache_lib.aucs(y[sample], pa[sample], cache.targets, strict=False)["macro_auc"]
            mb = cache_lib.aucs(y[sample], pb[sample], cache.targets, strict=False)["macro_auc"]
            boots[b] = (ma - mb) if (ma is not None and mb is not None) else np.nan
        per_run.append(boots)
    stacked = np.nanmean(per_run, axis=0)
    lo, hi = np.nanpercentile(stacked, [2.5, 97.5])
    return {"mean": float(np.nanmean(stacked)), "ci95": [float(lo), float(hi)], "seed": seed, "n_boot": n_boot}
