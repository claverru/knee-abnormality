import json
from pathlib import Path

import numpy as np

OUT = Path("/home/claudio/src/knee-abnormality/outputs/analysis/allocation-neyman-20260917")
DEFAULT = (12, 12, 10, 10, 10, 10)

_raw = json.loads((OUT / "pattern_weights.json").read_text())

WEIGHTS_BY_AVAIL = {}
for pattern, r in _raw.items():
    present = r["present"]
    key = tuple(i in present for i in range(6))
    vecs = {}
    for method in ("neyman_mean_label", "neyman_worst_label", "sep_share_mean_label", "efficiency_norm"):
        vec = np.zeros(6)
        for pos, val in zip(present, r[method]):
            vec[pos] = val
        vecs[method] = vec
    WEIGHTS_BY_AVAIL[key] = vecs


def largest_remainder(raw_floats, budget):
    floor = np.floor(raw_floats).astype(int)
    remainder = raw_floats - floor
    deficit = int(budget) - int(floor.sum())
    if deficit > 0:
        order = np.argsort(-remainder)
        floor[order[:deficit]] += 1
    elif deficit < 0:
        order = np.argsort(remainder)
        floor[order[: -deficit]] = np.maximum(floor[order[: -deficit]] - 1, 0)
    return floor


def allocate(weights, budget, capacity):
    weights = np.asarray(weights, dtype=float)
    capacity = np.asarray(capacity, dtype=int)
    n = len(weights)
    counts = np.zeros(n, dtype=int)
    active = weights > 0
    remaining = int(budget)
    for _ in range(10):
        if remaining <= 0 or not active.any():
            break
        w = np.where(active, weights, 0.0)
        wsum = w.sum()
        if wsum <= 0:
            break
        raw = remaining * w / wsum
        add = largest_remainder(raw, remaining)
        add = np.maximum(add, 0)
        headroom = capacity - counts
        over = add > headroom
        if over.any():
            add = np.where(over, headroom, add)
            counts = counts + add
            active = active & (counts < capacity)
            remaining = budget - int(counts.sum())
        else:
            counts = counts + add
            remaining = 0
    return counts


def make_rule(method, budget_kind, fixed_budget=None):
    def rule(available, capacity):
        key = tuple(bool(v) for v in available)
        weights = WEIGHTS_BY_AVAIL.get(key)
        if weights is None:
            weights = {m: np.array([1.0 if a else 0.0 for a in available]) for m in (
                "neyman_mean_label", "neyman_worst_label", "sep_share_mean_label", "efficiency_norm"
            )}
        w = weights[method]
        if budget_kind == "variant1":
            budget = sum(DEFAULT[s] for s in range(6) if available[s])
        elif budget_kind == "variant2":
            budget = fixed_budget
        else:
            raise ValueError(budget_kind)
        return allocate(w, budget, capacity)

    return rule


def diagnostic_rule(W_per_study, S_per_study, study_index):
    def rule_factory(study):
        i = study_index[study]

        def rule(available, capacity):
            wraw = (W_per_study[i] * S_per_study[i]).mean(axis=1)
            budget = sum(DEFAULT[s] for s in range(6) if available[s])
            return allocate(wraw, budget, capacity)

        return rule

    return rule_factory
