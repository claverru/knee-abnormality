import sys
import time
import numpy as np
import polars as pl
from scipy.optimize import curve_fit

sys.path.insert(0, "outputs/analysis/token-cache-20260917")
import cache_lib

cache = cache_lib.load()
SLOTS = list(cache_lib.SLOTS)
DEFAULT = list(cache_lib.DEFAULT_COUNTS)

t0 = time.time()
default_res = cache.score_rule(tuple(DEFAULT), half="search")
default_macro = default_res["macro_auc_mean"]
default_mean_triplets = default_res["mean_triplets"]
print("default", default_macro, default_mean_triplets)

fracs = [0.0, 0.5, 1.0, 1.5, 2.5]
curves = {}
for k, name in enumerate(SLOTS):
    xs, ys = [], []
    for f in fracs:
        n = int(round(DEFAULT[k] * f))
        counts = list(DEFAULT)
        counts[k] = n
        r = cache.score_rule(tuple(counts), half="search")
        xs.append(n)
        ys.append(r["macro_auc_mean"])
    curves[name] = (xs, ys)
    print(name, list(zip(xs, ys)))

def sat(n, a, w, alpha):
    return a + w * (1 - np.exp(-alpha * np.asarray(n)))

fits = {}
for name, (xs, ys) in curves.items():
    xs_a = np.asarray(xs, dtype=float)
    ys_a = np.asarray(ys, dtype=float)
    a0 = ys_a.min()
    w0 = max(ys_a.max() - ys_a.min(), 1e-4)
    try:
        popt, _ = curve_fit(
            sat, xs_a, ys_a, p0=[a0, w0, 0.1], bounds=([0.5, 0.0, 1e-4], [1.0, 0.5, 5.0]), maxfev=5000
        )
    except Exception as e:
        popt = [a0, w0, 0.1]
        print("fit failed", name, e)
    fits[name] = tuple(popt)
    print("fit", name, popt)

print("fit time", time.time() - t0)

def marginal(name, n):
    a, w, alpha = fits[name]
    return w * alpha * np.exp(-alpha * max(n, 0))

def waterfill(target_total, step=1, max_n=200):
    n = {name: 0 for name in SLOTS}
    total = 0
    while total < target_total:
        gains = {name: marginal(name, n[name]) for name in SLOTS if n[name] < max_n}
        if not gains:
            break
        best = max(gains, key=gains.get)
        n[best] += step
        total += step
    return n

target_requested = sum(DEFAULT)  # 64, matches default's requested sum
alloc = waterfill(target_requested)
print("waterfill counts (requested)", alloc)

candidate = tuple(alloc[name] for name in SLOTS)
cand_res = cache.score_rule(candidate, half="search")
print("candidate search", candidate, cand_res["macro_auc_mean"], cand_res["mean_triplets"])

# adjust total to match default's realised mean_triplets within 1%
lo, hi = 20, 160
best_candidate, best_res, best_diff = None, None, 1e9
for total in range(30, 100, 2):
    alloc_t = waterfill(total)
    cand_t = tuple(alloc_t[name] for name in SLOTS)
    res_t = cache.score_rule(cand_t, half="search")
    diff = abs(res_t["mean_triplets"] - default_mean_triplets) / default_mean_triplets
    if diff < best_diff:
        best_diff = diff
        best_candidate = cand_t
        best_res = res_t

print("best matched-compute candidate", best_candidate, best_res["macro_auc_mean"], best_res["mean_triplets"], "diff", best_diff)

import json
out = {
    "default": {"counts": DEFAULT, "macro_auc_mean": default_macro, "mean_triplets": default_mean_triplets},
    "curves": curves,
    "fits": fits,
    "candidate_requested_64": {"counts": list(candidate), "macro_auc_mean": cand_res["macro_auc_mean"], "mean_triplets": cand_res["mean_triplets"]},
    "candidate_matched_compute": {"counts": list(best_candidate), "macro_auc_mean": best_res["macro_auc_mean"], "mean_triplets": best_res["mean_triplets"], "compute_diff_pct": best_diff * 100},
}
with open("outputs/analysis/allocation-search-theory-20260917/waterfill_search.json", "w") as f:
    json.dump(out, f, indent=2)
print("total time", time.time() - t0)
