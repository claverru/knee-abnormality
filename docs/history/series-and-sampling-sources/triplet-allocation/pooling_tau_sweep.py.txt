import json
import sys

import numpy as np
import pooling_lib as pl_

TAUS = [0.4, 0.5, 0.6, 0.7, 0.75, 0.85, 1.0]
RULE = (12, 12, 10, 10, 10, 10)

rows = []
which = sys.argv[1]
for _ in (0,):
    cache = pl_.load_cache(which)
    full = np.ones(len(cache.studies), dtype=bool)
    ids, per_study = cache.allocation_ids(RULE, full)
    prep = pl_.prepare_allocation(cache, ids, np.asarray(per_study))
    masks = {"full": full, "search": cache.half_mask("search"), "confirm": cache.half_mask("confirm")}
    base = {}
    for tau in TAUS:
        probs = {run: 1 / (1 + np.exp(-pl_.pool_from_prep(cache, run, prep, tau=tau))) for run in cache.runs}
        for half, mask in masks.items():
            aucs = {run: float(cache.evaluate(probs[run], mask)["macro_auc"]) for run in cache.runs}
            row = {"cache": which, "tau": tau, "half": half, "macro": float(np.mean(list(aucs.values()))), "runs": aucs}
            if tau == 1.0:
                base[half] = row["macro"]
            rows.append(row)
        print(which, tau, {h: round(r["macro"], 5) for h, r in [(x["half"], x) for x in rows[-3:]]}, flush=True)
    for row in rows:
        if row["cache"] == which:
            row["delta_vs_tau1"] = row["macro"] - base[row["half"]]
json.dump(rows, open(f"tau_sweep_{which}.json", "w"), indent=2)
