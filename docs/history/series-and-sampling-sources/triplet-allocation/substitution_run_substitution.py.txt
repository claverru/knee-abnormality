def apply_shares(counts, idxs, shares, budget):
    counts = list(counts)
    floor_shares = [int(np.floor(s)) for s in shares]
    remainder = budget - sum(floor_shares)
    order = sorted(range(len(idxs)), key=lambda i: -(shares[i] - floor_shares[i]))
    add = list(floor_shares)
    i = 0
    while remainder > 0 and order:
        add[order[i % len(order)]] += 1
        remainder -= 1
        i += 1
    for idx, a in zip(idxs, add):
        counts[idx] += a
    return counts


def make_rule(t, strategy, u=None):
    def rule(available, capacity):
        counts = list(DEFAULT)
        budget = counts[t]
        counts[t] = 0
        if strategy == "lost":
            pass
        elif strategy == "proportional":
            others = [i for i in range(6) if i != t and available[i]]
            wsum = sum(DEFAULT[i] for i in others)
            if wsum > 0:
                shares = [budget * DEFAULT[i] / wsum for i in others]
                counts = apply_shares(counts, others, shares, budget)
        elif strategy == "same_plane":
            partner = PLANE_PARTNER[t]
            if available[partner]:
                counts[partner] += budget
        elif strategy == "single":
            if available[u]:
                counts[u] += budget
        elif strategy == "fill_chain":
            partner = PLANE_PARTNER[t]
            chain = []
            for c in (partner, 3, 1, 5):
                if c != t and c not in chain:
                    chain.append(c)
            remaining = budget
            for c in chain:
                if remaining <= 0:
                    break
                if not available[c]:
                    continue
                room = max(capacity[c] - counts[c], 0)
                take = min(room, remaining)
                counts[c] += take
                remaining -= take
        elif strategy == "fluid_equal":
            fl = [i for i in FLUID if i != t and available[i]]
            if fl:
                shares = [budget / len(fl)] * len(fl)
                counts = apply_shares(counts, fl, shares, budget)
        else:
            raise ValueError(strategy)
        return counts

    return rule


def score(rule_or_counts, mask, runs=None):
    runs = runs or cache.runs
    ids, per_study = cache.allocation_ids(rule_or_counts, mask)
    logits = {run: cache.logits_from_ids(run, ids, per_study) for run in runs}
    result = {"studies": int(mask.sum()), "mean_triplets": float(per_study[mask].mean())}
    result["runs"] = {}
    for run in runs:
        probs = 1 / (1 + np.exp(-logits[run]))
        m = cache_lib.aucs(cache.hard[mask], probs[mask], cache.targets, strict=False)
        result["runs"][run] = {"probabilities": probs, **m}
    macro_vals = [v["macro_auc"] for v in result["runs"].values() if v["macro_auc"] is not None]
    result["macro_auc_mean"] = float(np.mean(macro_vals)) if macro_vals else None
    result["counts"] = cache.counts_for(rule_or_counts)
    return result


def per_label_mean(result):
    out = {}
    for target in cache.targets:
        vals = [r["per_target_auc"].get(target) for r in result["runs"].values()]
        vals = [v for v in vals if v is not None]
        out[target] = float(np.mean(vals)) if vals else None
    return out


def mean_abs_prob_delta(res_a, res_b, mask):
    diffs = []
    for run in cache.runs:
        pa = res_a["runs"][run]["probabilities"][mask]
        pb = res_b["runs"][run]["probabilities"][mask]
        diffs.append(np.nanmean(np.abs(pa - pb)))
    return float(np.mean(diffs))


def paired_bootstrap_macro(res_a, res_b, mask, n_boot=1000, seed=0):
    rng = np.random.default_rng(seed)
    idx = np.where(mask)[0]
    n = len(idx)
    deltas = []
    for run in cache.runs:
        pa = res_a["runs"][run]["probabilities"]
        pb = res_b["runs"][run]["probabilities"]
        y = cache.hard
        boots = np.empty(n_boot)
        for b in range(n_boot):
            sample = rng.choice(idx, size=n, replace=True)
            ma = cache_lib.aucs(y[sample], pa[sample], cache.targets, strict=False)["macro_auc"]
            mb = cache_lib.aucs(y[sample], pb[sample], cache.targets, strict=False)["macro_auc"]
            boots[b] = ma - mb
        deltas.append(boots)
    stacked = np.mean(deltas, axis=0)
    lo, hi = np.percentile(stacked, [2.5, 97.5])
    return {"mean": float(stacked.mean()), "ci95": [float(lo), float(hi)], "seed": seed, "n_boot": n_boot}
