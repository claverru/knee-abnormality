import json
from pathlib import Path

import numpy as np
import polars as pl

ROOT = Path("/home/claudio/src/knee-abnormality")
RAW = ROOT / "outputs/analysis/type-contribution-20260917/raw"
OUT = ROOT / "outputs/analysis/type-contribution-20260917"
RUNS = ("ranked-filtered-seed42", "ranked-filtered-seed43", "ranked-filtered-seed44")
TYPE_NAMES = ("ax_t1pd", "ax_fluid", "cor_t1pd", "cor_fluid", "sag_t1pd", "sag_fluid")
TARGET_THRESHOLD = 0.5

runs = [np.load(RAW / f"{name}.npz", allow_pickle=True) for name in RUNS]
targets = list(runs[0]["targets"])
study = runs[0]["study"]
presence = runs[0]["presence"]
target_values = runs[0]["target_values"]
binary = target_values >= TARGET_THRESHOLD

group_masks = {
    "b_five_no_ax_t1pd": (presence == np.array([False, True, True, True, True, True])).all(1),
    "a_all_six": presence.all(1),
}

contributions = [run["contribution"] for run in runs]
logits = [run["logits"] for run in runs]
attention_share = [run["attention_share"] for run in runs]
token_counts = [run["token_counts"] for run in runs]

rows = []
efficiency_rows = []
count_rows = []
spread_rows = []

for group_name, mask in group_masks.items():
    idx = np.nonzero(mask)[0]
    n = len(idx)
    group_binary = binary[idx]
    for k, label in enumerate(targets):
        pos = group_binary[:, k]
        n_pos = int(pos.sum())
        n_neg = int((~pos).sum())
        count_rows.append(
            {
                "group": group_name,
                "label": label,
                "n_studies": n,
                "n_pos": n_pos,
                "n_neg": n_neg,
                "flag_low_positives": n_pos < 20,
            }
        )
        run_sep = []
        run_var = []
        run_att = []
        run_tok = []
        for r in range(3):
            C = contributions[r][idx][:, :, k]
            L = logits[r][idx, k]
            A = attention_share[r][idx][:, :, k]
            T = token_counts[r][idx]
            mean_pos = C[pos].mean(0)
            mean_neg = C[~pos].mean(0)
            D = mean_pos - mean_neg
            total_D = D.sum()
            sep_share = D / total_D if total_D != 0 else np.full_like(D, np.nan)
            var_logit = L.var()
            cov = ((C - C.mean(0)) * (L - L.mean())[:, None]).mean(0)
            var_share = cov / var_logit if var_logit != 0 else np.full_like(cov, np.nan)
            att_mean = A.mean(0)
            tok_share = T.mean(0) / T.mean(0).sum()
            run_sep.append(sep_share)
            run_var.append(var_share)
            run_att.append(att_mean)
            run_tok.append(tok_share)
        run_sep = np.stack(run_sep)
        run_var = np.stack(run_var)
        run_att = np.stack(run_att)
        run_tok = np.stack(run_tok)
        mean_sep = run_sep.mean(0)
        mean_var = run_var.mean(0)
        mean_att = run_att.mean(0)
        mean_tok = run_tok.mean(0)
        for t, type_name in enumerate(TYPE_NAMES):
            rows.append(
                {
                    "group": group_name,
                    "label": label,
                    "type": type_name,
                    "separation_share_pct": 100 * mean_sep[t],
                    "variance_share_pct": 100 * mean_var[t],
                    "attention_share_pct": 100 * mean_att[t],
                    "token_share_pct": 100 * mean_tok[t],
                    "efficiency": (mean_sep[t] / mean_tok[t]) if mean_tok[t] > 0 else np.nan,
                    "sep_share_min_pct": 100 * run_sep[:, t].min(),
                    "sep_share_max_pct": 100 * run_sep[:, t].max(),
                }
            )

table = pl.DataFrame(rows)
counts_table = pl.DataFrame(count_rows)

table.write_csv(OUT / "type_label_summary.csv")
counts_table.write_csv(OUT / "label_group_counts.csv")

macro_rows = []
for group_name in group_masks:
    sub = table.filter(pl.col("group") == group_name)
    macro = (
        sub.group_by("type", maintain_order=True)
        .agg(
            pl.col("separation_share_pct").mean(),
            pl.col("variance_share_pct").mean(),
            pl.col("attention_share_pct").mean(),
            pl.col("token_share_pct").mean(),
            pl.col("efficiency").mean(),
            pl.col("sep_share_min_pct").min(),
            pl.col("sep_share_max_pct").max(),
        )
        .with_columns(pl.lit(group_name).alias("group"), pl.lit("MACRO").alias("label"))
    )
    macro_rows.append(macro)
macro_table = pl.concat(macro_rows)
macro_table.write_csv(OUT / "type_label_summary_macro.csv")

print(table.sort(["group", "label", "type"]))
print(macro_table)
print(counts_table.filter(pl.col("flag_low_positives")))
