# ---------------- Step 3: curve fits ----------------


def log_model(n, a, w):
    return a + w * np.log(n)


def sat_exp_model(n, a, w, tau):
    return a + w * (1 - np.exp(-n / tau))


def power_model(n, a, w, b):
    return a + w * np.power(n, b)


def r2(y, yhat):
    ss_res = np.sum((y - yhat) ** 2)
    ss_tot = np.sum((y - np.mean(y)) ** 2)
    return 1 - ss_res / ss_tot if ss_tot > 0 else float("nan")


def fit_all(n_arr, y_arr):
    out = {}
    try:
        p, _ = curve_fit(log_model, n_arr, y_arr, p0=[y_arr[0], 0.01], maxfev=10000)
        out["log"] = {"params": p.tolist(), "r2": r2(y_arr, log_model(n_arr, *p))}
    except Exception as e:
        out["log"] = {"error": str(e)}
    try:
        p, _ = curve_fit(
            sat_exp_model,
            n_arr,
            y_arr,
            p0=[y_arr[0], y_arr[-1] - y_arr[0], 8.0],
            maxfev=20000,
            bounds=([0, -1, 0.1], [1, 1, 200]),
        )
        out["sat_exp"] = {"params": p.tolist(), "r2": r2(y_arr, sat_exp_model(n_arr, *p))}
    except Exception as e:
        out["sat_exp"] = {"error": str(e)}
    try:
        p, _ = curve_fit(
            power_model,
            n_arr,
            y_arr,
            p0=[y_arr[0], y_arr[-1] - y_arr[0], 0.3],
            maxfev=20000,
            bounds=([0, -1, -2], [1, 1, 2]),
        )
        out["power"] = {"params": p.tolist(), "r2": r2(y_arr, power_model(n_arr, *p))}
    except Exception as e:
        out["power"] = {"error": str(e)}
    return out

# ---------------- Step 5: greedy water-filling ----------------

def slot_curve(slot):
    sub = mean_df.filter((pl.col("background") == "default") & (pl.col("type") == slot)).sort("n")
    return sub["n"].to_numpy().astype(np.float64), sub["macro_auc_whole"].to_numpy()


curves = {slot: slot_curve(slot) for slot in SLOTS}


def slope_at(slot, n):
    xs, ys = curves[slot]
    if n >= xs[-1]:
        return 0.0
    idx = np.searchsorted(xs, n, side="right") - 1
    idx = max(idx, 0)
    x0, x1 = xs[idx], xs[idx + 1]
    y0, y1 = ys[idx], ys[idx + 1]
    if x1 == x0:
        return 0.0
    return (y1 - y0) / (x1 - x0)


def greedy_fill(present_idx, target_total):
    n = {i: 0 for i in present_idx}
    total = 0
    while total < target_total:
        gains = {i: slope_at(SLOTS[i], n[i]) for i in present_idx}
        best = max(gains, key=gains.get)
        n[best] += 1
        total += 1
    return n


def proportional_fill(present_idx, target_total, weights):
    w = np.array([max(weights[SLOTS[i]], 1e-9) for i in present_idx])
    w = w / w.sum()
    raw = w * target_total
    base = np.floor(raw).astype(int)
    remainder = target_total - base.sum()
    frac = raw - base
    order = np.argsort(-frac)
    for k in range(remainder):
        base[order[k]] += 1
    return {i: int(v) for i, v in zip(present_idx, base)}

def make_rule(lookup):
    default_tuple = tuple(int(v) for v in DEFAULT)

    def rule(avail, capacity):
        key = tuple(1 if a else 0 for a in avail)
        if key in lookup:
            return tuple(int(v) for v in lookup[key])
        return default_tuple

    return rule
