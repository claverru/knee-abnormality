SLOT_NAMES = cache_lib.SLOTS

repeated = []
for i in search_idx:
    study = studies[i]
    state = cache.states[study]
    for slot in range(6):
        pool = cache._pool(study, slot, frozenset(), frozenset())
        uids = sorted({v.uid for v in pool.candidates})
        if len(uids) < 2:
            continue
        ranking = [u for u in pool.ranking if u in uids]
        top_uid = ranking[0]
        other_uids = ranking[1:]
        sources = dict(state["slots"][slot]["sources"])
        sig_top = sources[top_uid].summary.signature
        sig_next = sources[ranking[1]].summary.signature
        if sig_top is None or sig_next is None:
            cls = "minimal_headers"
        elif sig_top == sig_next:
            cls = "same_protocol_retake"
        elif sig_top[-2:] == sig_next[-2:]:
            cls = "same_geometry_diff_protocol"
        else:
            cls = "different_geometry_protocol"
        repeated.append(dict(
            study=study, i=int(i), slot=slot, slot_name=SLOT_NAMES[slot],
            n_sources=len(ranking), cls=cls, top_uid=top_uid, other_uids=tuple(other_uids),
        ))

print("repeated slots (search half):", len(repeated), time.time() - t0, flush=True)

def default_ids(study, slot):
    n = DEFAULT[slot]
    return cache.selected_ids(study, slot, n)

def variant_ids(study, slot, source_uids, n):
    state = cache.states[study]
    kind = cache.series_types[slot]
    sources = tuple((u, s) for u, s in state["slots"][slot]["sources"] if u in source_uids)
    pool = build_pool(sources, kind.plane, state["reference"], cache.spacing_mm, cache.radius, set(), set())
    sel = select_triplets(pool, n, cache.sigma, cache.radius, study,
                           (kind.plane, kind.fluid_sensitive, kind.fat_suppression), None, 0)
    keys = tuple((t.uid, str(t.records[1].path)) for t in sel.triplets)
    return cache.ids(keys, study, slot)

def base_ids_for_all(counts_override=None):
    id_sets = []
    for i, study in enumerate(studies):
        parts = []
        for slot in range(6):
            n = DEFAULT[slot] if counts_override is None else counts_override.get((i, slot), DEFAULT[slot])
            if not cache.available[i, slot]:
                continue
            parts.append(cache.selected_ids(study, slot, n))
        id_sets.append(np.concatenate(parts) if parts else np.zeros(0, dtype=np.int64))
    return id_sets

for r in repeated:
    key = (r["study"], r["slot"])

    other_only_cache[key] = variant_ids(*key, set(r["other_uids"]), n)

def build_scenario_v2(condition):
    id_sets = []
    rep_by_i = {}
    for r in repeated:
        rep_by_i.setdefault(r["i"], []).append(r)
    for i, study in enumerate(studies):
        parts = []
        rslots = {r["slot"]: r for r in rep_by_i.get(i, [])}
        for slot in range(6):
            if not cache.available[i, slot]:
                continue
            if slot in rslots:
                r = rslots[slot]
                key = (r["study"], r["slot"])
                if condition == "default":
                    ids = default_split_cache[key]
                elif condition == "top_only":
                    ids = top_only_cache[key]
                elif condition == "other_only":
                    ids = other_only_cache[key]
                parts.append(ids)
            else:
                parts.append(cache.selected_ids(study, slot, DEFAULT[slot]))
        id_sets.append(np.concatenate(parts) if parts else np.zeros(0, dtype=np.int64))
    return id_sets

# Simpler paired bootstrap directly on macro auc mean over runs, resampling studies with replacement
def bootstrap_macro_auc(condition, mask, n_boot=500, seed=0):
    rng = np.random.default_rng(seed)
    idxs = np.nonzero(mask)[0]
    out = np.zeros(n_boot)
    for b in range(n_boot):
        samp = rng.choice(idxs, size=len(idxs), replace=True)
        vals = []
        for r in runs:
            p = probs(condition, r)
            hard = cache.hard
            from knee_abnormality.evaluation.metrics import aucs
            m = aucs(hard[samp], p[samp], cache.targets)
            vals.append(m["macro_auc"])
        out[b] = np.mean(vals)
    return out
