import sys
import json
from pathlib import Path

import numpy as np
import polars as pl

sys.path.insert(0, "/home/claudio/src/knee-abnormality/outputs/analysis/token-cache-20260917")
import cache_lib  # noqa: E402

OUT = Path("/home/claudio/src/knee-abnormality/outputs/analysis/allocation-neyman-20260917")
OUT.mkdir(parents=True, exist_ok=True)

cache = cache_lib.load()
SLOTS = cache_lib.SLOTS
DEFAULT = cache_lib.DEFAULT_COUNTS
n_studies = len(cache.studies)
n_labels = len(cache.targets)

search_mask = cache.half_mask("search")
search_idx = np.nonzero(search_mask)[0]

depth_arr = cache.triplets["depth"].to_numpy()
sigma = cache.sigma

patterns = cache.split["pattern"].to_numpy()

per_study_W = np.zeros((n_studies, 6, n_labels))
per_study_S = np.zeros((n_studies, 6, n_labels))
per_study_C = np.zeros((n_studies, 6, n_labels))

for run in cache.runs:
    lse_run = cache.lse[run]
    mixed_run = cache.mixed[run]
    for i in search_idx:
        study = cache.studies[i]
        cap_row = cache.capacity[i]
        avail_row = cache.available[i]
        slot_ids = []
        for slot in range(6):
            if not avail_row[slot]:
                continue
            n = min(DEFAULT[slot], int(cap_row[slot]))
            ids = cache.selected_ids(study, slot, n)
            slot_ids.append((slot, ids))
        all_ids = np.concatenate([ids for _, ids in slot_ids]) if slot_ids else np.zeros(0, dtype=np.int64)
        if len(all_ids) == 0:
            continue
        lse_vals = lse_run[all_ids]  # [T, 12]
        peak = lse_vals.max(0)
        w = np.exp(lse_vals - peak)
        w = w / w.sum(0)  # softmax weight per label, sums to 1 over all_ids
        mixed_vals = mixed_run[all_ids]
        offset = 0
        for slot, ids in slot_ids:
            m = len(ids)
            sl = slice(offset, offset + m)
            per_study_W[i, slot] += w[sl].sum(0)
            per_study_C[i, slot] += (w[sl] * mixed_vals[sl]).sum(0)
            offset += m

        for slot in range(6):
            if not avail_row[slot]:
                continue
            cap = int(cap_row[slot])
            ids_full = cache.selected_ids(study, slot, cap)
            if len(ids_full) < 2:
                continue
            depth_vals = depth_arr[ids_full]
            gw = np.exp(-(depth_vals ** 2) / (2 * sigma ** 2))
            gw = gw / gw.sum()
            mixed_full = mixed_run[ids_full]  # [m, 12]
            mu = (gw[:, None] * mixed_full).sum(0)
            var = (gw[:, None] * (mixed_full - mu) ** 2).sum(0)
            per_study_S[i, slot] += np.sqrt(np.maximum(var, 0))

per_study_W /= len(cache.runs)
per_study_C /= len(cache.runs)
per_study_S /= len(cache.runs)

np.savez(
    OUT / "per_study_stats.npz",
    W=per_study_W,
    S=per_study_S,
    C=per_study_C,
    search_idx=search_idx,
    available=cache.available,
    capacity=cache.capacity,
)

rows = []
for i in search_idx:
    rows.append((cache.studies[i], patterns[i]))
pattern_df = pl.DataFrame(rows, schema=["study", "pattern"], orient="row")
pattern_df.write_parquet(OUT / "study_patterns.parquet")

print("done", per_study_W.shape)
print("patterns in search half:", sorted(set(patterns[search_idx].tolist())))
