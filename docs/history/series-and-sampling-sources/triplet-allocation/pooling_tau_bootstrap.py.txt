import json

import numpy as np
import pooling_lib as pl_

cache = pl_.load_cache("main")
from knee_abnormality.evaluation.metrics import bootstrap_auc

full = np.ones(len(cache.studies), dtype=bool)
ids, per_study = cache.allocation_ids((12, 12, 10, 10, 10, 10), full)
prep = pl_.prepare_allocation(cache, ids, np.asarray(per_study))
truth = (np.asarray(cache.y) >= 0.5).astype(int)
rng = np.random.default_rng(20260917)
counts = rng.multinomial(len(cache.studies), np.full(len(cache.studies), 1 / len(cache.studies)), size=500)
deltas = []
for run in cache.runs:
    a = 1 / (1 + np.exp(-pl_.pool_from_prep(cache, run, prep, tau=0.75)))
    b = 1 / (1 + np.exp(-pl_.pool_from_prep(cache, run, prep, tau=1.0)))
    deltas.append((bootstrap_auc(truth, a, counts) - bootstrap_auc(truth, b, counts)) / truth.shape[1])
d = np.mean(deltas, axis=0)
out = {"mean": float(d.mean()), "ci95": [float(np.percentile(d, 2.5)), float(np.percentile(d, 97.5))], "resamples": 500, "seed": 20260917}
print(out)
json.dump(out, open("tau_bootstrap.json", "w"), indent=2)
