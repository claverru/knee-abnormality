import pickle
import re
from collections import Counter
from pathlib import Path

import numpy as np
import polars as pl

ROOT = Path("/home/claudio/src/knee-abnormality")
OUT = Path(__file__).resolve().parent
entries = pickle.load(open(OUT / "entries.pkl", "rb"))
headers = pl.read_parquet(OUT / "repeated_headers.parquet")
acq = pl.read_parquet(
    ROOT / "outputs/analysis/input-resolution-research-20260915/acquisition/series_acquisition.parquet"
).filter(pl.col("split") == "train")
acq_cols = [
    "SeriesInstanceUID",
    "acq_res_best",
    "acq_res_worst",
    "acq_estimate_quality",
    "ps_min",
    "ps_max",
    "fov_min",
    "fov_max",
    "SliceThickness",
    "SpacingBetweenSlices",
    "spacing",
    "gap",
    "is_3d",
    "vendor",
    "n_images",
    "depth_extent",
    "NumberOfAverages",
    "PixelBandwidth",
    "ParallelReductionFactorInPlane",
    "acq_freq",
    "acq_phase",
    "interp_min",
]
acq = acq.select(acq_cols).rename({"SliceThickness": "thickness", "NumberOfAverages": "nex_acq", "PixelBandwidth": "bw_acq"})
table = headers.join(acq, on="SeriesInstanceUID", how="left")
by_uid = {r["SeriesInstanceUID"]: r for r in table.iter_rows(named=True)}


def f(x):
    try:
        return float(x)
    except (TypeError, ValueError):
        return None


SIDE = re.compile(r"(^|[^a-z])(rt|lt|right|left|r|l|izq|der)([^a-z]|$)")


def side(desc):
    if not desc:
        return None
    d = desc.lower()
    tokens = set()
    for m in SIDE.finditer(d):
        t = m.group(2)
        tokens.add("R" if t in {"rt", "right", "r", "der"} else "L")
    return "".join(sorted(tokens)) or None


def norm_desc(desc):
    if not desc or desc.lower().startswith("dummyseriesdesc"):
        return None
    return re.sub(r"[^a-z0-9]", "", desc.lower())


rows = []
for e in entries:
    series = []
    for s in e["series"]:
        h = by_uid[s["uid"]]
        series.append({**s, **h})
    series.sort(key=lambda s: (f(s["SeriesNumber"]) or 0, s["uid"]))
    row = {
        "study": e["study"],
        "kind": e["kind"],
        "n": len(series),
        "cov": e["cov"],
        "physical": e["physical"],
        "vendor": series[0]["vendor"],
        "manufacturer": series[0]["Manufacturer"],
    }
    same_geom = all(
        len(s["depths"]) == len(series[0]["depths"])
        and len(s["depths"]) > 0
        and np.allclose(s["depths"], series[0]["depths"], atol=0.1)
        for s in series[1:]
    )
    tr = [f(s["RepetitionTime"]) for s in series]
    te = [f(s["EchoTime"]) for s in series]
    have = all(v is not None for v in tr + te)
    same_tr = have and max(tr) / max(min(tr), 1e-6) <= 1.01
    same_te = have and max(te) - min(te) <= 1.0
    descs = [norm_desc(s["SeriesDescription"]) for s in series]
    same_desc = all(d is not None for d in descs) and len(set(descs)) == 1
    sig = [
        (
            round(f(s["RepetitionTime"]) or -1),
            round(f(s["EchoTime"]) or -1),
            s["EchoTrainLength"],
            s["FlipAngle"],
            s["AcquisitionMatrix"],
            s["MRAcquisitionType"],
            s["SequenceVariant"],
            s["ScanOptions"],
        )
        for s in series
    ]
    same_sig = have and len(set(sig)) == 1
    same_image_type = len({s["ImageType"] for s in series}) == 1
    same_pixels = len({(s["Rows"], s["Columns"], s["PixelSpacing"]) for s in series}) == 1
    types3d = {s["MRAcquisitionType"] for s in series}
    sides = [side(s["SeriesDescription"]) for s in series]
    if not have:
        cls = "missing_tr_te"
    elif same_geom and same_tr and same_te:
        cls = "same_geom_same_protocol"
    elif same_geom and same_tr:
        cls = "same_geom_multi_echo"
    elif same_geom:
        cls = "same_geom_diff_protocol"
    elif same_sig:
        cls = "retake_same_protocol"
    elif same_tr and same_te:
        cls = "diff_geom_same_tr_te"
    else:
        cls = "diff_geom_diff_protocol"
    row.update(
        cls=cls,
        same_geom=same_geom,
        same_tr=same_tr,
        same_te=same_te,
        same_desc=same_desc,
        same_sig=same_sig,
        same_image_type=same_image_type,
        same_pixels=same_pixels,
        mixed_3d=len(types3d - {None}) > 1,
        any_3d="3D" in types3d,
        side_conflict=len({x for x in sides if x}) > 1,
        sides=sides,
        descs=[s["SeriesDescription"] for s in series],
        series_numbers=[s["SeriesNumber"] for s in series],
        acq_numbers=[s["AcquisitionNumber"] for s in series],
        image_types=[s["ImageType"] for s in series],
        te=te,
        tr=tr,
        acq_best=[s["acq_res_best"] for s in series],
        acq_worst=[s["acq_res_worst"] for s in series],
        acq_quality=[s["acq_estimate_quality"] for s in series],
        ps=[s["ps_min"] for s in series],
        rows_=[s["Rows"] for s in series],
        thickness=[s["thickness"] for s in series],
        spacing=[s["median_gap"] for s in series],
        n_records=[s["n_records"] for s in series],
        n_candidates=[s["n_candidates"] for s in series],
        fov=[s["fov_min"] for s in series],
        nex=[f(s["NumberOfAverages"]) for s in series],
        etl=[f(s["EchoTrainLength"]) for s in series],
        bw=[f(s["PixelBandwidth"]) for s in series],
        mr3d=[s["MRAcquisitionType"] for s in series],
        obliq=[s["obliquity_deg"] for s in series],
        support=[
            (min(a for a, _ in s["supports"]), max(b for _, b in s["supports"])) if s["supports"] else None
            for s in series
        ],
        pool_status=[s["pool_status"] for s in series],
        uids=[s["uid"] for s in series],
        targets=e.get("targets"),
        eligible=e.get("eligible"),
        nearest=e.get("nearest"),
        perm_outcomes=e.get("perm_outcomes"),
        trace=e.get("trace_seed42"),
        pool_sources=e["pool_sources"],
        count=e.get("count"),
    )
    rows.append(row)

pickle.dump(rows, open(OUT / "slots.pkl", "wb"))
frame = pl.DataFrame([{k: v for k, v in r.items() if not isinstance(v, (list, dict, tuple))} | {"kind": str(r["kind"])} for r in rows])
print(frame.shape)
print(frame["cls"].value_counts().sort("count", descending=True))
print(frame.group_by("cls").agg(pl.len(), pl.col("same_desc").mean(), pl.col("same_image_type").mean(), pl.col("same_pixels").mean(), pl.col("mixed_3d").sum(), pl.col("any_3d").sum(), pl.col("side_conflict").sum()).sort("len", descending=True))
print(frame.group_by("vendor", "cls").len().sort("vendor", "len", descending=[False, True]))
print("side conflicts", frame["side_conflict"].sum())
for r in rows:
    if r["side_conflict"]:
        print("SIDE", r["kind"], r["descs"], r["sides"], r["cls"])
