import pickle
from collections import Counter
from pathlib import Path

import numpy as np

OUT = Path(__file__).resolve().parent
entries = pickle.load(open(OUT / "entries.pkl", "rb"))


def frac(xs):
    xs = list(xs)
    return f"{sum(xs)}/{len(xs)} ({np.mean(xs):.3f})" if xs else "0/0"


print("== geometry sharing ==")
mixed_share = []
for e in entries:
    if e.get("n_groups"):
        mixed_share.append(e["mixed_groups"] / e["n_groups"])
mixed_share = np.asarray(mixed_share)
print("slots with any mixed group", frac(mixed_share > 0))
print("mixed share quantiles", np.quantile(mixed_share, [0.1, 0.25, 0.5, 0.75, 0.9]).round(3))

print("== nearest-slice distance differences at targets ==")
equal_rows = 0
rows = 0
diffs = []
slot_all_equal = []
for e in entries:
    if not e.get("nearest") or len(e["pool_sources"]) < 2:
        continue
    eq = []
    for row, elig in zip(e["nearest"], e["eligible"]):
        vals = [row[u] for u in elig if row[u] is not None]
        if len(vals) < 2:
            continue
        rows += 1
        spread = max(vals) - min(vals)
        diffs.append(spread)
        eq.append(spread <= 0.1)
    slot_all_equal.append(all(eq) if eq else True)
diffs = np.asarray(diffs)
print("multi-eligible rows", rows, "spread<=0.1mm", frac(diffs <= 0.1), "spread<=0.5", frac(diffs <= 0.5), "spread>1mm", frac(diffs > 1))
print("spread quantiles", np.quantile(diffs, [0.25, 0.5, 0.75, 0.9, 0.99]).round(2))
print("slots where all multi-eligible rows have equal nearest distance (<=0.1mm)", frac(slot_all_equal))

print("== phase depth error for 2-source full coverage ==")
better = []
gain = []
for e in entries:
    if e.get("cov") != "all_cover_all_targets" or len(e["pool_sources"]) != 2:
        continue
    po = e["perm_outcomes"]
    errs = {}
    for order, out in po.items():
        pass
    a, b = e["pool_sources"]
    targets = np.asarray(e["targets"])
    order_idx = np.argsort(targets)
    tot = {}
    for first in (a, b):
        other = b if first == a else a
        err = 0.0
        for k, row in enumerate(order_idx):
            src = first if k % 2 == 0 else other
            err += e["nearest"][row][src] ** 2
        tot[first] = err
    better.append(abs(tot[a] - tot[b]) > 1e-6)
    gain.append(abs(np.sqrt(tot[a] / len(targets)) - np.sqrt(tot[b] / len(targets))))
print("2-source full-coverage slots:", len(better), "phase changes nearest-distance error:", frac(better))
print("rms error difference quantiles (mm)", np.quantile(gain, [0.25, 0.5, 0.75, 0.9]).round(3))

print("== per-pair series differences (2-source slots) ==")
keys = ["n_records", "median_gap", "rows", "columns", "obliquity_deg", "n_candidates"]
for key in keys:
    same = []
    for e in entries:
        if len(e["series"]) != 2:
            continue
        x, y = (s[key] for s in e["series"])
        if x is None or y is None:
            continue
        same.append(abs(x - y) <= (0.05 if isinstance(x, float) else 0))
    print(key, "equal:", frac(same))
same = []
for e in entries:
    if len(e["series"]) != 2:
        continue
    x, y = (s["pixel_spacing"] for s in e["series"])
    if x and y:
        same.append(np.allclose(x, y, atol=1e-3))
print("pixel spacing equal:", frac(same))
same = []
for e in entries:
    if len(e["series"]) != 2:
        continue
    x, y = (s["frame_uid"] for s in e["series"])
    same.append(x == y)
print("frame uid equal:", frac(same))
same = []
for e in entries:
    if len(e["series"]) != 2:
        continue
    x, y = (s["supports"] for s in e["series"])
    if x and y:
        lo = [min(a for a, _ in s) for s in (x, y)]
        hi = [max(b for _, b in s) for s in (x, y)]
        same.append(abs(lo[0] - lo[1]) <= 0.5 and abs(hi[0] - hi[1]) <= 0.5)
print("support extent equal within 0.5mm:", frac(same))
same = []
for e in entries:
    if len(e["series"]) != 2:
        continue
    x, y = (s["depths"] for s in e["series"])
    if len(x) == len(y) and len(x):
        same.append(np.allclose(x, y, atol=0.1))
    else:
        same.append(False)
print("identical candidate depth lists (0.1mm):", frac(same))
g = Counter()
for e in entries:
    if len(e["series"]) != 2:
        continue
    a, b = e["series"]
    if a["median_gap"] and b["median_gap"]:
        r = max(a["median_gap"], b["median_gap"]) / min(a["median_gap"], b["median_gap"])
        g["gap ratio>1.2"] += r > 1.2
        g["gap ratio>1.5"] += r > 1.5
        g["n"] += 1
print(g)
print("obliquity quantiles", np.quantile([s["obliquity_deg"] for e in entries for s in e["series"] if s["obliquity_deg"] is not None], [0.5, 0.9, 0.99]).round(2))
