def value(ds, key):
    v = ds.get(key)
    if v is None:
        return None
    if isinstance(v, pydicom.multival.MultiValue):
        return "|".join(str(x) for x in v)
    return str(v)


def read(item):
    study, series, plane, fluid, fat = item
    directory = ROOT / "data/raw/train_series" / study / series
    files = sorted(directory.glob("*.dcm"))
    ds = pydicom.dcmread(files[0], stop_before_pixels=True)
    row = {
        "StudyInstanceUID": study,
        "SeriesInstanceUID": series,
        "plane": plane,
        "contrast": f"{fluid}{fat}",
        "files": len(files),
        "keywords": [e.keyword or str(e.tag) for e in ds],
        "sequences": [e.keyword for e in ds if e.VR == "SQ"],
    }
    for key in FIELDS:
        row[key] = value(ds, key)
    return row


def main():
    frame = pl.read_csv(ROOT / "data/raw/train_series.csv")
    groups = frame.group_by("StudyInstanceUID", *KEYS).agg(pl.len().alias("n"), pl.col("SeriesInstanceUID"))
    repeated = groups.filter(pl.col("n") >= 2).explode("SeriesInstanceUID")
    items = list(
        repeated.select("StudyInstanceUID", "SeriesInstanceUID", *KEYS).iter_rows()
    )
    print(len(items), repeated["StudyInstanceUID"].n_unique())
    with ThreadPoolExecutor(24) as pool:
        rows = list(pool.map(read, items))
    keywords = Counter(k for r in rows for k in set(r["keywords"]))
    sequences = Counter(k for r in rows for k in set(r["sequences"]))
    for r in rows:
        r.pop("keywords")
        r.pop("sequences")
    pl.DataFrame(rows, infer_schema_length=None).write_parquet(OUT / "repeated_headers.parquet")
    json.dump(
        {"series": len(rows), "keywords": dict(keywords.most_common()), "sequences": dict(sequences)},
        open(OUT / "tag_inventory.json", "w"),
        indent=1,
    )


if __name__ == "__main__":
    main()
