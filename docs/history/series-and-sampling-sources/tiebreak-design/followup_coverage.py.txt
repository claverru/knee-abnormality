ROOT = m.ROOT
CENSUS = ROOT / "outputs/analysis/series-filtering-20260917"
FOLLOW = CENSUS / "followup"
HALF = 70.0
GAP = 0.20
VARIANT = ("reformat", "superseded", "coverage", "acq_worst", "uid")
ROBUST = {}


def crop_coverage(indexed, point):
    if point is None or not indexed.records:
        return None, None
    middle = indexed.records[len(indexed.records) // 2]
    if middle.pose is None:
        return None, None
    layout = middle.pose.layout
    orientation = np.asarray(layout.orientation, dtype=float).reshape(2, 3)
    delta = np.asarray(point) - np.asarray(middle.pose.position, dtype=float)
    u, v = float(orientation[0] @ delta), float(orientation[1] @ delta)
    width = (layout.columns - 1) * layout.spacing[1]
    height = (layout.rows - 1) * layout.spacing[0]
    ox = max(0.0, min(u + HALF, width) - max(u - HALF, 0.0))
    oy = max(0.0, min(v + HALF, height) - max(v - HALF, 0.0))
    fill = min(layout.columns * layout.spacing[1], 2 * HALF) * min(layout.rows * layout.spacing[0], 2 * HALF)
    return ox * oy / (2 * HALF) ** 2, fill / (2 * HALF) ** 2


def mass(intervals, sigma):
    if not len(intervals):
        return 0.0
    return float(sum(np.exp(interval_log_mass(lo / sigma, hi / sigma)) for lo, hi in intervals))


def deciding(tied, values, criteria):
    for c in criteria:
        if len({values[s][c] for s in tied}) > 1:
            return c
    return None


def process(study_uid):
    config = m.base.load_config(ROOT / "configs/training.yaml")
    dataset = m.base.WORKER["dataset"]
    study = dataset.series_index[study_uid]
    dataset._index_study_geometry(study)
    reference = dataset._study_references[study_uid]
    point = reference.point if reference.point is not None else ROBUST.get(study_uid)
    out = []
    for slot, values in enumerate(study.slots):
        if len(values) < 2:
            continue
        kc = config.data.series_types[slot]
        kind = (kc.plane, kc.fluid_sensitive, kc.fat_suppression)
        sources = []
        coverage, fill = {}, {}
        for c in sorted(values, key=lambda c: c.uid):
            dataset._records(c)
            indexed = dataset._slice_index[c.directory]
            sources.append((c.uid, indexed))
            coverage[c.uid], fill[c.uid] = crop_coverage(indexed, point)
        pool = build_pool(tuple(sources), kc.plane, reference, set(), set())
        budget = kc.max_images * config.data.validation_image_multiplier // 3
        sigma = config.data.sampling_sigma_mm
        radius = config.data.image_size * config.data.spacing_mm / 2
        pool_sources = sorted({v.uid for v in pool.candidates})
        if len(pool_sources) < 2:
            continue
        depth = {}
        if pool.physical:
            total = mass(union_support(pool.candidates), sigma)
            for s in pool_sources:
                own = tuple(v for v in pool.candidates if v.uid == s)
                depth[s] = mass(union_support(own), sigma) / total
        values_q = m.criterion_values(pool_sources)
        known = all(coverage[s] is not None for s in pool_sources)
        values_v = {s: {**values_q[s], "coverage": -round(coverage[s], 1) if known else 0} for s in pool_sources}
        q_order = m.quality_order(pool_sources)
        v_order = sorted(pool_sources, key=lambda s: tuple(values_v[s][c] for c in VARIANT))
        rules = {
            "proposed": m.select_custom(pool, budget, sigma, radius, q_order, tol=1e-9),
            "coverage_first": m.select_custom(pool, budget, sigma, radius, v_order, tol=1e-9),
            "seed42": m.select_custom(pool, budget, sigma, radius, m.base.seeded_order(42, study_uid, kind, pool_sources)),
        }
        out.append(
            {
                "study": study_uid,
                "slot": slot,
                "kind": kind,
                "physical": pool.physical,
                "reference_accepted": reference.point is not None,
                "sources": pool_sources,
                "coverage": {s: coverage[s] for s in pool_sources},
                "fill": {s: fill[s] for s in pool_sources},
                "depth": depth,
                "q_order": q_order,
                "v_order": v_order,
                "values_v": values_v,
                "rules": {k: (m.describe(sel, pool.physical), dec) for k, (sel, dec) in rules.items()},
            }
        )
    return study_uid, out


def init(config_path, robust):
    m.init(config_path)
    ROBUST.update(robust)


def summarize(slots, flags):
    small = set(flags.filter(pl.col("f_small_fov_axial"))["SeriesInstanceUID"])
    final = pickle.load(open(HERE / "rules_final.pkl", "rb"))
    final_keys = {(e["study"], e["slot"]): e["rules"]["quality"][0]["key"] for r in final.values() for e in r}
    mismatch = sum(final_keys.get((e["study"], e["slot"])) != e["rules"]["proposed"][0]["key"] for e in slots)
    known = [e for e in slots if all(e["coverage"][s] is not None for s in e["sources"])]
    report = {
        "slots": len(slots),
        "slots_with_coverage": len(known),
        "slots_without_centre": len(slots) - len(known),
        "proposed_vs_rules_final_slot_mismatch": mismatch,
        "gap_pp": GAP * 100,
    }
    cov_values = [e["coverage"][s] for e in known for s in e["sources"]]
    report["series_coverage_mean"] = float(np.mean(cov_values))
    report["series_coverage_lt_0.8"] = int(sum(c < 0.8 for c in cov_values))
    report["series_small_fov_axial_coverage_mean"] = float(np.mean([e["coverage"][s] for e in known for s in e["sources"] if s in small]))
    for rule, order_key in (("proposed", "q_order"), ("coverage_first", "v_order"), ("seed42", None)):
        low_slots = low_slots_small = 0
        depth_low_slots = 0
        tie_rows = low_rows = low_rows_small = 0
        won, triplet_cov, triplet_depth, won_depth = [], [], [], []
        criteria = Counter()
        for e in known:
            cov = e["coverage"]
            sel, decisions = e["rules"][rule]
            order = e[order_key] if order_key else None
            if order:
                first = order[0]
                if cov[first] <= max(cov[s] for s in e["sources"]) - GAP:
                    low_slots += 1
                    low_slots_small += first in small
                if e["depth"] and e["depth"][first] <= max(e["depth"].values()) - GAP:
                    depth_low_slots += 1
            triplet_cov.extend(cov[u] for u in sel["uids"])
            if e["depth"]:
                triplet_depth.extend(e["depth"][u] for u in sel["uids"])
            for _, tied, chosen in decisions:
                if len(tied) < 2:
                    continue
                tie_rows += 1
                won.append(cov[chosen])
                if e["depth"]:
                    won_depth.append(e["depth"][chosen])
                if cov[chosen] <= max(cov[s] for s in tied) - GAP:
                    low_rows += 1
                    low_rows_small += chosen in small
                if rule == "coverage_first":
                    criteria[deciding(tied, e["values_v"], VARIANT)] += 1
        entry = {
            "tie_rows": tie_rows,
            "tie_rows_low_coverage_winner": low_rows,
            "tie_rows_low_coverage_winner_small_fov_axial": low_rows_small,
            "tie_won_mean_crop_coverage": float(np.mean(won)),
            "tie_won_mean_depth_coverage_physical": float(np.mean(won_depth)),
            "repeated_slot_triplets": len(triplet_cov),
            "triplet_mean_crop_coverage": float(np.mean(triplet_cov)),
            "triplet_mean_depth_coverage_physical": float(np.mean(triplet_depth)),
        }
        if order_key:
            entry.update(
                slots_low_coverage_first=low_slots,
                slots_low_coverage_first_small_fov_axial=low_slots_small,
                slots_low_depth_coverage_first=depth_low_slots,
            )
        if criteria:
            total = sum(criteria.values())
            entry["deciding_criteria"] = {k: {"rows": v, "share": round(v / total, 4)} for k, v in criteria.most_common()}
        report[rule] = entry
    report["slots_coverage_differs_by_gap"] = sum(max(e["coverage"].values()) - min(e["coverage"].values()) >= GAP for e in known)
    report["slots_coverage_differs_rounded"] = sum(len({v["coverage"] for v in e["values_v"].values()}) > 1 for e in known)
    report["slots_small_fov_axial"] = sum(any(s in small for s in e["sources"]) for e in known)
    agree = lambda a, b: sum(e["rules"][a][0]["key"] != e["rules"][b][0]["key"] for e in slots)
    report["slots_differ_proposed_vs_coverage_first"] = agree("proposed", "coverage_first")
    report["slots_differ_proposed_vs_seed42"] = agree("proposed", "seed42")
    return report


def main():
    config_path = ROOT / "configs/training.yaml"
    data = pickle.load(open(HERE / "sensitivity.pkl", "rb"))
    robust = {
        r["StudyInstanceUID"]: tuple(r["robust_point"])
        for r in pl.read_parquet(CENSUS / "robust_studies.parquet").filter(pl.col("robust_point").is_not_null()).iter_rows(named=True)
    }
    start = time.time()
    results = {}
    with get_context("spawn").Pool(16, initializer=init, initargs=(config_path, robust)) as pool:
        for study, out in pool.imap_unordered(process, data["repeated"], chunksize=2):
            results[study] = out
    slots = [e for r in results.values() for e in r]
    pickle.dump(results, open(HERE / "followup_coverage.pkl", "wb"))
    flags = pl.read_parquet(CENSUS / "series_flags.parquet")
    report = summarize(slots, flags)
    FOLLOW.mkdir(parents=True, exist_ok=True)
    (FOLLOW / "tiebreak_coverage.json").write_text(json.dumps(report, indent=1) + "\n")
    small = set(flags.filter(pl.col("f_small_fov_axial"))["SeriesInstanceUID"])
    rows = [
        {
            "StudyInstanceUID": e["study"],
            "slot": e["slot"],
            "SeriesInstanceUID": s,
            "physical": e["physical"],
            "reference_accepted": e["reference_accepted"],
            "crop_coverage": e["coverage"][s],
            "image_centred_fill": e["fill"][s],
            "depth_coverage": e["depth"].get(s),
            "proposed_rank": e["q_order"].index(s),
            "coverage_first_rank": e["v_order"].index(s),
            "small_fov_axial": s in small,
        }
        for e in slots
        for s in e["sources"]
    ]
    pl.DataFrame(rows).write_parquet(FOLLOW / "tiebreak_coverage_series.parquet")
    print(json.dumps(report, indent=1))
    print("done", round(time.time() - start, 1))


if __name__ == "__main__":
    main()
