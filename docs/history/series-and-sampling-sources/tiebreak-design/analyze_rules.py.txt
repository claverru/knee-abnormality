import itertools
import pickle
from collections import Counter
from pathlib import Path

import numpy as np

OUT = Path(__file__).resolve().parent
import os
results = pickle.load(open(OUT / os.environ.get("RULES_OUT", "rules.pkl"), "rb"))
slots = [e for r in results.values() for e in r]
rules = list(slots[0]["rules"])
print("slots", len(slots), "rules", rules)


def overlap(a, b):
    return sum((Counter(a) & Counter(b)).values())


print("\n== triplet agreement between rules (share of triplets identical) ==")
for a, b in itertools.combinations(rules, 2):
    same = tot = slot_diff = 0
    studies = set()
    for e in slots:
        ka, kb = e["rules"][a][0]["key"], e["rules"][b][0]["key"]
        same += overlap(ka, kb)
        tot += max(len(ka), len(kb))
        if ka != kb:
            slot_diff += 1
            studies.add(e["study"])
    print(f"{a:>22} vs {b:<22} triplets differ {1 - same / tot:.3f}  slots differ {slot_diff}/{len(slots)}  studies differ {len(studies)}")

print("\n== deciding criterion at tie rows (quality rule) ==")
dec = Counter(d for e in slots for d in e["deciding"])
total = sum(dec.values())
for k, v in dec.most_common():
    print(f"{k}: {v} ({v / total:.3f})")
first = Counter(e["deciding"][0] if e["deciding"] else "no_tie" for e in slots)
print("per slot (first tie row):", first.most_common())
slot_dec = Counter()
for e in slots:
    for c in set(e["deciding"]):
        slot_dec[c] += 1
print("slots in which each criterion decides at least one tie:", slot_dec.most_common())
cd = [x for e in slots for x in e["closest_deciding"]]
print("closest-slice decides tie rows (rounded 0.1mm):", sum(cd), "/", len(cd))
print("slots where closest decides any tie:", sum(any(e["closest_deciding"]) for e in slots))

print("\n== deciding criterion by vendor ==")
by = {}
for e in slots:
    if e["deciding"]:
        by.setdefault(e["vendor"], Counter())[e["deciding"][0]] += 1
for v, c in by.items():
    print(v, dict(c))

print("\n== depth error (physical slots, mm) ==")
for r in rules:
    errs = [x for e in slots if e["physical"] for x in e["rules"][r][0]["abs_error"]]
    print(f"{r:>22} mean {np.mean(errs):.3f} rms {np.sqrt(np.mean(np.square(errs))):.3f} p95 {np.quantile(errs, 0.95):.3f} n {len(errs)}")

print("\n== quality proxies of realized triplets ==")
import simulate_rules as m

m.INFO.update(m.load_info())
for r in rules:
    worst, best, thick, reformat, superseded, n3d = [], [], [], 0, 0, 0
    n = 0
    for e in slots:
        for uid in e["rules"][r][0]["uids"]:
            i = m.INFO[uid]
            n += 1
            if i["acq_worst"] is not None:
                worst.append(max(i["acq_worst"], m.SPACING))
            if i["acq_best"] is not None:
                best.append(max(i["acq_best"], m.SPACING))
            if i["thickness"] is not None:
                thick.append(i["thickness"])
            reformat += i["reformat"]
            superseded += e["values"][uid]["superseded"]
            n3d += i["is_3d"]
    print(f"{r:>22} n {n} clipped acq worst {np.mean(worst):.4f} best {np.mean(best):.4f} thickness {np.mean(thick):.3f} reformat {reformat} superseded {superseded} 3D {n3d}")

print("\n== criterion availability and difference in slots ==")
for c in m.CRITERIA[:-1]:
    differs = sum(len({e["values"][s][c] for s in e["sources"]}) > 1 for e in slots)
    print(c, "differs between sources in", differs, "slots")
unknown = Counter()
for e in slots:
    for c in ("acq_worst", "thickness"):
        if all(e["values"][s][c] == 0 for s in e["sources"]):
            unknown[c] += 1
print("criterion neutralized (unknown for some source)", unknown)
sup = [e for e in slots if any(e["values"][s]["superseded"] for s in e["sources"])]
print("slots with a superseded same-protocol series:", len(sup))
ref = [e for e in slots if any(e["values"][s]["reformat"] for s in e["sources"])]
print("slots with a reformat:", len(ref))
for e in ref:
    print("   ", e["kind"], [m.INFO[s]["desc"] for s in e["sources"]], [e["values"][s]["reformat"] for s in e["sources"]], "reformat triplets seed42/quality:", sum(m.INFO[u]["reformat"] for u in e["rules"]["seed42"][0]["uids"]), sum(m.INFO[u]["reformat"] for u in e["rules"]["quality"][0]["uids"]), "of", len(e["rules"]["quality"][0]["uids"]))
